# 3) 유틸 함수 (야후 규격 변환, 청크, 유효성 검증)
# ==========================================
def clean_to_yahoo_symbol(sym: str) -> str:
    """
    야후 호환 심볼로 정규화:
    - 대문자화
    - '.' -> '-' 변환 (예: BRK.B -> BRK-B, BF.B -> BF-B)
    """
    s = sym.strip().upper().replace('.', '-')
    alias = {"BRK.B": "BRK-B", "BF.B": "BF-B"}
    return alias.get(sym, s)

def chunk(lst, n=50):
    for i in range(0, len(lst), n):
        yield lst[i:i+n]

def validate_with_yfinance(tickers, period="5d", min_ok_days=1, chunk_size=50, sleep_sec=0.05):
    """
    yfinance로 최근 period 구간에서 종가가 min_ok_days 이상 존재하는 심볼만 통과.
    입력 순서를 보존해서 반환.
    """
    valid = []
    for batch in chunk(tickers, chunk_size):
        try:
            px = yf.download(batch, period=period, auto_adjust=False, progress=False)["Close"]
            if isinstance(px, pd.Series):  # 단일 티커 대응
                px = px.to_frame()
            for col in px.columns:
                if px[col].dropna().shape[0] >= min_ok_days:
                    valid.append(col)
        except Exception:
            pass
        time.sleep(sleep_sec)  # 과다 호출 방지
    valid_set = set(valid)
    return [t for t in tickers if t in valid_set]

# ==========================================
# 4) 한경 S&P500 페이지에서 티커 추출 (한글+티커 패턴만)
#    URL: https://www.hankyung.com/globalmarket/usa-stock-sp500
#    - 메뉴/배너 노이즈 제거 위해 '한글 포함 링크 텍스트'에서만 대문자 티커 패턴 검색
# ==========================================
HK_URL = "https://www.hankyung.com/globalmarket/usa-stock-sp500"
headers = {"User-Agent": "Mozilla/5.0 (compatible; ColabScraper/1.0)"}
html = requests.get(HK_URL, headers=headers, timeout=30).text
soup = BeautifulSoup(html, "lxml")

tickers_raw = []

# (1) '엔비디아(NVDA)' 같은 괄호 패턴 우선 추출
pattern_paren = re.compile(r"\(([A-Z]{1,6})\)")
for node in soup.find_all(string=True):
    for m in pattern_paren.finditer(node):
        tickers_raw.append(m.group(1))

# (2) 보조: 한글이 섞인 링크 텍스트에서만 대문자 1~5자(또는 BRK-B 같은 하이픈 포함) 추출
ticker_pat  = re.compile(r"\b[A-Z]{1,5}(?:-[A-Z]{1,2})?\b")  # AAPL, GE, BRK-B 등
has_hangul  = re.compile(r"[가-힣]")
blacklist   = {
    "USA","US","USD","S","P","SP","SNP","CPI","PPI","EPS","PE","YOY","QOQ",
    "ETF","AI","EV","GDP","FOMC","FED","ECB","BOJ","SEC","IRS","NY","EU",
    "BOOK","POLL","MAP","KLA","DR","WR","SBA","CH","JM","AO","JB","BF","RSS","AT","TESAT"  # 한경 고유 노이즈
}
for a in soup.find_all("a"):
    text = a.get_text(" ", strip=True)
    if not text or len(text) > 300:
        continue
    if not has_hangul.search(text):
        continue
    cands = ticker_pat.findall(text)
    for c in cands:
        if c not in blacklist and c.isalpha() or "-" in c:
            tickers_raw.append(c)

# 정규화(야후 규격) + 기본 필터 + 순서 보존 중복 제거
filtered = []
for s in (clean_to_yahoo_symbol(x) for x in tickers_raw):
    if 1 <= len(s) <= 6 and re.fullmatch(r"[A-Z\-]+", s):
        if s not in filtered:
            filtered.append(s)

print(f"[한경] 후보 추출: {len(filtered)}개")

# ==========================================
# 5) yfinance로 유효 심볼만 최종 확정
# ==========================================
universe = validate_with_yfinance(filtered, period="5d", min_ok_days=1, chunk_size=50, sleep_sec=0.03)
# 안전장치: S&P500은 500개. 혹시 500 초과면 컷, 부족하면 그대로 진행(한경 페이지 업데이트/노이즈 영향 가능)
if len(universe) > 500:
    universe = universe[:500]
print(f"[야후 유효] 최종: {len(universe)}개")

# 저장(선택)
CSV_PATH  = "/content/sp500_from_hankyung_yahoo.csv"
XLSX_PATH = "/content/sp500_from_hankyung_yahoo.xlsx"

# 6) RS(60/120/250 + 평균) 계산
# ==========================================
spy_rs = yf.download("SPY", period="3y", auto_adjust=False, progress=False)["Close"].dropna()

def compute_mansfield_rs(price_series, benchmark_series, ma_period):
    rel = price_series / benchmark_series
    ma  = rel.rolling(window=ma_period, min_periods=ma_period).mean()
    return ((rel / ma) - 1) * 100

def normalize_to_100_scale(x, scale=12):
    return 100 * (1 / (1 + np.exp(-x / scale)))

RS_WINDOWS = [60, 120, 250]
mansfield_rows, all_rows_for_fallback = [], []

for ticker in universe:
    try:
        px = yf.download(ticker, period="3y", auto_adjust=False, progress=False)["Close"].dropna()
        aligned = pd.concat([px, spy_rs], axis=1, join="inner")
        aligned.columns = ["etf", "spy"]
        if aligned.empty or len(aligned) < min(RS_WINDOWS):
            continue

        latest = {}
        for win in RS_WINDOWS:
            rs = compute_mansfield_rs(aligned["etf"], aligned["spy"], ma_period=win).dropna()
            if not rs.empty:
                raw = float(rs.iloc[-1])
                latest[f"RS_{win}d"]       = round(raw, 2)
                latest[f"Norm_RS_{win}d"]  = round(normalize_to_100_scale(raw), 2)

        raw_keys = [k for k in latest if k.startswith("RS_")]
        if not raw_keys:
            continue

        rs_avg      = round(sum(latest[k] for k in raw_keys) / len(raw_keys), 2)
        norm_rs_avg = round(normalize_to_100_scale(rs_avg), 2)

        row = {
            "Ticker": ticker,
            "RS_60d": np.nan, "RS_120d": np.nan, "RS_250d": np.nan, "RS_avg": rs_avg,
            "Norm_RS_60d": np.nan, "Norm_RS_120d": np.nan, "Norm_RS_250d": np.nan, "Norm_RS_avg": norm_rs_avg
        }
        for win in RS_WINDOWS:
            if f"RS_{win}d" in latest:
                row[f"RS_{win}d"]      = latest[f"RS_{win}d"]
                row[f"Norm_RS_{win}d"] = latest[f"Norm_RS_{win}d"]

        mansfield_rows.append(row)
        all_rows_for_fallback.append(row)
    except Exception:
        pass

mansfield_df = pd.DataFrame(mansfield_rows).sort_values(by="Norm_RS_avg", ascending=False)

if not mansfield_df.empty:
    mansfield_df_70 = mansfield_df[mansfield_df["Norm_RS_avg"] >= 70].copy()
    if mansfield_df_70.empty:
        print("🚫 Norm_RS_avg ≥ 70 조건 만족 종목 없음 → 상위 10개 출력:")
        display(mansfield_df.head(10))
    else:
        print("📌 정규화 Mansfield RS (60/120/250 + 평균) - Norm_RS_avg ≥ 70:")
        display(mansfield_df_70)
else:
    if all_rows_for_fallback:
        fb = pd.DataFrame(all_rows_for_fallback).sort_values(by="Norm_RS_avg", ascending=False).head(10)
        print("🚫 유효 RS 결과가 거의 없습니다 → 상위 10개 임시 출력:")
        display(fb)
    else:
        print("🚫 RS 결과가 비어 있습니다. 데이터/네트워크를 확인하세요.")

print("\n📌 변동성 조정 모멘텀 상위 20 (S&P500 기준):")
data = yf.download(universe, period="1y", auto_adjust=False, progress=False)["Close"]
data = data.dropna(axis=1, how="any").sort_index()

def calculate_risk_adjusted_momentum(data, period_3m=63, period_6m=126, period_12m=252, last_n_days=10):
    daily_top_20_stocks = {}
    trading_days = data.index[-last_n_days:]
    for day in trading_days:
        end_idx  = data.index.get_loc(day)
        start_3m = max(0, end_idx - period_3m  + 1)
        start_6m = max(0, end_idx - period_6m  + 1)
        start_12m= max(0, end_idx - period_12m + 1)

        r3  = data.iloc[start_3m:end_idx+1].pct_change().dropna().mean()
        r6  = data.iloc[start_6m:end_idx+1].pct_change().dropna().mean()
        r12 = data.iloc[start_12m:end_idx+1].pct_change().dropna().mean()

        v3  = data.iloc[start_3m:end_idx+1].pct_change().dropna().std()
        v6  = data.iloc[start_6m:end_idx+1].pct_change().dropna().std()
        v12 = data.iloc[start_12m:end_idx+1].pct_change().dropna().std()

        avg = (r3/v3 + r6/v6 + r12/v12) / 3
        top_20 = avg.nlargest(20).index.tolist()
        daily_top_20_stocks[day] = top_20

    df = pd.DataFrame.from_dict(daily_top_20_stocks, orient="index",
                                columns=[f"Top {i+1}" for i in range(20)])
    df.index.name = "Date"
    return df

momentum_df = calculate_risk_adjusted_momentum(data, last_n_days=10)
display(momentum_df)
