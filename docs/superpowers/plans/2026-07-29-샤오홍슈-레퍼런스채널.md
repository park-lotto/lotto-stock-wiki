# 샤오홍슈 레퍼런스 채널 랭킹 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** "레퍼런스 랭킹" 화면의 샤오홍슈 탭을(현재는 번역 딥링크 카드뿐) 인스타·틱톡과 동일한
"계정 등록 → 자동 크롤 → 강도별 랭킹" 진짜 탭으로 만든다.

**Architecture:** 등록된 샤오홍슈 프로필 URL마다 Playwright(로그인 세션 재사용, 서버 데이터센터
IP 직결)로 프로필 페이지를 열어 최근 48h 노트를 추출 → 해외HOT에서 이미 검증된 view-less 랭킹
공식(`build_overseas_items`, window_hours=48로 재사용)으로 정규화·랭킹 → 기존 `platform_seeds`/
`platform_snapshots`/`last_run::{platform}` 저장소·API(`/api/seeds`, `/api/collect`,
`/api/reference`)가 이미 플랫폼 문자열 하나로 전부 범용이라 백엔드 라우트 변경은 0.

**Tech Stack:** Python(FastAPI, Playwright sync API), SQLite(기존 `Store` 클래스), 바닐라 JS(기존
`static/index.html` / `static/refs.html`).

```python
"""샤오홍슈 레퍼런스 채널 수집 — 프로필 페이지 크롤(2026-07-29).

로그인 세션(storageState) 재사용 + 서버 데이터센터 IP 직결로 rednote.com 프로필 페이지를
헤드리스로 연다(Phase 0 스파이크 검증: docs/superpowers/specs/2026-07-29-샤오홍슈-레퍼런스채널-design.md).
build_overseas_items가 기대하는 raw dict 스키마로 반환 — 해외HOT과 랭킹 공식을 그대로 공유한다.

채널 하나가 실패해도 전체가 죽지 않는다(instagram_playwright.py와 동일 원칙).
테스트는 _scrape_one을 주입해 브라우저 없이 돈다(test_xiaohongshu_playwright.py).
"""
import re
from shopping_shorts import config

LAST_TALLY = {"ok": 0, "login_wall": 0, "not_found": 0, "error": 0}

# Task 1 스파이크에서 채택된 노트ID 패턴으로 교체할 것(현재는 검색결과 페이지에서 확인된
# search_result 패턴을 기본값으로 둠 — 프로필 페이지가 다른 패턴을 쓰면 Task 1 결과로 바꾼다).
_NOTE_ID_RE = re.compile(r"/search_result/([0-9a-f]{24})")


def _profile_username(url):
    """프로필 URL에서 표시용 계정 식별자 추출(URL 마지막 path segment)."""
    return (url or "").rstrip("/").rsplit("/", 1)[-1]


def _parse_note_card(anchor_href, block_text):
    """앵커 href + 주변 텍스트 블록 → note dict 하나. 검색결과 실측 형식
    '제목 | 작성자 | 날짜 | 좋아요'을 1차 가정 — Task 1 dump로 확인 후 조정."""
    m = _NOTE_ID_RE.search(anchor_href or "")
    if not m:
        return None
    parts = [p.strip() for p in (block_text or "").split("|") if p.strip()]
    title = parts[0] if len(parts) > 0 else ""
    likes_raw = parts[-1] if len(parts) > 0 else "0"
    likes_num = re.sub(r"[^\d.]", "", likes_raw) or "0"
    likes = int(float(likes_num) * (1000 if "천" in likes_raw else
                                     10000 if ("만" in likes_raw or "w" in likes_raw.lower()) else 1))
    return {
        "note_id": m.group(1),
        "title": title,
        "published_at": "",   # Task 1 dump로 실제 날짜 텍스트 위치 확인 후 ISO 변환 로직 채움
        "likes": likes,
        "comments": 0,
        "collects": 0,
        "shares": 0,
        "thumbnail": "",
        "url": f"https://www.rednote.com{anchor_href}" if anchor_href.startswith("/") else anchor_href,
    }


def _scrape_one_playwright(profile_url):
    """계정 1개 → (notes, page_url, error). 브라우저를 실제로 띄우는 유일한 함수."""
    from playwright.sync_api import sync_playwright   # 지연 import

    captured = []
    try:
        with sync_playwright() as p:
            browser = p.chromium.launch(headless=True)
            ctx = browser.new_context(storage_state=config.XIAOHONGSHU_SESSION_PATH)
            page = ctx.new_page()
            page.goto(profile_url, timeout=config.XIAOHONGSHU_PW_TIMEOUT_MS,
                      wait_until="domcontentloaded")
            page.wait_for_timeout(4000)
            final_url = page.url
            if "website-login/error" in final_url or "login" in final_url:
                ctx.close(); browser.close()
                return [], final_url, "login_wall"
            anchors = page.eval_on_selector_all(
                "a", "els => els.map(e => [e.getAttribute('href'), e.closest('section,div')?.innerText || e.innerText])")
            seen = set()
            for href, block_text in anchors:
                note = _parse_note_card(href, block_text)
                if note and note["note_id"] not in seen:
                    seen.add(note["note_id"])
                    captured.append(note)
            ctx.close()
            browser.close()
        return captured, final_url, None
    except Exception as e:                        # noqa: BLE001 — 계정 하나의 실패로 전체가 죽지 않게
        return [], profile_url, str(e)[:200]


def fetch_notes(profile_urls, on_progress=None, _scrape_one=None):
    """profile_urls → build_overseas_items 스키마 raw dict 리스트.

    on_progress(done, total, items_so_far, tally): 계정 1개 끝날 때마다 호출(선택).
    _scrape_one: 테스트 주입용. 기본은 실제 Playwright 스크레이퍼.
    """
    scrape = _scrape_one or _scrape_one_playwright
    urls = [(u or "").strip() for u in (profile_urls or [])]
    urls = [u for u in urls if u]
    total = len(urls)
    tally = {"ok": 0, "login_wall": 0, "not_found": 0, "error": 0}
    items = []
    for i, url in enumerate(urls, start=1):
        notes, page_url, error = scrape(url)
        if error == "login_wall":
            verdict = "login_wall"
        elif error:
            verdict = "error"
        elif not notes:
            verdict = "not_found"
        else:
            verdict = "ok"
        tally[verdict] = tally.get(verdict, 0) + 1
        uname = _profile_username(url)
        for n in notes:
            items.append({
                "video_id": n.get("note_id", ""),
                "title": n.get("title", ""),
                "published_at": n.get("published_at", ""),
                "views": 0,
                "likes": int(n.get("likes") or 0),
```
