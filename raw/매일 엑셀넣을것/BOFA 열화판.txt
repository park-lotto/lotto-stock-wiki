# 1) HEAT(0~10): '과열 레벨'만 말하게 (VIX/금리변동성 제거)
#    - HYG/IEF (risk appetite)
#    - HYG/LQD (credit calm)
#    - RSP/SPY (style concentration: mega-cap 쏠림)
#    - SPY/QQQ 200MA 이격 (position stretch)
#
# 2) SHOCK FLAG(ON/OFF): 이벤트/정책 충격 대응용 (예측이 아니라 대응)
#    - VIX 3일 +30% 이상
#    - HYG/LQD 5일 -2% 이하
#
# 3) TREND FILTER & HEAT FLOOR: 추세가 강할 때 '낮은 heat 착시' 보정
#    - SPY > 200MA AND 200MA 상승이면 Heat_floor=2.5
#
# 출력: 첫 번째 그래프 한 장 (SPY/QQQ + Heat + Shock 마커)

import yfinance as yf
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ----------------------------
# 0) 설정
# ----------------------------
START  = "2015-01-01"
END    = None

ZS_WIN = 252          # 표준화 창(1년)
SMOOTH_HEAT = 10      # Heat를 '레벨형'으로 만들기 위한 스무딩(권장 10~20)
HEAT_FLOOR = 2.5      # 강한 추세(상승장)에서 heat 바닥 보정

# Shock thresholds
VIX_3D_UP = 0.30      # VIX 3일 +30%
CREDIT_5D_DN = -0.02  # HYG/LQD 5일 -2%

TICKERS = {
    "SPY": "SPY",
    "QQQ": "QQQ",
    "RSP": "RSP",
    "VIX": "^VIX",
    "HYG": "HYG",
    "IEF": "IEF",
    "LQD": "LQD",
}

# ----------------------------
# 1) 다운로드
# ----------------------------
def download_adjclose(tickers, start, end=None):
    data = yf.download(list(tickers.values()), start=start, end=end, progress=False, auto_adjust=False)
    if isinstance(data.columns, pd.MultiIndex):
        col = "Adj Close" if ("Adj Close" in data.columns.get_level_values(0)) else "Close"
        px = data[col].copy()
    else:
        px = data.copy()
    inv = {v: k for k, v in tickers.items()}
    px = px.rename(columns=inv).sort_index()
    return px

px = download_adjclose(TICKERS, START, END)

# ----------------------------
# 2) 유틸
# ----------------------------
def zscore(s, win=252):
    mu = s.rolling(win).mean()
    sd = s.rolling(win).std(ddof=0)
    return (s - mu) / sd

def clip01(x):
    return x.clip(0.0, 1.0)

def norm01_from_z(z, lo=-2.0, hi=2.0):
    return clip01((z - lo) / (hi - lo))

def dist_from_ma(price, n=200):
    m = price.rolling(n).mean()
    return (price / m - 1.0)

def normalize100(s):
    s = s.dropna()
    return (s / s.iloc[0]) * 100.0

def weighted_mean_renorm(df_vals: pd.DataFrame, weights: dict):
    cols = [c for c in df_vals.columns if c in weights]
    W = pd.Series({c: float(weights[c]) for c in cols})
    W = W / W.sum()

    v = df_vals[cols]
    mask = v.notna().astype(float)
    denom = (mask * W).sum(axis=1)
    numer = (v.mul(W, axis=1)).sum(axis=1)
    out = numer / denom
    return out.where(denom > 0)

# ----------------------------
# 3) HEAT (과열 레벨) 컴포넌트 (0~1)
# ----------------------------
df = pd.DataFrame(index=px.index)

# 3-1) Risk appetite: HYG/IEF (과열이면 ↑)
if {"HYG","IEF"}.issubset(px.columns):
    ra = px["HYG"] / px["IEF"]
    df["heat_risk_appetite"] = norm01_from_z(zscore(ra, ZS_WIN), lo=0.0, hi=2.0)

# 3-2) Credit calm: HYG/LQD (과열이면 ↑)
if {"HYG","LQD"}.issubset(px.columns):
    cc = px["HYG"] / px["LQD"]
    df["heat_credit_calm"] = norm01_from_z(zscore(cc, ZS_WIN), lo=0.0, hi=2.0)

# 3-3) Style concentration: RSP/SPY 낮아질수록(=메가캡 쏠림) 경고 ↑
if {"RSP","SPY"}.issubset(px.columns):
    rsp_spy = px["RSP"] / px["SPY"]
    rsp_spy_z = zscore(rsp_spy, ZS_WIN)
    df["heat_style_concentration"] = clip01((0.0 - rsp_spy_z) / 2.0)  # z<=-2 -> 1, z>=0 -> 0

# 3-4) Position stretch: SPY/QQQ 200MA 이격(과열이면 ↑)
if "SPY" in px.columns:
    spy_ext_z = zscore(dist_from_ma(px["SPY"], 200), ZS_WIN)
    df["heat_spy_extension"] = norm01_from_z(spy_ext_z, lo=0.5, hi=2.0)

if "QQQ" in px.columns:
    qqq_ext_z = zscore(dist_from_ma(px["QQQ"], 200), ZS_WIN)
    df["heat_qqq_extension"] = norm01_from_z(qqq_ext_z, lo=0.5, hi=2.0)

# HEAT 가중치 (단순/견고하게)
WEIGHTS_HEAT = {
    "heat_risk_appetite": 1.2,
    "heat_credit_calm":  1.0,
    "heat_style_concentration": 0.8,
    "heat_spy_extension": 0.8,
    "heat_qqq_extension": 0.8,
}

heat_cols = [c for c in df.columns if c.startswith("heat_")]
df["heat_raw01"] = weighted_mean_renorm(df[heat_cols], WEIGHTS_HEAT)

# 레벨형으로 만들기: 스무딩(변동성 감소)
df["heat_0_10"] = (df["heat_raw01"] * 10.0).rolling(SMOOTH_HEAT).mean()

# ----------------------------
# 4) TREND FILTER + HEAT FLOOR
# ----------------------------
df["trend_on"] = False
df["heat_floor"] = 0.0

if "SPY" in px.columns:
    spy = px["SPY"]
    spy_ma200 = spy.rolling(200).mean()
    spy_ma200_up = spy_ma200.diff(20) > 0  # 1개월(20거래일) 기준 상승 기울기
    trend_on = (spy > spy_ma200) & spy_ma200_up

    df["trend_on"] = trend_on
    df["heat_floor"] = np.where(trend_on, HEAT_FLOOR, 0.0)

# 최종 Heat (바닥 보정)
df["heat_final_0_10"] = df[["heat_0_10", "heat_floor"]].max(axis=1)

# ----------------------------
# 5) SHOCK FLAG (ON/OFF)
# ----------------------------
df["shock_flag"] = False

# (a) VIX 3일 +30% 이상
if "VIX" in px.columns:
    vix = px["VIX"]
    df["shock_vix"] = vix.pct_change(3) >= VIX_3D_UP
else:
    df["shock_vix"] = False

# (b) Credit shock: HYG/LQD 5일 -2% 이하
if {"HYG","LQD"}.issubset(px.columns):
    credit = px["HYG"] / px["LQD"]
    df["shock_credit"] = credit.pct_change(5) <= CREDIT_5D_DN
else:
    df["shock_credit"] = False

df["shock_flag"] = (df["shock_vix"] | df["shock_credit"]).fillna(False)

# ----------------------------
# 6) PLOT: 한 장만 출력
# ----------------------------
plt.figure(figsize=(14, 7))

spy_n = normalize100(px["SPY"]).reindex(df.index) if "SPY" in px.columns else None
qqq_n = normalize100(px["QQQ"]).reindex(df.index) if "QQQ" in px.columns else None

if spy_n is not None:
    plt.plot(spy_n.index, spy_n.values, label="SPY (Normalized=100)")
if qqq_n is not None:
    plt.plot(qqq_n.index, qqq_n.values, label="QQQ (Normalized=100)")

ax = plt.gca()
ax2 = ax.twinx()
ax2.plot(df.index, df["heat_final_0_10"], linestyle="--", label="Heat (0~10, refined)")

# Shock 마커(점)
shock_idx = df.index[df["shock_flag"] == True]
ax2.scatter(shock_idx, df.loc[shock_idx, "heat_final_0_10"], marker="o", s=18, label="Shock ON")

# 기준선
ax2.axhline(7.5, linestyle="--", linewidth=1)
ax2.axhline(2.0, linestyle="--", linewidth=1)

ax.set_title("SPY/QQQ + Refined Heat(0~10) + Shock Flags (Yahoo only)")
ax.set_ylabel("Price (Normalized)")
ax2.set_ylabel("Heat Score (0~10)")
ax.grid(True)

# 범례 합치기
lines1, labels1 = ax.get_legend_handles_labels()
lines2, labels2 = ax2.get_legend_handles_labels()
ax2.legend(lines1 + lines2, labels1 + labels2, loc="upper left")

plt.show()

print("최근 Heat:", df["heat_final_0_10"].dropna().iloc[-1])
print("최근 Trend ON 여부:", bool(df["trend_on"].dropna().iloc[-1]) if df["trend_on"].notna().any() else None)
print("최근 Shock ON 여부:", bool(df["shock_flag"].dropna().iloc[-1]) if df["shock_flag"].notna().any() else None)
print("Heat 구성요소:", [c for c in heat_cols if c in WEIGHTS_HEAT])
