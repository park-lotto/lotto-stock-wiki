# ----------------------------
# 1) Global params
# ----------------------------
# 일봉은 최근 6개월(거래일 약 126) 뷰, 주봉은 최근 1년(52주) 뷰
PLOT_DAYS_D   = 126      # Daily plot window
PLOT_WEEKS_W  = 52       # Weekly plot window
BUFFER_DAYS   = 120      # indicator warmup for daily window
# 주봉을 안정적으로 만들기 위해 일봉을 최소 500일 확보 (≈ 1.5년)
LOOKBACK_ALL  = 500

VOL_MA        = 20       # volume moving average for spike
ZSCORE_WIN    = 60       # z-score lookback
ACC_WIN       = 7        # 7-day rolling sum (oscillator core)

# Component weights for TI_raw
W_VOL_SPIKE   = 0.50
W_TURNOVER    = 0.30
W_GAP_TREND   = 0.20

# 티커 컬럼이 없고 시트도 1개일 때 그래프 제목용
DEFAULT_TICKER_NAME = "DATA"

# (선택) 유통/발행주식수(턴오버 계산용). 모르면 None
SHARES_HINT = None

# (선택) 특정 시트만 읽고 싶으면 시트명 입력. 기본(None)은 모든 시트/또는 티커 컬럼 자동 처리
SHEET_NAME = None  # e.g., "Sheet1"

# ----------------------------
# 2) Helper functions
# ----------------------------
def rolling_zscore(series, window=60, min_periods=20):
    mean = series.rolling(window, min_periods=min_periods).mean()
    std  = series.rolling(window, min_periods=min_periods).std(ddof=0)
    return (series - mean) / (std.replace(0, np.nan))

def minmax_0_100(series: pd.Series):
    s = series.dropna()
    if s.empty:
        return series * np.nan
    vmin, vmax = s.min(), s.max()
    if vmin == vmax:
        return series * np.nan
    return (series - vmin) * 100.0 / (vmax - vmin)

def build_trading_intensity(df_ohlcv: pd.DataFrame, shares_hint: float | None):
    """Compute 7D trading-intensity oscillator and components from OHLCV (daily)."""
    df = df_ohlcv.copy()
    df["PrevClose"] = df["Close"].shift(1)

    # Volume spike vs 20D MA
    df["VolMA"] = df["Volume"].rolling(VOL_MA, min_periods=max(5, VOL_MA//2)).mean()
    df["VolSpike"] = df["Volume"] / df["VolMA"]

    # Turnover (prefer shares if given; else fallback)
    if shares_hint and shares_hint > 0:
        df["Turnover"] = df["Volume"] / shares_hint
    else:
        df["Turnover"] = df["Volume"] / df["Volume"].rolling(60, min_periods=20).mean()

    # Gap + intraday trend
    df["Gap"] = (df["Open"] / df["PrevClose"] - 1.0)
    df["IntradayTrend"] = (df["Close"] / df["Open"] - 1.0)
    df["GapTrend"] = df["Gap"] + df["IntradayTrend"]

    # Z-scores
    df["z_VolSpike"] = rolling_zscore(df["VolSpike"], window=ZSCORE_WIN)
    df["z_Turnover"] = rolling_zscore(df["Turnover"],  window=ZSCORE_WIN)
    df["z_GapTrend"] = rolling_zscore(df["GapTrend"],  window=ZSCORE_WIN)

    # Composite daily score
    df["TI_raw"] = (
        W_VOL_SPIKE * df["z_VolSpike"].fillna(0) +
        W_TURNOVER  * df["z_Turnover"].fillna(0) +
        W_GAP_TREND * df["z_GapTrend"].fillna(0)
    )

    # 7D rolling sum -> normalize to 0-100
    df["TI_acc7"] = df["TI_raw"].rolling(ACC_WIN, min_periods=ACC_WIN).sum()
    df["TI_acc7_0_100"] = minmax_0_100(df["TI_acc7"])
    return df

def plot_price_and_osc(close_s, osc_s, ticker, title_suffix):
    """Draw price + raw osc + SMA3/EMA5/Signal EMA7 on shared axes."""
    plot_close = close_s.copy()
    plot_osc   = osc_s.copy()

    sma3 = plot_osc.rolling(3, min_periods=1).mean()
    ema5 = plot_osc.ewm(span=5, adjust=False).mean()
    sig7 = plot_osc.ewm(span=7, adjust=False).mean()

    fig, ax1 = plt.subplots(figsize=(12, 5))
    ax1.plot(plot_close.index, plot_close.values, lw=1.6, color="black", label="Price")
    ax1.set_xlabel("Date")
    ax1.set_ylabel("Price", color="black")
    ax1.tick_params(axis="y", labelcolor="black")

    ax2 = ax1.twinx()
    ax2.plot(plot_osc.index, plot_osc.values, ls="--", lw=1.3, color="black", label="7D Osc (raw)")
    ax2.plot(sma3.index,    sma3.values,      ls="-.", lw=1.5, color="tab:blue",   label="SMA(3)")
    ax2.plot(ema5.index,    ema5.values,      ls="-",  lw=1.6, color="tab:orange", label="EMA(5)")
    ax2.plot(sig7.index,    sig7.values,      ls=":",  lw=1.6, color="tab:green",  label="Signal EMA(7)")
    ax2.axhline(50, ls="--", lw=1, alpha=0.4, color="gray")
    ax2.set_ylabel("Oscillator (0-100)")

    l1, lb1 = ax1.get_legend_handles_labels()
    l2, lb2 = ax2.get_legend_handles_labels()
    ax1.legend(l1 + l2, lb1 + lb2, loc="upper left")

    plt.title(f"{ticker} - Price & 7D Oscillator ({title_suffix})")
    plt.tight_layout()
    plt.show()

def normalize_columns(df: pd.DataFrame) -> pd.DataFrame:
    """
    Map 다양한 컬럼명을 Date/Open/High/Low/Close/Volume로 매핑.
    예: 일자/시가/고가/저가/종가/거래량 지원
    """
    colmap = {
        # Date
        "date": "Date", "datetime": "Date", "day": "Date",
        "일자": "Date", "날짜": "Date",

        # OHLCV
        "open": "Open", "시가": "Open",
        "high": "High", "고가": "High",
        "low":  "Low",  "저가": "Low",
        "close":"Close","종가": "Close", "adj close": "Close", "adjclose": "Close",
        "volume":"Volume","거래량": "Volume",
    }

    newcols = {}
    for c in df.columns:
        key = str(c).strip().lower()
        key = key.replace("_", " ").replace("-", " ")
        key = " ".join(key.split())
        newcols[c] = colmap.get(key, c)
    df = df.rename(columns=newcols)

    # 필수 컬럼 체크
    required = ["Date", "Open", "Close", "Volume"]
    missing = [c for c in required if c not in df.columns]
    if missing:
        raise ValueError(f"필수 컬럼이 없습니다: {missing}\n현재 컬럼: {list(df.columns)}")

    return df

def preprocess_ohlcv(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df = normalize_columns(df)

    df["Date"] = pd.to_datetime(df["Date"])
    df = df.sort_values("Date").drop_duplicates(subset=["Date"])
    df = df.set_index("Date")

    # numeric coercion
    for c in ["Open","High","Low","Close","Volume"]:
        if c in df.columns:
            df[c] = pd.to_numeric(df[c], errors="coerce")

    df = df.dropna(subset=["Open","Close","Volume"])

    # Ensure columns exist
    if "High" not in df.columns:
        df["High"] = np.nan
    if "Low" not in df.columns:
        df["Low"] = np.nan

    df = df[["Open","High","Low","Close","Volume"]]
    return df

# ----------------------------
# 3) Load Excel (supports: single sheet / multi-sheet / ticker column)
# ----------------------------
xls = pd.ExcelFile(io.BytesIO(uploaded[FILE_NAME]))
sheet_names = xls.sheet_names if SHEET_NAME is None else [SHEET_NAME]

datasets = []

for sh in sheet_names:
    tmp = pd.read_excel(xls, sheet_name=sh)

    # Case 1) ticker/symbol column exists
    ticker_cols = [c for c in tmp.columns if str(c).strip().lower() in ["ticker", "symbol", "티커", "종목", "종목코드"]]
    if ticker_cols:
        tc = ticker_cols[0]
        for tkr, g in tmp.groupby(tc):
            g = g.drop(columns=[tc])
            datasets.append((str(tkr).strip(), g))
        continue

    # Case 2) each sheet is a dataset
    datasets.append((sh, tmp))

# If only one dataset and sheet name is generic, use default ticker name
if len(datasets) == 1:
    tkr, df0 = datasets[0]
    if str(tkr).strip().lower() in ["데이터", "data", "sheet1", "sheet"]:
        datasets = [(DEFAULT_TICKER_NAME, df0)]

# ----------------------------
# 4) Main loop (Excel-based)
# ----------------------------
for ticker, raw_df in datasets:
    print(f"\n=== Processing {ticker} ===")

    df_all = preprocess_ohlcv(raw_df)

    if df_all.empty or len(df_all) < (ZSCORE_WIN + ACC_WIN + 10):
        print(f"[Skip] {ticker}: 데이터가 너무 짧습니다. (rows={len(df_all)})")
        continue

    # Use last LOOKBACK_ALL rows (or all if shorter)
    df_use = df_all.tail(min(len(df_all), LOOKBACK_ALL)).copy()

    # Compute oscillator on daily data
    feat_full = build_trading_intensity(df_use, SHARES_HINT)

    # ---------- DAILY view (최근 6M) ----------
    daily_view = feat_full.tail(PLOT_DAYS_D + BUFFER_DAYS).copy()
    daily_close = daily_view["Close"].tail(PLOT_DAYS_D)
    daily_osc   = daily_view["TI_acc7_0_100"].tail(PLOT_DAYS_D)
    plot_price_and_osc(daily_close, daily_osc, ticker, "Daily (6M)")

    # ---------- WEEKLY view (최근 1Y) ----------
    wk = pd.DataFrame({
        "Close": feat_full["Close"],
        "Osc":   feat_full["TI_acc7_0_100"]
    }).resample("W-FRI").last().dropna()

    weekly_close = wk["Close"].tail(PLOT_WEEKS_W)
    weekly_osc   = wk["Osc"].tail(PLOT_WEEKS_W)
    plot_price_and_osc(weekly_close, weekly_osc, ticker, "Weekly (1Y)")

print("\nDone.")
