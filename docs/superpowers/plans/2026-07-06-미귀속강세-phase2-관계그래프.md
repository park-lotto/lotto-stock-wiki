`id INTEGER PK, src TEXT, dst TEXT, relation_type TEXT, source_atom_id TEXT NULL, source TEXT, confidence REAL, created_at TEXT`

# pipeline/atoms/edges.py
"""관계그래프 엣지 저장소(atom_edges). Phase 2 예측망의 그래프-홉 귀속에 쓰인다."""
from datetime import datetime
from pipeline.atoms import db as _db

def init_edges() -> None:
    conn = _db.get_conn()
    conn.execute("""
        CREATE TABLE IF NOT EXISTS atom_edges(
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            src TEXT NOT NULL, dst TEXT NOT NULL,
            relation_type TEXT NOT NULL, source_atom_id TEXT,
            source TEXT, confidence REAL DEFAULT 1.0, created_at TEXT NOT NULL,
            UNIQUE(src, dst, relation_type)
        )""")
    conn.execute("CREATE INDEX IF NOT EXISTS idx_edges_dst ON atom_edges(dst)")
    conn.execute("CREATE INDEX IF NOT EXISTS idx_edges_src ON atom_edges(src)")
    conn.commit(); conn.close()

def insert_edge(src, dst, relation_type, source, source_atom_id=None, confidence=1.0) -> None:
    conn = _db.get_conn()
    conn.execute("""INSERT OR IGNORE INTO atom_edges
        (src,dst,relation_type,source_atom_id,source,confidence,created_at)
        VALUES (?,?,?,?,?,?,?)""",
        (src, dst, relation_type, source_atom_id, source, confidence,
         datetime.now().isoformat()))
    conn.commit(); conn.close()

def assets_in_sector(sector: str) -> list:
    conn = _db.get_conn()
    rows = conn.execute("SELECT DISTINCT src FROM atom_edges WHERE dst=?", (sector,)).fetchall()
    conn.close()
    return [r[0] for r in rows]

def sectors_of_asset(asset: str) -> list:
    conn = _db.get_conn()
    rows = conn.execute("SELECT DISTINCT dst FROM atom_edges WHERE src=?", (asset,)).fetchall()
    conn.close()
    return [r[0] for r in rows]
```

# add to pipeline/atoms/edges.py
import json, pathlib
_ATOMS_DIR = pathlib.Path(__file__).parent

def _as_list(v):
    if v is None: return []
    return v if isinstance(v, list) else [v]

def seed_sector_edges(stock_sector_map: dict, foreign_sector_map: dict) -> int:
    init_edges()
    n = 0
    for stock, secs in (stock_sector_map or {}).items():
        for sec in _as_list(secs):
            insert_edge(stock, sec, "sector_member", "sector_map"); n += 1
    for ent, secs in (foreign_sector_map or {}).items():
        for sec in _as_list(secs):
            insert_edge(ent, sec, "foreign_sector", "foreign_map"); n += 1
    return n

def seed_from_files() -> int:
    def _load(name):
        p = _ATOMS_DIR / name
        return json.loads(p.read_text(encoding="utf-8")) if p.exists() else {}
    return seed_sector_edges(_load("stock_sector_map.json"), _load("foreign_sector_map.json"))
```

```python
# strength_net.py — attribute_mover 교체
def attribute_mover(mover: dict, days: int = 3, query_fn=None, related_fn=None) -> dict:
    """직접 원자 → 없으면 그래프-홉(같은 섹터 관련자산의 호재) → 없으면 미귀속."""
    if query_fn is None:
        query_fn = _db.query_atoms
    name = mover.get("name")
    base = {"name": name, "code": mover.get("code"),
            "sector": mover.get("sector"), "rate": mover.get("rate")}
    atoms = query_fn(asset=name, days=days, active_only=True) or []
    hits = [a for a in atoms if (a.get("signal") or "").strip().lower() in _BULLISH_SIGNALS
            or (a.get("signal") or "").strip() in _BULLISH_SIGNALS]
    if hits:
        hits.sort(key=lambda a: a.get("strength_score", 1), reverse=True)
        top = hits[0]
        return {**base, "attributed": True, "issue": top.get("content"),
                "trust": trust_tier(top), "source": top.get("source_name"),
                "atom_ids": [a["id"] for a in hits], "status": "attributed",
                "priority": 1, "flag": None, "via": None}
    # 그래프-홉 폴백
    if related_fn is not None:
        for rel in related_fn(mover.get("sector")) or []:
            if rel == name:
                continue
            r_atoms = query_fn(asset=rel, days=days, active_only=True) or []
            r_hits = [a for a in r_atoms if (a.get("signal") or "").strip().lower() in _BULLISH_SIGNALS
                      or (a.get("signal") or "").strip() in _BULLISH_SIGNALS]
            if r_hits:
                r_hits.sort(key=lambda a: a.get("strength_score", 1), reverse=True)
                rt = r_hits[0]
                return {**base, "attributed": True,
                        "issue": f"연계: {rel} — {rt.get('content')}",
                        "trust": "🔵", "source": rt.get("source_name"),
                        "atom_ids": [a["id"] for a in r_hits], "status": "attributed",
                        "priority": 1, "flag": None, "via": rel}
    return {**base, "attributed": False, "issue": None, "trust": None,
            "source": None, "atom_ids": [], "status": "unattributed",
            "priority": 0, "flag": _UNATTR_FLAG, "via": None}
```
