# 4. ETF 및 KOSPI 데이터 다운로드

data = yf.download(tickers + ["^KS11"], period="1y", interval="1d")["Close"]

# ----------------------------
# 5. 변동성 조정 모멘텀 계산 함수
# ----------------------------
def calculate_risk_adjusted_momentum(data, last_n_days=10):
    daily_top = {}
    trading_days = data.dropna().index[-last_n_days:]

    for day in trading_days:
        end_idx = data.index.get_loc(day)
        start_3m = max(0, end_idx - 62)
        start_6m = max(0, end_idx - 125)
        start_12m = max(0, end_idx - 251)

        returns_3m = data.iloc[start_3m:end_idx + 1].pct_change(fill_method=None).mean()
        returns_6m = data.iloc[start_6m:end_idx + 1].pct_change(fill_method=None).mean()
        returns_12m = data.iloc[start_12m:end_idx + 1].pct_change(fill_method=None).mean()

        vol_3m = data.iloc[start_3m:end_idx + 1].pct_change(fill_method=None).std()
        vol_6m = data.iloc[start_6m:end_idx + 1].pct_change(fill_method=None).std()
        vol_12m = data.iloc[start_12m:end_idx + 1].pct_change(fill_method=None).std()

        risk_adj = (returns_3m / vol_3m + returns_6m / vol_6m + returns_12m / vol_12m) / 3
        top_15 = risk_adj.nlargest(15).index
        daily_top[day] = [f"{ticker} ({korea_industry_etfs[ticker]})" for ticker in top_15]

    df = pd.DataFrame.from_dict(daily_top, orient="index", columns=[f"Top {i+1}" for i in range(15)])
    df.index.name = "Date"
    return df

# ----------------------------
# 6. Mansfield RS 계산 함수
# ----------------------------
def compute_mansfield_rs(etf, benchmark, ma_period=52):
    relative = etf / benchmark
    ma = relative.rolling(window=ma_period).mean()
    return ((relative / ma) - 1) * 100

def normalize_to_100_scale(x, scale=12):
    return 100 * (1 / (1 + np.exp(-x / scale)))

# ----------------------------
# 7. RS 결과 추출
# ----------------------------
mansfield_results = []
for ticker in tickers:
    try:
        etf_series = etf_data[ticker].dropna()
        common_idx = etf_series.index.intersection(kospi.index)
        etf_series = etf_series.loc[common_idx]
        kospi_series = kospi.loc[common_idx]

        rs = compute_mansfield_rs(etf_series, kospi_series)
        latest_raw = rs.dropna().iloc[-1]
        norm_rs = normalize_to_100_scale(latest_raw)

        if norm_rs >= 70:
            mansfield_results.append((ticker, korea_industry_etfs[ticker], round(norm_rs, 2), round(latest_raw, 2)))
    except:
        continue

mansfield_df = pd.DataFrame(mansfield_results, columns=["Ticker", "Name", "Normalized_RS", "Raw_RS"])
mansfield_df = mansfield_df.sort_values(by="Normalized_RS", ascending=False)
