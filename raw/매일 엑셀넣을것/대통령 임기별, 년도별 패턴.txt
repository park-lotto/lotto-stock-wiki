# STEP 1: 라이브러리 불러오기
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime
from google.colab import files

# STEP 2: 파일 업로드
uploaded = files.upload()
file_name = list(uploaded.keys())[0]

# STEP 3: 엑셀 파일 읽기 (A열: 날짜, B열: NASDAQCOM)
raw_df = pd.read_excel(file_name, sheet_name='Daily, Close', engine='openpyxl', header=None)
df = raw_df.iloc[1:, :2].copy()
df.columns = ['observation_date', 'NASDAQCOM']
df['observation_date'] = pd.to_datetime(df['observation_date'], errors='coerce')
df['NASDAQCOM'] = pd.to_numeric(df['NASDAQCOM'], errors='coerce')
df = df.dropna(subset=['observation_date', 'NASDAQCOM']).copy()

# STEP 4: 기본 전처리
df = df.sort_values('observation_date').copy()
df['year'] = df['observation_date'].dt.year.astype(int)

# STEP 5-1: 대통령 임기 주기 연차 구분
# (1961부터 4년마다 취임)
inauguration_years = list(range(1961, datetime.now().year + 1, 4))

def get_cycle_year(y):
    for start in inauguration_years:
        if start <= y <= start + 3:
            return (y - start + 1)
    return None

df['cycle_year'] = df['year'].apply(get_cycle_year)
df = df.dropna(subset=['cycle_year']).copy()
df['cycle_year'] = df['cycle_year'].astype(int)

# ------------------------------------------------------------
# 핵심 수정 1) 연도별 누적 수익률 보간: "연도 내부"에서 pct_change()
# ------------------------------------------------------------
yearly_cum_returns = {}
for year, g in df.groupby('year'):
    g = g.sort_values('observation_date')
    r = g['NASDAQCOM'].pct_change().fillna(0)   # <- 연도 내부 수익률
    cum_return = (1 + r).cumprod() - 1

    # 최소 거래일 조건(원 코드 유지)
    if len(cum_return) >= 200:
        interpolated = np.interp(
            np.linspace(0, len(cum_return) - 1, 252),
            np.arange(len(cum_return)),
            cum_return.values
        )
        yearly_cum_returns[year] = interpolated

# ------------------------------------------------------------
# 핵심 수정 2) cycle_year별 평균 곡선: "연도 내부"에서 pct_change()
# ------------------------------------------------------------
composite_data = {1: [], 2: [], 3: [], 4: []}

for (yr, cy), g in df.groupby(['year', 'cycle_year']):
    g = g.sort_values('observation_date')
    if len(g) < 200:
        continue

    r = g['NASDAQCOM'].pct_change().fillna(0)   # <- 연도 내부 수익률
    cum_return = (1 + r).cumprod() - 1

    cum_return_interp = np.interp(
        np.linspace(0, len(cum_return) - 1, 252),
        np.arange(len(cum_return)),
        cum_return.values
    )
    composite_data[cy].append(cum_return_interp)

cycle_averages = {}
for cy in [1, 2, 3, 4]:
    if len(composite_data[cy]) > 0:
        cycle_averages[cy] = np.mean(composite_data[cy], axis=0)
    else:
        cycle_averages[cy] = np.zeros(252)

# STEP 7: 현재 연도 누적 수익률 계산
# "파일에 있는 최신 연도"를 current_year로 쓰는 게 안전
current_year = int(df['year'].max())

df_current = df[df['year'] == current_year].copy()
df_current = df_current.sort_values('observation_date')

# 연도 내부 수익률
df_current['Return'] = df_current['NASDAQCOM'].pct_change().fillna(0)
df_current['Cumulative Return'] = (1 + df_current['Return']).cumprod() - 1

valid_returns = df_current['Cumulative Return'].dropna()
actual_days = int(len(valid_returns))
has_data = actual_days >= 1

# ------------------------------------------------------------
# 핵심 수정 3) YTD는 252로 억지 보간하지 말고 "실제 길이"만 그림
# ------------------------------------------------------------
ytd_x = np.arange(actual_days)
ytd_y = valid_returns.values * 100  # %

# STEP 8: 시각화 ① 전체 대통령 주기 평균 vs 현재 연도(YTD 길이까지만)
colors = {1: 'blue', 2: 'orange', 3: 'green', 4: 'red'}
labels = {
    1: '1st Year (Post-election)',
    2: '2nd Year (Midterm)',
    3: '3rd Year (Pre-election)',
    4: '4th Year (Election)'
}

plt.figure(figsize=(14, 6))

# 평균 곡선은 252일 전체를 그리되,
# YTD와 같은 길이만 비교하고 싶으면 [:actual_days]로 잘라서 그려도 됨(아래 주석 참고)
for cy in [1, 2, 3, 4]:
    plt.plot(np.arange(252), cycle_averages[cy] * 100, label=labels[cy], color=colors[cy], linewidth=2)
    # 비교를 더 깔끔하게 하려면:
    # plt.plot(np.arange(actual_days), cycle_averages[cy][:actual_days] * 100, ...)

if has_data:
    plt.plot(ytd_x, ytd_y, label=f'{current_year} YTD (NASDAQ)', color='black', linestyle='--', linewidth=2)
    plt.annotate(f'{current_year} YTD',
                 xy=(actual_days-1, ytd_y[-1]),
                 xytext=(actual_days-1, ytd_y[-1] + 2),
                 arrowprops=dict(facecolor='black', arrowstyle='->'),
                 fontsize=10, color='black')

plt.title(f'① NASDAQ Presidential Cycle (All Years) vs. {current_year} YTD')
plt.xlabel('Trading Days (0–252)')
plt.ylabel('Cumulative Return (%)')
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

# STEP 9: 시각화 ② "현재 연도의 cycle_year" 평균 vs 현재 연도(YTD)
current_cycle_year = get_cycle_year(current_year)  # 2026이면 2가 나와야 정상
cy_label = labels.get(current_cycle_year, f'Cycle Year {current_cycle_year}')

plt.figure(figsize=(14, 6))
plt.plot(np.arange(252), cycle_averages[current_cycle_year] * 100,
         label=f'{cy_label} Avg', color=colors.get(current_cycle_year, 'blue'), linewidth=2)

if has_data:
    plt.plot(ytd_x, ytd_y, label=f'{current_year} YTD (NASDAQ)', color='black', linestyle='--', linewidth=2)
    plt.annotate(f'{current_year} YTD',
                 xy=(actual_days-1, ytd_y[-1]),
                 xytext=(actual_days-1, ytd_y[-1] + 2),
                 arrowprops=dict(facecolor='black', arrowstyle='->'),
                 fontsize=10, color='black')

plt.title(f'② NASDAQ {cy_label} vs. {current_year} YTD')
plt.xlabel('Trading Days (0–252)')
plt.ylabel('Cumulative Return (%)')
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

# STEP 10: 끝자리 기반 연도 누적 수익률 평균
def get_1yr_avg_curve_by_last_digit(digit):
    curves = []
    for year in yearly_cum_returns:
        if str(year)[-1] == str(digit):
            curve = yearly_cum_returns[year]
            if curve is not None and len(curve) == 252:
                curves.append(curve)
    if curves:
        return np.mean(curves, axis=0) * 100
    else:
        return np.zeros(252)

digit_1 = str(current_year)[-1]
digit_2 = str((int(digit_1) + 1) % 10)

avg_curve_1 = get_1yr_avg_curve_by_last_digit(digit_1)
avg_curve_2 = get_1yr_avg_curve_by_last_digit(digit_2)

# STEP 11: 시각화 ③ ④ 끝자리 기준
fig, axes = plt.subplots(1, 2, figsize=(16, 6), sharey=True)

axes[0].plot(np.arange(252), avg_curve_1, color='blue', linewidth=2)
axes[0].set_title(f'③ Start Year Ending in {digit_1}\nAverage Cumulative Return (1 Year)', fontsize=14)
axes[0].set_xlabel('Trading Days (0~252)')
axes[0].set_ylabel('Cumulative Return (%)')
axes[0].grid(True)

axes[1].plot(np.arange(252), avg_curve_2, color='orange', linewidth=2)
axes[1].set_title(f'④ Start Year Ending in {digit_2}\nAverage Cumulative Return (1 Year)', fontsize=14)
axes[1].set_xlabel('Trading Days (0~252)')
axes[1].grid(True)

plt.tight_layout()
plt.show()

# 디버그 출력(원하면 지워도 됨)
print(f"[DEBUG] current_year={current_year}, current_cycle_year={current_cycle_year}, actual_days={actual_days}")
