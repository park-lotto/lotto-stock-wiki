**Goal:** 렌즈 모달 안에서 중국어 후보 검색어 버튼을 눌러 샤오홍슈·도우인을 그 자리에서 검색·담기하는 60대 친화 UI를 붙인다.

**Architecture:** 비전이 프레임에서 중국어 후보 검색어 3~4개(+한국어 뜻)를 뽑는다(`cn_search_candidates`). 엔드포인트를 둘로 나눠 `/api/lens/cn/keywords`(비전, 후보만)와 `/api/lens/cn/search`(Apify, 검색어 1개)로 분리한다. 프론트는 후보를 큰 버튼으로 렌더하고 첫 후보를 자동검색, 나머지는 클릭 재검색.

**Tech Stack:** Python/FastAPI, google-genai(Gemini), Apify(zen-studio 액터), 순수 JS(index.html), pytest + node 슬라이스 JS 테스트.

_CN_CANDIDATES_PROMPT = """이 이미지는 한국어 쇼츠 영상의 한 장면(썸네일)이다. 화면에 박힌 글자\
(제품명·주제어)와 물건의 생김새를 아래 캡션과 종합해 '이 영상이 소개하는 바로 그 제품/소재'를 \
특정하라. 그 제품과 같은 영상을 중국 SNS(샤오홍슈/도우인)에서 찾을 **중국어 검색어 후보 3~4개**를 만들라.
- 넓은 제품+방식(예: 에어프라이어 감자칩→空气炸锅土豆片) → 좁은 제품명(예: 气泡土豆) 순으로 다양하게.
- 화면 글자에 제품명이 있으면 최우선 반영. 서로 다른 각도의 검색어(재료·조리법·모양)를 섞어라.
- 각 후보에 한국어 뜻(ko)을 짧게 달라(사용자가 뭘 누르는지 알게).
- JSON만: {{"product": "한국어 제품명(짧게)", \
"candidates": [{{"ko": "한국어 뜻", "zh": "중국어 검색어"}}, ...]}}
캡션: {caption}"""

_CN_CANDIDATES_SCHEMA = {
    "type": "object",
    "properties": {
        "product": {"type": "string"},
        "candidates": {
            "type": "array",
            "items": {
                "type": "object",
                "properties": {"ko": {"type": "string"}, "zh": {"type": "string"}},
                "required": ["ko", "zh"],
            },
            "minItems": 1, "maxItems": 4,
        },
    },
    "required": ["product", "candidates"],
}


def cn_search_candidates(image_bytes, caption, max_retries=3, quota_sleep=8):
    """프레임(+캡션) → {"product": str, "candidates": [{"ko","zh"}]}. 실패/키없음 시 빈 리스트."""
    empty = {"product": "", "candidates": []}
    if not image_bytes or not SHORTS_GEMINI_KEYS:
        return empty
    prompt = _CN_CANDIDATES_PROMPT.format(caption=(caption or "(캡션 없음)")[:400])
    for attempt in range(max_retries):
        key, idx = comment_gen._current_key_and_idx()
        if key is None:
            return empty
        try:
            client = _client_for_key(key)
            resp = client.models.generate_content(
                model=_MODEL,
                contents=[prompt, types.Part.from_bytes(data=image_bytes, mime_type="image/jpeg")],
                config=types.GenerateContentConfig(
                    response_mime_type="application/json",
                    response_schema=_CN_CANDIDATES_SCHEMA,
                ),
            )
            data = json.loads(resp.text)
            cands = []
            for c in (data.get("candidates") or []):
                ko, zh = (c.get("ko") or "").strip(), (c.get("zh") or "").strip()
                if zh:
                    cands.append({"ko": ko, "zh": zh})
            return {"product": (data.get("product") or "").strip(), "candidates": cands}
        except Exception as e:
            if key_vault.is_daily_exhausted_error(e) or key_vault.is_account_disabled_error(e):
                comment_gen._mark_key_exhausted(idx)
                continue
            if key_vault.is_quota_error(e):
                time.sleep(quota_sleep)
                continue
            if attempt < max_retries - 1 and any(c in str(e) for c in ("503", "UNAVAILABLE", "overloaded")):
                time.sleep((attempt + 1) * 5)
                continue
            return empty
    return empty
```

@app.post("/api/lens/cn/search")
async def api_lens_cn_search(request: Request, keyword: str = Form(""),
                              max_results: int = Form(8)):
    """중국어 검색어 1개 → 샤오홍슈+도우인 병렬 검색. Gemini 안 부름(후보는 사람이 고름).
    프론트가 후보 버튼 클릭 시 호출한다(2026-07-19)."""
    kw = (keyword or "").strip()
    if not kw:
        return {"ok": True, "items": [], "count": 0, "keyword": ""}
    if not APIFY_TOKENS:
        return {"ok": True, "items": [], "count": 0, "keyword": kw, "note": "APIFY 토큰 없음"}
    n = max(1, min(int(max_results or 8), 60))

    def _run(platform, mod):
        try:
            rows = mod.search(kw, max_results=n)
        except Exception:
            return []
        for r in rows:
            r["platform"] = platform
            r["match"] = None
        return rows

    with ThreadPoolExecutor(max_workers=2) as ex:
        fx = ex.submit(_run, "xiaohongshu", xiaohongshu_search)
        fd = ex.submit(_run, "douyin", douyin_search)
        items = fx.result() + fd.result()
    return {"ok": True, "items": items, "count": len(items), "keyword": kw}
```

`fetchLensYt` 함수 정의 **아래**에 신규 함수들 추가:

```javascript
// 순수 헬퍼: CN 검색 클릭 상한(비용 가드). 렌즈 1회당 6번까지.
function _lensCnCapReached(st){ return (st.cnClicks||0) >= 6; }
// ── CN후보 끝 ──

// 프레임(비전)으로 중국어 후보 검색어를 받아 버튼으로 그린다. 첫 후보는 자동검색.
async function fetchLensCnKeywords(shortcode, caption, blob){
  try{
    const fd=new FormData();
    if(blob) fd.append('frame', blob, 'f.jpg');
    if(caption) fd.append('source_caption', caption);
    const d=await (await fetch('/api/lens/cn/keywords',{method:'POST',body:fd})).json();
    const st=LENS_STATE[shortcode]; if(!st) return;
    st.cnCands=(d.candidates||[]).filter(c=>c && c.zh);
    renderLens(shortcode);
    if(st.cnCands.length) doLensCnSearch(shortcode, st.cnCands[0].zh);   // 첫 후보 자동
  }catch(e){ /* 후보 실패해도 렌즈 나머지는 정상 */ }
}

// 후보 검색어 하나로 샤오홍슈+도우인 검색 → CN 결과만 교체(유튜브 등은 유지).
async function doLensCnSearch(shortcode, zh){
  const st=LENS_STATE[shortcode]; if(!st) return;
  if(_lensCnCapReached(st)){ showToast('이번 검색은 여기까지예요 🙏'); return; }
  st.cnActive=zh; st.cnClicks=(st.cnClicks||0)+1; st.cnLoading=true; renderLens(shortcode);
  try{
    const fd=new FormData(); fd.append('keyword', zh); fd.append('max_results','12');
    const d=await (await fetch('/api/lens/cn/search',{method:'POST',body:fd})).json();
    const st2=LENS_STATE[shortcode]; if(!st2) return;
    // 기존 CN 결과 제거 후 새 결과로 교체
    st2.items=st2.items.filter(i=>i.platform!=='xiaohongshu' && i.platform!=='douyin');
    const add=(d.items||[]);
    add.forEach(i=>{ i.savesc='lens_'+i.platform+'_'+_lensHash(i.url); });
    const have=new Set(st2.items.map(i=>i.savesc));
    st2.items.push(...add.filter(i=>!have.has(i.savesc)));
    st2.cnLoading=false;
  }catch(e){ const s=LENS_STATE[shortcode]; if(s) s.cnLoading=false; }
  renderLens(shortcode);
}
```

    ${st.cnCands && st.cnCands.length ? `<div style="margin-top:8px;padding-top:8px;border-top:1px solid var(--line)">
      <div style="font-size:12px;color:var(--sub);margin-bottom:5px">📕🎬 중국 앱에서 찾기 — 눌러보세요</div>

        ${st.cnCands.map(c=>{

          return `<button onclick="doLensCnSearch('${shortcode}', ${JSON.stringify(c.zh)})"
            style="cursor:pointer;padding:8px 14px;border-radius:12px;border:2px solid ${on?'var(--accent)':'var(--line)'};

        }).join('')}
      </div>
    </div>` : ''}
