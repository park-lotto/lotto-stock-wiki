import pandas as pd
import yfinance as yf
from datetime import datetime

# Step 1: Define the tickers and their corresponding names for Korean ETFs
risky_assets = {
    "133690.KS": "TIGER 미국나스닥100",
    "143850.KS": "TIGER 미국S&P500선물(H)",
    "280930.KS": "KODEX 미국러셀2000(H)",
    "195980.KS": "ARIRANG 신흥국MSCI(합성 H)",
    "251350.KS": "KODEX 선진국MSCI World",
    "261220.KS": "KODEX WTI원유선물(H)",
    "182480.KS": "TIGER 미국MSCI리츠(합성 H)",
    "102110.KS": "TIGER 200"
}

safe_assets = {
    "304660.KS": "KODEX 미국채울트라30년선물(H)",
    "305080.KS": "TIGER 미국채10년선물",
    "148070.KS": "KOSEF 국고채10년",
    "261240.KS": "KODEX 미국달러선물",
    "130730.KS": "KOSEF 단기자금"
}

# Canary Asset (TIP)
canary_asset = {"TIP": "TIP 물가연동채"}

# Step 2: Download data (past 3 years)
tickers = list(risky_assets.keys()) + list(safe_assets.keys()) + list(canary_asset.keys())
end_date = datetime.today().strftime('%Y-%m-%d')
start_date = (datetime.today() - pd.DateOffset(years=3)).strftime('%Y-%m-%d')

data = yf.download(tickers, start=start_date, end=end_date)['Close']

# 월말 데이터(백테스트용)
monthly_data = data.resample('M').last()

# ─────────────────────────────────────────────────────────
# 모멘텀 계산 함수
def calculate_momentum(series):
    """
    단일 티커 series에 대해 1M,3M,6M,12M 모멘텀 평균을 구해 반환.
    월말 데이터에서 periods=1,3,6,12면
    '직전 1개월, 3개월, 6개월, 12개월 전 대비 수익률'의 평균
    """
    momentum = pd.DataFrame(index=series.index)
    momentum['1M']  = series.pct_change(periods=1, fill_method=None)
    momentum['3M']  = series.pct_change(periods=3, fill_method=None)
    momentum['6M']  = series.pct_change(periods=6, fill_method=None)
    momentum['12M'] = series.pct_change(periods=12, fill_method=None)
    return momentum.mean(axis=1)

momentum_data = monthly_data.apply(calculate_momentum)

# ─────────────────────────────────────────────────────────
# TIP 신호를 이용한 카나리아 모멘텀 계산
def canary_momentum(tip_series):
    tip_mom = calculate_momentum(tip_series)
    return (tip_mom > 0), tip_mom

canary_signal, tip_momentum = canary_momentum(monthly_data['TIP'])

# ─────────────────────────────────────────────────────────
# 안전 자산 선택
def select_safe_assets(momentum_df, bil_momentum, top_n=2):
    """
    safe_assets 중 모멘텀 상위 top_n 개를 뽑되,
    bil_momentum(=단기자금 모멘텀)보다 모멘텀 낮으면 제외.
    최소 1개는 '130730.KS'(단기자금) 유지.
    """
    safe_mom = momentum_df[safe_assets.keys()].iloc[-1]
    top_safe_assets = safe_mom.nlargest(top_n)
    selected_assets = []
    for asset in top_safe_assets.index:
        if top_safe_assets[asset] > bil_momentum:
            selected_assets.append(asset)
    # 최소 1개는 '130730.KS'
    if not selected_assets:
        return ['130730.KS']
    if '130730.KS' not in selected_assets:
        selected_assets.append('130730.KS')
    return selected_assets[:top_n]

# ─────────────────────────────────────────────────────────
# 공격 자산 선택
def select_top_assets(momentum_df, canary_signal_series, top_n=4):
    """
    카나리아 신호(True) 시 공격자산 상위 top_n, 
    False 시 안전자산(select_safe_assets)
    """
    if canary_signal_series.iloc[-1]:  # TIP 모멘텀이 양수면 공격
        top_assets = momentum_df.iloc[-1].nlargest(top_n)
        return top_assets.index.tolist()
    else:
        bil_momentum = calculate_momentum(data['130730.KS']).iloc[-1]
        return select_safe_assets(momentum_df, bil_momentum, top_n=2)

# ─────────────────────────────────────────────────────────
# (A) "이번 달 말 => 다음 달 말" 구조의 백테스트 실행
#     예) 11월 말까지 데이터로 자산 선정 -> 12월 한 달 보유 -> 12월 말 수익률 기록
# ─────────────────────────────────────────────────────────

portfolio_data = []

dates = monthly_data.index  # 월말 인덱스
# 1) 처음 12개월 전에는 12M 모멘텀이 정확치 않으므로 i=12부터 시작
# 2) 마지막 달은 (다음 달 X) 수익률 계산 불가하므로 (len(dates) - 1)까지만
for i in range(12, len(dates) - 1):
    # ① 이번 달 말 (모멘텀 계산 & 투자 결정 시점)
    decision_date = dates[i]
    # ② 다음 달 말 (실제로 한 달간 보유 후 수익률을 측정할 시점)
    invest_date = dates[i + 1]

    # (1) decision_date까지의 데이터로 모멘텀/카나리아 계산
    decision_data = monthly_data.loc[:decision_date]

    current_canary_signal, current_tip_momentum = canary_momentum(decision_data['TIP'])
    decision_momentum = decision_data.apply(calculate_momentum)

    # (2) 자산 선정
    top_assets = select_top_assets(decision_momentum, current_canary_signal, top_n=4)
    weight = 1.0 / len(top_assets)

    # (3) 다음 달 말(invest_date)에 월간 수익률 계산
    #     => (decision_date 달) ~ (invest_date 달) 한 달 변동률
    monthly_return = 0.0
    for asset in top_assets:
        if asset in monthly_data.columns:
            # invest_date에 대한 pct_change 값:
            #   "직전 월말 대비 invest_date 월말" 변동률
            ret = monthly_data[asset].pct_change().loc[invest_date]
            monthly_return += (weight * ret)

    # (4) 포트폴리오 기록 (수익률은 invest_date에 실현)
    portfolio_data.append({
        "Date": invest_date.strftime('%Y-%m-%d'), 
        "Assets": ', '.join(top_assets),
        "Monthly Return": round(monthly_return * 100, 2),
        # TIP Momentum은 'decision_date' 기준 (의사결정 시점)
        "TIP Momentum": round(current_tip_momentum.loc[decision_date] * 100, 2)
    })

# (B) 포트폴리오 DataFrame
portfolio = pd.DataFrame(portfolio_data)

# ─────────────────────────────────────────────────────────
# (C) CAGR, MDD 계산
portfolio['Cumulative'] = (1 + portfolio['Monthly Return'] / 100).cumprod()

start_dt = datetime.strptime(portfolio['Date'].iloc[0], '%Y-%m-%d')
end_dt   = datetime.strptime(portfolio['Date'].iloc[-1], '%Y-%m-%d')
years = (end_dt - start_dt).days / 365.25

cagr = portfolio['Cumulative'].iloc[-1] ** (1 / years) - 1

def calculate_mdd(cumulative_returns):
    cum_max = cumulative_returns.cummax()
    dd = cumulative_returns / cum_max - 1
    return dd.min()

mdd = calculate_mdd(portfolio['Cumulative'])

# ─────────────────────────────────────────────────────────
# (D) 결과 출력
print("\n월별 투자 자산 및 수익률:")
print(portfolio[['Date', 'Assets', 'Monthly Return', 'TIP Momentum']])

print(f"\n최근 3년간 한국형 HAA 전략 (월말→다음달말) 백테스트 결과:")
print(f"CAGR(연평균 성장률): {cagr * 100:.2f}%")
print(f"MDD(최대 낙폭): {mdd * 100:.2f}%")

if len(portfolio) > 1:
    last_month_date = portfolio['Date'].iloc[-2]
    current_month_date = portfolio['Date'].iloc[-1]
    print(f"\n점검: {last_month_date} 시점 결정 → {current_month_date}까지 한 달간 보유 후 수익률 기록")
