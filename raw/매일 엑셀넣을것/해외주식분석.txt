import yfinance as yf
import matplotlib.pyplot as plt
from datetime import datetime, timedelta
import pandas as pd

# 기본 폰트 설정 (NanumGothic 설정을 제거하여 경고 방지)
plt.rcParams['font.family'] = 'sans-serif'  # 기본 폰트로 설정

# 사용자가 여러 티커를 쉼표로 구분하여 입력
tickers_input = input("Enter stock tickers separated by commas (e.g., AAPL, MSFT, TSLA): ")
tickers = [ticker.strip() for ticker in tickers_input.split(",")]

# 티커 입력 및 결과 출력 반복
for ticker in tickers:
    # Yahoo Finance에서 데이터 다운로드
    stock = yf.Ticker(ticker)
    info = stock.info
    company_name = info.get('longName', ticker)  # Use company name if available, otherwise use ticker
    current_price = info.get('currentPrice', None)

    # PER TTM 및 Forward PER 계산
    trailing_eps = info.get('trailingEps', None)
    forward_eps = info.get('forwardEps', None)

    # PER 계산 (TTM 및 12개월 예상)
    per_ttm = None
    forward_per = None
    if trailing_eps and current_price:
        per_ttm = current_price / trailing_eps
    if forward_eps and current_price:
        forward_per = current_price / forward_eps

    # PER 관련 데이터 출력
    print(f"\n{company_name} ({ticker}) - PER Data:")
    print(f"PER TTM (Trailing 12 Months PER): {per_ttm:.2f} (Price: {current_price}, EPS: {trailing_eps})" if per_ttm else "PER TTM data not available.")
    print(f"Forward PER (Next 12 Months Expected PER): {forward_per:.2f} (Price: {current_price}, Expected EPS: {forward_eps})" if forward_per else "Forward PER data not available.")

    # 재무 데이터 (과거 5개 분기) 정리
    financials = stock.quarterly_financials

    # 필요한 항목만 선택 (5개 분기만 선택)
    financials_filtered = financials.loc[['Total Revenue', 'Gross Profit', 'Operating Income', 'Net Income']].iloc[:, :5]

    # 가로 방향으로 출력 (transpose)
    financials_transposed = financials_filtered.T.dropna()

    print(f"\n{company_name} ({ticker}) - Last 5 Quarters Financial Data (NaN Removed):")
    print(financials_transposed)

    # 매출액, 영업이익률, 매출총이익률 계산
    revenue = financials.loc['Total Revenue'].dropna().iloc[:5]
    operating_income = financials.loc['Operating Income'].dropna().iloc[:5]
    gross_profit = financials.loc['Gross Profit'].dropna().iloc[:5]

    # 매출총이익률과 영업이익률 계산 (NaN 제거 후)
    gross_profit_margin = (gross_profit / revenue) * 100
    operating_profit_margin = (operating_income / revenue) * 100

    print("\nGross Profit Margin (%) (NaN Removed):")
    print(gross_profit_margin.T.dropna())  # Transpose and print

    print("\nOperating Profit Margin (%) (NaN Removed):")
    print(operating_profit_margin.T.dropna())  # Transpose and print

    # 애널리스트 추천 데이터 출력
    try:
        recommendations = stock.recommendations
        target_price = info.get('targetMeanPrice', 'Data not available')
        print(f"\n{company_name} ({ticker}) - Analyst Recommendations and Target Price:")
        print(f"Target Mean Price: {target_price}")
        print(f"Analyst Recommendations (Most Recent):\n{recommendations.tail()}")
    except:
        print("\nFailed to retrieve analyst recommendations and EPS estimate data.")

    # 현재 날짜로부터 5년 전 날짜 계산
    end_date = datetime.today().strftime('%Y-%m-%d')
    start_date = (datetime.today() - timedelta(days=5*365)).strftime('%Y-%m-%d')

    # 최근 5년치 주가 데이터를 가져오기
    data = stock.history(start=start_date, end=end_date)

    # 주가(종가) 그래프 그리기
    plt.figure(figsize=(10, 6))
    plt.plot(data.index, data['Close'], label=f'{company_name} ({ticker}) Stock Price')
    plt.title(f'{company_name} ({ticker}) - 5 Year Stock Price')
    plt.xlabel('Date')
    plt.ylabel('Stock Price (Close)')
    plt.legend()
    plt.grid(True)
    plt.show(block=False)  # block=False를 통해 각 티커 이후에도 코드가 진행됨

# 마지막으로 plt.show() 호출하여 모든 그래프 창이 닫히도록 설정
plt.show()
