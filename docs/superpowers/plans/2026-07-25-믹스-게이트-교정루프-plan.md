# 믹스 게이트 교정 루프 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** `plan_gate`를 경고→교정 관문으로 바꿔 장면 반복·연속재생 쳇바퀴를 종결하고, 끝 자막 겹침을 고친다.

**Architecture:** `_plan_and_tts`의 1회성 게이트 검사를 재검사 루프(상한 3회)로. 위반이면 `repick_for_gate`(신규 순수함수)가 primary 연속 런을 비연속·미사용소스로 스왑 + 기존 dedup 재적용. 못 고친 잔여는 프론트가 역할별로 표시. 끝 자막은 headcopy가 마지막 비트와 안 겹치게 + CTA를 여운까지 홀드.

**Tech Stack:** Python 3, pytest, ffmpeg(자막), 순수함수(무과금·나레이션불변).

```python
def repick_for_gate(beats, pool_sources, gate):
    """게이트 위반(연속·반복·파편)을 재픽으로 교정한다. 원본 mutate 안 함, Gemini/IO 없음,
    나레이션·tts_path 불변. 후보 없으면 그 위반은 그대로 둠(루프가 수렴 판정으로 종료).
    ★primary 비트 간 연속(is_continuous)을 여기서 처음으로 끊는다(dedup_*는 alternates만 봄)."""
    out = [dict(b) for b in beats]
    # 1) primary 연속 런 끊기 — 앞 비트 primary와 연속인 primary를 비연속·미사용소스 우선으로 스왑
    src_count = Counter(_vid_of(b.get("primary")) for b in out if b.get("primary"))
    used = {(b.get("primary") or {}).get("seg_id") for b in out}
    used |= {a.get("seg_id") for b in out for a in (b.get("alternates") or [])}
    for i in range(1, len(out)):
        prev_p = out[i - 1].get("primary") or {}
        cur_p = out[i].get("primary") or {}
        if not is_continuous(prev_p, cur_p):
            continue
        # 포인트 비트 primary는 불가침 → 앞 비트를 바꿔 연속을 깬다
        target = i - 1 if is_point_beat(out[i]) and not is_point_beat(out[i - 1]) else i
        tb = out[target]
        anchor = out[target - 1].get("primary") if target > 0 else None
        action = segment_action(tb.get("primary") or {}) or action_dict.tag_action(tb.get("narration", ""))
        # 같은 행위 + 비연속 후보(미사용 소스·seg 우선)
        cands = [c for c in (pick_clips_for_action(action, pool_sources) if action else [])
                 if c.get("seg_id") not in used and not is_continuous(anchor, c)]
        if not cands:
            cands = [c for c in _broll_segs(pool_sources, src_count, used)
                     if not is_continuous(anchor, c)]
        if not cands:
            continue  # 후보 없음 → 그대로(수렴)
        cands.sort(key=lambda c: (src_count.get(_vid_of(c), 0), -_seg_dur(c)))  # 미사용소스·긴컷 우선
        pick = cands[0]
        used.discard(tb["primary"].get("seg_id")); used.add(pick.get("seg_id"))
        src_count[_vid_of(pick)] += 1
        tb["primary"] = pick
    # 2) 반복·파편·alternates 연속은 기존 dedup 재적용(테스트된 로직 재사용)
    out = dedup_and_balance(out, pool_sources)
    out = dedup_clips_global(out, pool_sources)
    return out
```

```python
_MAX_REPICK = 3
# 재픽으로 고칠 수 있는 위반의 신호어(생성영역=길이·비트수는 제외)
_REPICKABLE_HINTS = ("이어지는 구간", "같은 장면", "잘게 쪼개진", "1개만 사용")

def _has_repickable(gate):
    return any(any(h in v for h in _REPICKABLE_HINTS) for v in (gate.get("violations") or []))

def _run_gate_correction(plan, source_scripts, target_seconds):
    """게이트 검사→재픽 루프. 최종 gate를 plan["gate"]에 항상 저장(순수·무과금·나레이션불변)."""
    from shopping_shorts import backbone, plan_gate
    pool_ct = len({s.get("video_id") for s in (source_scripts or []) if s.get("segments")} - {None})
    beats = plan.get("beats")
    gate = plan_gate.check_plan(beats, target_seconds, pool_ct)
    rounds = 0
    while rounds < _MAX_REPICK and _has_repickable(gate):
        new_beats = backbone.repick_for_gate(beats, source_scripts, gate)
        if new_beats == beats:
            break  # 더 못 고침 → 종료
        beats = new_beats
        rounds += 1
        gate = plan_gate.check_plan(beats, target_seconds, pool_ct)
    plan["beats"] = beats
    plan["gate"] = gate
    plan["repick_rounds"] = rounds
    if not gate["ok"]:
        print("plan_gate 잔여 위반: " + " / ".join(gate["violations"]), file=sys.stderr)
```
