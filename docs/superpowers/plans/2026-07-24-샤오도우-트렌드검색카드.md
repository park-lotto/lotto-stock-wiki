```python
    def get_translation(self, ko):
        """한국어 소재 → 저장된 중국어. 없으면 None, 번역실패로 빈값 저장됐으면 ""."""
        if not ko:
            return None
        with self._conn() as c:
            row = c.execute("SELECT zh FROM translations WHERE ko=?", (ko,)).fetchone()
        return None if row is None else (row[0] or "")

    def save_translation(self, ko, zh):
        """번역 저장(덮어쓰기). zh 빈 문자열도 저장 — 반복 호출 방지."""
        if not ko:
            return
        with self._conn() as c:
            c.execute(
                "INSERT INTO translations(ko, zh, updated_at) VALUES(?,?,datetime('now')) "
                "ON CONFLICT(ko) DO UPDATE SET zh=excluded.zh, updated_at=excluded.updated_at",
                (ko, zh or ""),
            )

    def translations_map(self, kos):
        """여러 ko → {ko: zh} (있는 것만). sqlite 변수상한 회피 청크."""
        keys = [k for k in (kos or []) if k]
        if not keys:
            return {}
        out = {}
        with self._conn() as c:
            for i in range(0, len(keys), 400):
                chunk = keys[i:i + 400]
                q = "SELECT ko, zh FROM translations WHERE ko IN (%s)" % ",".join("?" * len(chunk))
                for k, zh in c.execute(q, chunk).fetchall():
                    out[k] = zh or ""
        return out
```

def api_translate(q: str = ""):
    """트렌드카드용 한→중 소재 번역(캐시 우선). 미스면 translate_keyword 1회 후 저장.

    ko = (q or "").strip()
    if not ko:
        return {"ok": True, "ko": "", "zh": ""}
    if len(ko) > _TRANSLATE_MAXLEN:
        return {"ok": False, "ko": ko, "zh": "", "error": "too_long"}
    store = Store(DB_PATH)
    cached = store.get_translation(ko)

        return {"ok": True, "ko": ko, "zh": cached}
    from shopping_shorts import video_analysis
    zh = ""

    except Exception:
        zh = ""
    store.save_translation(ko, zh)   # 빈 zh도 저장 → 반복 호출 방지
    return {"ok": True, "ko": ko, "zh": zh}

def _translate_new_subjects(items):
    """수집 직후 백그라운드: 새 vision_subject를 한→중 번역해 translations 캐시에 채운다.

    try:
        from shopping_shorts import video_analysis
    except Exception:
        return
    store = Store(DB_PATH)
    subjects = []
    seen = set()
    for it in items:
        s = (it.get("vision_subject") or "").strip()
        if s and s not in seen:
            seen.add(s)
            subjects.append(s)
    have = store.translations_map(subjects)
    todo = [s for s in subjects if s not in have]
    for s in todo[:_TRANSLATE_CAP]:

        except Exception:
            zh = ""
        store.save_translation(s, zh)
    if len(todo) > _TRANSLATE_CAP:
        print(f"[translations] {len(todo) - _TRANSLATE_CAP}건 다음 수집으로 미룸(상한 {_TRANSLATE_CAP})")

// ── 트렌드 검색카드(2026-07-24) ──────────────────────
// 인스·틱·유튜 상위 통합에서 소재를 뽑아 샤오·도우 딥링크 카드로. 순수함수는 node 테스트됨.
const _TREND_STOP = new Set(['만들기','완성','레시피','털기','또','집에서','하는','추천','리뷰','종','신상털기']);
function _titleKeyword(title){
  // 제목에서 핵심 명사구 근사: 이모지·해시태그·괄호 제거 후 불용어 밖 최장 토큰.
  let t = String(title||'').replace(/[#\[\](){}!?~…·|]/g,' ')
            .replace(/[\u{1F000}-\u{1FFFF}☀-➿]/gu,' ');
  const toks = t.split(/\s+/).map(s=>s.trim()).filter(Boolean).filter(w=>!_TREND_STOP.has(w));
  if(!toks.length) return '';
  // 가장 긴 토큰(정보량 근사). 동률이면 앞선 것.
  return toks.reduce((a,b)=> b.length>a.length ? b : a, toks[0]);
}
function _extractSubject(item){
  const vs = (item.vision_subject||'').trim();
  if(vs) return vs;
  return _titleKeyword(item.title||'');
}
function _normSubjectKey(s){
  // 묶기 키: 공백 제거·소문자·흔한 조사 꼬리 제거(근사).
  return String(s||'').toLowerCase().replace(/\s+/g,'')
           .replace(/(을|를|이|가|은|는|의|도|에|와|과|만)$/,'');
}
function _score(item){
  const v = item.views;
  if(v!=null && !isNaN(v)) return Number(v);
  const l = item.likes;
  return (l!=null && !isNaN(l)) ? Number(l) : 0;   // 좋아요는 보정(가중 1로 단순화)
}
function _groupBySubject(items){
  const map = new Map();   // key → {subjectKo, items, platforms:Set, maxScore}
  for(const it of (items||[])){
    const subj = _extractSubject(it);
    if(!subj) continue;                    // 소재 없으면 제외
    const key = _normSubjectKey(subj);
    if(!key) continue;
    let g = map.get(key);
    if(!g){ g = {subjectKo:subj, items:[], platforms:new Set(), maxScore:0}; map.set(key,g); }
    g.items.push(it);
    if(it.platform) g.platforms.add(it.platform);
    g.maxScore = Math.max(g.maxScore, _score(it));
  }
  const groups = [...map.values()].map(g=>({
    subjectKo:g.subjectKo, items:g.items, platforms:[...g.platforms],
    count:g.items.length, maxScore:g.maxScore,
  }));
  groups.sort((a,b)=> b.maxScore - a.maxScore);
  return groups;
}
