# ----------------------------
# 5. Mansfield RS 계산 함수 (다중 윈도우 지원)
# ----------------------------
def compute_mansfield_rs(price_series, benchmark_series, ma_period):
    relative = price_series / benchmark_series
    ma = relative.rolling(window=ma_period, min_periods=ma_period).mean()
    return ((relative / ma) - 1) * 100

def normalize_to_100_scale(x, scale=12):
    return 100 * (1 / (1 + np.exp(-x / scale)))

# ----------------------------
# 6. RS 계산 실행 (60d / 120d / 250d + 평균)
# ----------------------------
RS_WINDOWS = [60, 120, 250]
mansfield_rows = []

for ticker in etf_list:
    if ticker == "SPY":
        continue
    if ticker not in data.columns:
        continue

    try:
        price = data[ticker].dropna()
        aligned = pd.concat([price, spy], axis=1, join='inner')
        aligned.columns = ['etf', 'spy']

        latest = {}
        for win in RS_WINDOWS:
            if len(aligned) >= win:
                rs_series = compute_mansfield_rs(aligned['etf'], aligned['spy'], ma_period=win).dropna()
                if not rs_series.empty:
                    raw = float(rs_series.iloc[-1])
                    latest[f"RS_{win}d"] = round(raw, 2)
                    latest[f"Norm_RS_{win}d"] = round(normalize_to_100_scale(raw), 2)

        raw_keys = [k for k in latest if k.startswith("RS_") and k.endswith("d")]
        if not raw_keys:
            continue

        raw_values = [latest[k] for k in raw_keys]
        rs_avg = round(sum(raw_values) / len(raw_values), 2)
        norm_rs_avg = round(normalize_to_100_scale(rs_avg), 2)

        row = {
            "Ticker": ticker,
            "Industry": industry_etfs.get(ticker, ""),
            "RS_60d": np.nan, "RS_120d": np.nan, "RS_250d": np.nan, "RS_avg": rs_avg,
            "Norm_RS_60d": np.nan, "Norm_RS_120d": np.nan, "Norm_RS_250d": np.nan, "Norm_RS_avg": norm_rs_avg
        }
        for win in RS_WINDOWS:
            if f"RS_{win}d" in latest:
                row[f"RS_{win}d"] = latest[f"RS_{win}d"]
                row[f"Norm_RS_{win}d"] = latest[f"Norm_RS_{win}d"]

        mansfield_rows.append(row)

    except Exception:
        continue

mansfield_df = pd.DataFrame(mansfield_rows).sort_values(by="Norm_RS_avg", ascending=False)

# 7. 변동성 조정 모멘텀 계산 함수 (+ 이동평균선 이탈 필터)
#    ✅ FutureWarning 제거: pct_change(fill_method=None)
#    ✅ vol=0 안전장치
# ----------------------------
def calculate_risk_adjusted_momentum(
    data,
    period_3m=63,
    period_6m=126,
    period_12m=252,
    last_n_days=10,
    ma_filter_days=None,
    top_n=20
):
    """
    3/6/12개월 '일간 평균 수익률'을 각 기간 표준편차로 나눠 평균한 위험조정 모멘텀 점수.
    ma_filter_days가 주어지면 해당 이동평균 하회 종목 제외.
    """
    daily_top_industries = {}
    trading_days = data.index[-last_n_days:]

    for day in trading_days:
        end_idx = data.index.get_loc(day)
        start_3m = max(0, end_idx - period_3m + 1)
        start_6m = max(0, end_idx - period_6m + 1)
        start_12m = max(0, end_idx - period_12m + 1)

        # ✅ FutureWarning 방지
        r3 = data.iloc[start_3m:end_idx + 1].pct_change(fill_method=None).dropna()
        r6 = data.iloc[start_6m:end_idx + 1].pct_change(fill_method=None).dropna()
        r12 = data.iloc[start_12m:end_idx + 1].pct_change(fill_method=None).dropna()

        returns_3m = r3.mean()
        returns_6m = r6.mean()
        returns_12m = r12.mean()

        vol_3m = r3.std()
        vol_6m = r6.std()
        vol_12m = r12.std()

        # ✅ 0 변동성 방지
        vol_3m = vol_3m.replace(0, np.nan)
        vol_6m = vol_6m.replace(0, np.nan)
        vol_12m = vol_12m.replace(0, np.nan)

        risk_adj_3m = returns_3m / vol_3m
        risk_adj_6m = returns_6m / vol_6m
        risk_adj_12m = returns_12m / vol_12m

        avg_risk_adj_momentum = (risk_adj_3m + risk_adj_6m + risk_adj_12m) / 3

        # 이동평균선 이탈 필터
        if ma_filter_days is not None:
            below_ma = []
            for t in data.columns:
                series = data[t].iloc[:end_idx + 1]
                if len(series) < ma_filter_days:
                    below_ma.append(True)
                else:
                    ma_value = series.rolling(window=ma_filter_days).mean().iloc[-1]
                    price_value = series.iloc[-1]
                    below_ma.append(price_value < ma_value)
            below_ma = pd.Series(below_ma, index=data.columns)
            avg_risk_adj_momentum = avg_risk_adj_momentum[~below_ma]

        # ✅ NaN 제거 후 TopN
        avg_risk_adj_momentum = avg_risk_adj_momentum.dropna()
        top_list = avg_risk_adj_momentum.nlargest(top_n).index.tolist()
        top_names = [f"{t} ({industry_etfs.get(t, '')})" for t in top_list]

        # 혹시 부족하면 None 채우기(표 모양 유지)
        top_names += [None] * max(0, top_n - len(top_names))
        daily_top_industries[day] = top_names

    df = pd.DataFrame.from_dict(
        daily_top_industries,
        orient='index',
        columns=[f'Top {i+1}' for i in range(top_n)]
    )
    df.index.name = 'Date'
    return df

def sortino_from_returns(ret: pd.Series, mar: float = 0.0, periods_per_year: int = 252) -> float:
    """Sortino Ratio, annualized."""
    ret = ret.dropna()
    if len(ret) < 5:
        return np.nan

    mar_daily = (1 + mar) ** (1 / periods_per_year) - 1
    excess = ret - mar_daily
    downside = excess[excess < 0]

    if len(downside) < 5:
        return np.nan

    downside_dev = downside.std(ddof=1) * np.sqrt(periods_per_year)
    if downside_dev == 0 or np.isnan(downside_dev):
        return np.nan

    annual_excess = excess.mean() * periods_per_year
    return float(annual_excess / downside_dev)


def ma_ok_mask(close_df: pd.DataFrame, end_idx: int, ma_days: int) -> pd.Series:
    """end_idx 시점에서 종가 >= MA(ma_days)이면 True."""
    ok = {}
    for t in close_df.columns:
        s = close_df[t].iloc[:end_idx + 1].dropna()
        if len(s) < ma_days:
            ok[t] = False
        else:
            ma = s.rolling(ma_days).mean().iloc[-1]
            ok[t] = bool(s.iloc[-1] >= ma)
    return pd.Series(ok)


def rank_sortino_ma_filter(
    close_df: pd.DataFrame,
    period_3m: int = 63,
    period_6m: int = 126,
    period_12m: int = 252,
    last_n_days: int = 10,
    ma_filter_days: int = 10,
    top_n: int = 20,
    periods_per_year: int = 252,
    mar: float = 0.0,
):
    """
    3M / 6M / 12M Sortino를 각각 계산한 뒤 평균 점수로 랭킹.
    ma_filter_days가 주어지면 해당 날짜 기준 종가 >= 이동평균선 종목만 통과.
    """
    daily_top = {}
    trading_days = close_df.dropna(how="all").index[-last_n_days:]
    wins = [period_3m, period_6m, period_12m]

    for day in trading_days:
        end_idx = close_df.index.get_loc(day)
        ok = ma_ok_mask(close_df, end_idx=end_idx, ma_days=ma_filter_days)

        scores = {}
        for t in close_df.columns:
            if not ok.get(t, False):
                continue

            vals = []
            for win in wins:
                start_idx = max(0, end_idx - win + 1)
                w_prices = close_df[t].iloc[start_idx:end_idx + 1].dropna()

                if len(w_prices) < max(20, int(win * 0.8)):
                    continue

                r = w_prices.pct_change(fill_method=None).dropna()
                v = sortino_from_returns(r, mar=mar, periods_per_year=periods_per_year)

                if pd.notna(v):
                    vals.append(float(v))

            if len(vals) > 0:
                scores[t] = float(np.mean(vals))

        if len(scores) == 0:
            daily_top[day] = [None] * top_n
            continue

        ranked = pd.Series(scores).sort_values(ascending=False).head(top_n).index.tolist()
        row = [f"{x} ({industry_etfs.get(x, '')})" for x in ranked]
        row += [None] * max(0, top_n - len(row))
        daily_top[day] = row

    out = pd.DataFrame.from_dict(
        daily_top,
        orient="index",
        columns=[f"Top {i+1}" for i in range(top_n)]
    )
    out.index.name = "Date"
    return out

#     30일 초과 이격도 + 최근 247거래일 97% 초과 이격도

# - 97구간대비현재위치 = 현재30일초과이격도 / 최근247거래일97구간초과이격도 * 100
#
# 표는 한글, 그래프는 영문 제목만 사용해서 Matplotlib 한글 폰트 경고를 방지

def _extract_ticker_from_label(x):
    """'SOXX (Semiconductors)' 형태에서 SOXX만 추출."""
    if pd.isna(x):
        return None
    return str(x).split(" ")[0].strip()


def build_sortino_ma10_gap247_p97_summary(
    close_df: pd.DataFrame,
    sortino_table: pd.DataFrame,
    industry_map: dict,
    ma_days: int = 30,
    lookback_days: int = 247,
    percentile: float = 0.97,
    top_n: int = 20,
):
    latest_row = sortino_table.iloc[-1].dropna().tolist()
    tickers = [_extract_ticker_from_label(x) for x in latest_row]
    tickers = [x for x in tickers if x in close_df.columns]

    if top_n is not None:
        tickers = tickers[:top_n]

    rows = []

    for rank_no, ticker in enumerate(tickers, start=1):
        price = close_df[ticker].dropna()
        if len(price) < ma_days + 20:
            continue

        ma30 = price.rolling(ma_days).mean()
        gap_excess = (price / ma30 - 1) * 100
        gap_excess = gap_excess.dropna()

        if len(gap_excess) < 20:
            continue

        hist = gap_excess.tail(lookback_days)
        current_excess = float(gap_excess.iloc[-1])
        p97_excess = float(hist.quantile(percentile))

        if pd.notna(p97_excess) and p97_excess != 0:
            heat_ratio = current_excess / p97_excess * 100
        else:
            heat_ratio = np.nan

        rows.append({
            "Sortino순위": rank_no,
            "Ticker": ticker,
            "Industry": industry_map.get(ticker, ""),
            "Date": gap_excess.index[-1].date(),
            "현재30일초과이격도": round(current_excess, 2),
            "최근247거래일97구간초과이격도": round(p97_excess, 2),
            "97구간대비현재위치": round(heat_ratio, 1) if pd.notna(heat_ratio) else np.nan,
            "사용거래일수": len(hist),
        })

    out = pd.DataFrame(rows)

    if not out.empty:
        out = out[[
            "Sortino순위",
            "Ticker",
            "Industry",
            "Date",
            "현재30일초과이격도",
            "최근247거래일97구간초과이격도",
            "97구간대비현재위치",
            "사용거래일수",
        ]]

    return out


def plot_sortino_ma10_gap247_p97_graphs(
    close_df: pd.DataFrame,
    summary_df: pd.DataFrame,
    ma_days: int = 30,
    lookback_days: int = 247,
    figsize=(16, 4),
    bar_alpha: float = 0.15,
):
    if summary_df.empty:
        print("그래프로 출력할 ETF가 없습니다.")
        return

    for _, row in summary_df.iterrows():
        ticker = row["Ticker"]
        industry = row["Industry"]

        price = close_df[ticker].dropna()
        ma30 = price.rolling(ma_days).mean()
        gap_excess = (price / ma30 - 1) * 100

        plot_df = pd.DataFrame({
            "Price": price,
            "MA30": ma30,
            "GapExcess30": gap_excess,
        }).dropna()

        if plot_df.empty:
            continue

        fig, ax1 = plt.subplots(figsize=figsize)
        ax2 = ax1.twinx()

        # 오른쪽 축: 30일 초과 이격도 막대
        ax2.bar(
            plot_df.index,
            plot_df["GapExcess30"],
            alpha=bar_alpha,
            width=1.5,
            label="GapExcess30",
            zorder=1,
        )

        # 97% 구간 기준선
        p97 = row["최근247거래일97구간초과이격도"]
        if pd.notna(p97):
            ax2.axhline(
                p97,
                linestyle="--",
                linewidth=1.2,
                label="P97_247D",
                zorder=2,
            )

        # 왼쪽 축을 앞으로 가져오기
        ax1.set_zorder(ax2.get_zorder() + 1)
        ax1.patch.set_visible(False)

        # 왼쪽 축: 가격 + MA30
        ax1.plot(
            plot_df.index,
            plot_df["Price"],
            linewidth=2.8,
            label=ticker,
            zorder=10,
        )

        ax1.plot(
            plot_df.index,
            plot_df["MA30"],
            linewidth=2.4,
            label="MA30",
            zorder=9,
        )

        ax1.grid(True, alpha=0.25, zorder=0)

        # ------------------------------------------------------------
        # 제목/수치 정보는 그래프 안쪽 좌측 상단에 표시
        # - 범례와 제목이 겹치지 않도록 set_title은 사용하지 않음
        # - 그래프 텍스트는 영문만 사용해서 Matplotlib 한글 폰트 경고 방지
        # ------------------------------------------------------------
        info_text_1 = f"#{int(row['Sortino순위'])} {ticker} | {industry}"
        info_text_2 = (
            f"GapExcess30={row['현재30일초과이격도']:.2f} | "
            f"P97_247D={row['최근247거래일97구간초과이격도']:.2f} | "
            f"Heat={row['97구간대비현재위치']:.1f}"
        )

        ax1.text(
            0.01,
            0.96,
            info_text_1,
            transform=ax1.transAxes,
            fontsize=12,
            fontweight="bold",
            va="top",
            ha="left",
            bbox=dict(facecolor="white", alpha=0.75, edgecolor="none"),
            zorder=20,
        )

        ax1.text(
            0.01,
            0.88,
            info_text_2,
            transform=ax1.transAxes,
            fontsize=10.5,
            fontweight="bold",
            va="top",
            ha="left",
            bbox=dict(facecolor="white", alpha=0.75, edgecolor="none"),
            zorder=20,
        )

        ax1.set_ylabel("Price")
        ax2.set_ylabel("Gap excess")

        lines1, labels1 = ax1.get_legend_handles_labels()
        lines2, labels2 = ax2.get_legend_handles_labels()

        # 범례는 그래프 위쪽에 한 줄로 배치하되, 제목을 없앴기 때문에 겹치지 않음
        ax1.legend(
            lines2 + lines1,
            labels2 + labels1,
            loc="upper center",
            bbox_to_anchor=(0.5, 1.10),
            ncol=4,
            frameon=False,
            fontsize=10,
        )

        plt.tight_layout()
        plt.show()
