import pandas as pd
import yfinance as yf
from datetime import datetime, timedelta

# ============== 1) 티커 설정 ==============
risky_assets = {
    "418660.KS": "TIGER 미국나스닥100레버리지(합성)",
    "225040.KS": "TIGER 미국S&P500레버리지(합성 H)",
    "280930.KS": "KODEX 미국러셀2000(H)",
    "241180.KS": "TIGER 일본니케이225",
    "251350.KS": "KODEX 선진국MSCI World",
    "261220.KS": "KODEX WTI원유선물(H)",
    "182480.KS": "TIGER 미국MSCI리츠(합성 H)",
    "229200.KS": "KODEX 코스닥 150",
    "069500.KS": "KODEX 200",
}

safe_assets = {
    "304660.KS": "KODEX 미국채울트라30년선물(H)",
    "305080.KS": "TIGER 미국채10년선물",
    "148070.KS": "KOSEF 국고채10년",
    "261240.KS": "KODEX 미국달러선물",
    "130730.KS": "KOSEF 단기자금"
}

canary_asset = {"TIP": "TIP 물가연동채"}

# 고정 자산(예시): 합계 35%
fixed_assets = {
    "225040.KS": 10,  # TIGER 미국S&P500레버리지(합성 H)
    "418660.KS": 10,  # TIGER 미국나스닥100레버리지(합성)
    # 아래 2개는 임의 예시
    "411060.KS": 5,   # ACE KRX금현물 (실제 존재하는 티커)
    "261220.KS": 10,  # KODEX WTI원유선물(H)
}


# ============== 2) 데이터 다운로드 (과거 3년) ==============
tickers = list(risky_assets.keys()) + list(safe_assets.keys()) + list(canary_asset.keys()) + list(fixed_assets.keys())

end_date = datetime.today().strftime('%Y-%m-%d')
start_date = (datetime.today() - pd.DateOffset(years=3)).strftime('%Y-%m-%d')

print(f"Downloading data from {start_date} to {end_date}...")
data = yf.download(tickers, start=start_date, end=end_date)['Close']

# 혹시 결측/중복 열 정리
data = data.loc[:, ~data.columns.duplicated()].copy()


# ============== 3) 월말 데이터 리샘플링 ==============
monthly_data = data.resample('M').last()


# ============== 4) 모멘텀 계산 함수 (월단위) ==============
def calculate_momentum(series):
    """
    월단위 종가(월말)에 대해
    1M, 3M, 6M, 12M 전 대비 수익률을 평균내서 모멘텀으로 사용
    """
    df = pd.DataFrame(index=series.index)
    df['1M']  = series.pct_change(periods=1)
    df['3M']  = series.pct_change(periods=3)
    df['6M']  = series.pct_change(periods=6)
    df['12M'] = series.pct_change(periods=12)
    return df.mean(axis=1)

# 전체 티커에 대한 모멘텀 DataFrame
momentum_data = monthly_data.apply(calculate_momentum)

# TIP(카나리아) 모멘텀 체크
def canary_momentum(tip_series):
    tip_mom = calculate_momentum(tip_series)
    # True/False로 반환
    return tip_mom > 0

canary_signal = canary_momentum(monthly_data['TIP'])


# ============== 5) 자산 선택 로직 ==============
def select_safe_assets(momentum_df, bil_momentum, top_n=2):
    """
    - 안전자산 중 모멘텀 상위 top_n
    - BIL 모멘텀(이 예시에서는 '130730.KS' 단기자금으로 대체 가능)보다 낮으면 제외
    - 최소 1개 '130730.KS' 보유
    """
    # 안전자산 중 최근 행
    safe_mom = momentum_df[safe_assets.keys()].iloc[-1]
    top_safe = safe_mom.nlargest(top_n)
    selected = []
    for asset in top_safe.index:
        if top_safe[asset] > bil_momentum:
            selected.append(asset)

    if not selected:
        return ['130730.KS']
    if '130730.KS' not in selected:
        selected.append('130730.KS')
    return selected[:top_n]

def select_top_assets(momentum_df, canary_signal_series, top_n=4):
    """
    canary_signal_series.iloc[-1] == True -> 공격 자산 (risky_assets)
    else -> 안전 자산
    """
    if canary_signal_series.iloc[-1]:
        # 공격 모드: risky_assets 중 모멘텀 상위 n개
        top_assets = momentum_df[risky_assets.keys()].iloc[-1].nlargest(top_n)
        return top_assets.index.tolist()
    else:
        # 안전 모드
        if '130730.KS' in data.columns:
            bil_momentum = calculate_momentum(data['130730.KS']).iloc[-1]
        else:
            bil_momentum = 0  # 단순 가정
        return select_safe_assets(momentum_df, bil_momentum, top_n=2)

def normalize_weights(weights_dict, total=100):
    """
    가중치 합을 total(%)로 맞추되, 정수 반올림 후
    합이 total보다 부족/초과 시, 최댓값 항목에 조정
    """
    rounded = {k: round(v) for k, v in weights_dict.items()}
    diff = total - sum(rounded.values())
    if diff != 0 and len(rounded) > 0:
        max_key = max(rounded, key=rounded.get)
        rounded[max_key] += diff
    return rounded


# ============== 6) 백테스트: "이번 달 말 -> 다음 달 말" ==============

portfolio_data = []

dates = monthly_data.index
# 12개월치 전에는 12M 모멘텀이 제대로 계산 안 되므로 i=12부터
# 마지막 달은 (다음 달이 없어서) -1
for i in range(12, len(dates) - 1):
    # 이번 달 말 (의사결정)
    decision_date = dates[i]
    # 다음 달 말 (투자 결과)
    invest_date = dates[i + 1]

    # (1) decision_date까지의 모멘텀/카나리아 계산
    decision_data = monthly_data.loc[:decision_date]
    decision_momentum = decision_data.apply(calculate_momentum)

    canary_series = canary_momentum(decision_data['TIP'])

    # (2) 자산 선정
    top_assets = select_top_assets(decision_momentum, canary_series, top_n=4)

    # (3) 고정 자산(예: 35%) + 변동 자산(65%)
    #     변동자산 개수(top_assets)가 n개면 각각 65/n %
    dynamic_share = 65
    if len(top_assets) == 0:
        # 방어모드에서 top_assets가 비었을 경우 등
        top_assets = ['130730.KS']
    weight_each = dynamic_share / len(top_assets)

    combined_weights = {}
    # 고정자산
    for asset, w in fixed_assets.items():
        combined_weights[asset] = w
    # 변동자산
    for asset in top_assets:
        combined_weights[asset] = combined_weights.get(asset, 0) + weight_each

    # 최종 비중 정수화 (합=100)
    final_weights = normalize_weights(combined_weights, total=100)

    # (4) 수익률 계산
    monthly_return = 0
    if invest_date in monthly_data.index:  # 혹시 invest_date가 누락되었을 경우 방어
        for asset, w in final_weights.items():
            if asset in monthly_data.columns:
                # (decision_date) ~ (invest_date) 한 달
                ret = monthly_data[asset].pct_change().loc[invest_date]
                monthly_return += (w/100) * ret

    # (5) 기록
    asset_list_str = ", ".join([f"{k}: {v}%" for k,v in final_weights.items()])
    portfolio_data.append({
        "Date": invest_date.strftime('%Y-%m-%d'),
        "Assets & Weights": asset_list_str,
        "Monthly Return": round(monthly_return * 100, 2),
    })


# (7) 포트폴리오 DataFrame 생성
portfolio = pd.DataFrame(portfolio_data)

# (8) CAGR, MDD 계산
portfolio['Cumulative'] = (1 + portfolio['Monthly Return'] / 100).cumprod()
if len(portfolio) > 1:
    start_dt = datetime.strptime(portfolio['Date'].iloc[0], '%Y-%m-%d')
    end_dt   = datetime.strptime(portfolio['Date'].iloc[-1], '%Y-%m-%d')
    years = (end_dt - start_dt).days / 365.25
    cagr = portfolio['Cumulative'].iloc[-1] ** (1/years) - 1

    def calculate_mdd(cum_returns):
        peak = cum_returns.cummax()
        dd = cum_returns / peak - 1
        return dd.min()

    mdd = calculate_mdd(portfolio['Cumulative'])
else:
    cagr = 0.0
    mdd  = 0.0

# (9) 결과 출력
print("\n백테스트 결과 (월말→다음달말):")
print(portfolio[['Date', 'Assets & Weights', 'Monthly Return']])

print(f"\nCAGR(연평균수익률): {cagr*100:.2f}%")
print(f"MDD(최대낙폭): {mdd*100:.2f}%")
