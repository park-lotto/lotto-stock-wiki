def search_full(keyword, max_results=40, token=None, timeout=180, poll_interval=5):
    """키워드 → build_tiktok_items 스키마 raw dict 리스트. video_id 없는 행은 제외."""
    tokens = [token] if token else APIFY_TOKENS
    if not tokens:
        raise RuntimeError("douyin_search: APIFY_TOKEN이 설정되지 않았습니다")
    payload = {"keywords": [keyword], "maxResultsPerQuery": max_results, "sort": "general"}
    items = _run_with_rotation(payload, tokens, timeout, poll_interval, actor=_ACTOR)
    out = []
    for item in items:
        vid = item.get("id") or item.get("awemeId")
        url = item.get("url")
        if not vid or not url or item.get("type") != "video":
            continue
        out.append({
            "video_id": str(vid),
            "title": item.get("itemTitle") or item.get("previewTitle") or item.get("text", ""),
            "published_at": _first(item, "createTimeISO", "createTime") or "",
            "views": _num(_first(item, ("statistics", "playCount"))) or 0,
            "likes": _num(_first(item, ("statistics", "diggCount"))) or 0,
            "comments": _num(_first(item, ("statistics", "commentCount"))) or 0,
            "channel_title": _first(item, ("authorMeta", "nickName"), ("authorMeta", "name")) or "",
            "thumbnail": (item.get("videoMeta") or {}).get("cover", ""),
            "url": url,
            "media_platform": "douyin",
        })
    return out
```

"""해외HOT 무료 깔때기 — 형식 하드컷·관련성 게이트·'안터진' 조회수 상한. 순수함수라
overseas_hot_jobs에서 조합해 쓰고 단위테스트가 쉽다. (설계 §4 STAGE 1·2, §5 상한)"""

# 비꿀템 잡영상 차단어(제목 소문자 매칭). 선점 목적과 무관한 포맷을 컷.
BLOCK_WORDS = ("dance", "prank", "storytime", "story time", "lip sync", "lipsync",
               "giveaway", "sub for sub", "follow me", "asmr")

# 이미 터진 것 제외 — 절대조회수 상한(설계 §5 "안터짐"). 국가별 감이라 튜닝 대상.
DEFAULT_VIEW_CEILING = 3_000_000


def passes_format(item):
    """재생 URL과 video_id가 있어야 통과(죽은/불완전 행 컷)."""
    return bool(item.get("url")) and bool(item.get("video_id"))


def passes_relevance(item, allow_words):
    """제목에 카테고리 허용어가 ≥1 있고 차단어가 없으면 통과."""
    title = (item.get("title") or "").lower()
    if any(b in title for b in BLOCK_WORDS):
        return False
    return any(w.lower() in title for w in allow_words) if allow_words else True

    return int(item.get("views") or 0) <= ceiling

def gap_badge(title, min_korean=1, translate=False):
    """제목 → '🔥선점가능' | '이미유입' | '미확인'.

    translate=True면(CN 영상) 제목을 한국어로 번역해 검색한다 — 중국어 제목 그대로면
    한글 재편집본을 못 찾아 거짓 '선점가능'이 나온다. 번역 실패/빈값이면 '미확인'."""
    q = (title or "").strip()
    if not q:
        return "미확인"
    if translate:
        from shopping_shorts import video_analysis
        try:
            q = (video_analysis.translate_keyword(q) or "").strip()
        except Exception:
            q = ""
        # 번역결과에 한글이 없으면(빈값 또는 방향반대로 중국어 그대로) 검색 무의미 → 미확인.
        if not _has_korean(q):
            return "미확인"
    try:
        results = youtube_search.search(q, max_results=10)
    except Exception:
        return "미확인"
    korean = sum(1 for r in results if _has_korean(r.get("title")))
    return "이미유입" if korean >= min_korean else "🔥선점가능"
```
