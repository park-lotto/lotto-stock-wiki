# 라이브러리 불러오기
import pandas as pd
import numpy as np
import yfinance as yf
from datetime import datetime

# 자산 리스트 설정
공격자산 = ["XLE", "XLB", "XLI", "XLY", "XLP", "XLV", "XLF", "XLK", "XLU"]
대표주식 = "SPY"
전체리스트 = 공격자산 + [대표주식]

# Yahoo Finance에서 조정 종가 데이터 가져오기
def get_yahoo_data(tickers, start, get_type="Adj Close"):
    df = yf.download(tickers, start=start, progress=False)
    df = df[get_type]
    df.dropna(inplace=True)
    return df

# 월말 데이터만 추출하는 함수
def get_rebal_date(df):
    df["year"] = df.index.year
    df["month"] = df.index.month
    days_df = df.groupby(["year", "month"]).tail(1)  # 각 달의 마지막 거래일 선택
    return days_df.drop(columns=["year", "month"])

# 모멘텀 스코어 계산 함수 (1, 3, 6, 9, 12개월 평균 수익률)
def calculate_momentum_score(df, tickers, i):
    scores = {}
    for ticker in tickers:
        m1 = (df[ticker].iloc[i] - df[ticker].iloc[i-1]) / df[ticker].iloc[i-1]
        m3 = (df[ticker].iloc[i] - df[ticker].iloc[i-3]) / df[ticker].iloc[i-3]
        m6 = (df[ticker].iloc[i] - df[ticker].iloc[i-6]) / df[ticker].iloc[i-6]
        m9 = (df[ticker].iloc[i] - df[ticker].iloc[i-9]) / df[ticker].iloc[i-9]
        m12 = (df[ticker].iloc[i] - df[ticker].iloc[i-12]) / df[ticker].iloc[i-12]
        scores[ticker] = (m1 + m3 + m6 + m9 + m12) / 5
    return scores

# 데이터 가져오기
data = get_yahoo_data(전체리스트, start="2003-01-01")
rebal_data = get_rebal_date(data)

# 월별 투자 자산 및 수익률 계산
portfolio_data = []

# BAA 전략 수행
for i in range(12, rebal_data.shape[0] - 1):  # 12개월째부터 시작, 마지막 달은 제외하여 미래 참조 방지
    date = rebal_data.index[i]
    next_month_date = rebal_data.index[i + 1]  # 다음 달 날짜 (투자 적용 달)

    # SPY의 10개월 평균 가격 계산
    spy_avg_10m = rebal_data[대표주식].iloc[i-10:i].mean()
    spy_condition = rebal_data[대표주식].iloc[i] > spy_avg_10m

    res = {}
    if spy_condition:
        # 공격자산 중 모멘텀 스코어가 높은 상위 3개 선택
        momentum_scores = calculate_momentum_score(rebal_data, 공격자산, i)
        top_3_assets = sorted(momentum_scores, key=momentum_scores.get, reverse=True)[:3]
        for asset in top_3_assets:
            res[asset] = 100 / 3
    else:
        # SPY 조건을 충족하지 못하면 현금 100% 보유
        res["Cash"] = 100

    # 매수할 자산 리스트 생성
    assets_list = ', '.join([f"{asset}" for asset in res.keys()])

    # 다음 달 수익률 계산 (가중평균 수익률)
    monthly_return = sum(
        (rebal_data[asset].pct_change().iloc[i + 1] * weight / 100) if asset != "Cash" else 0
        for asset, weight in res.items()
    )

    # 결과 저장
    portfolio_data.append({
        "Date": next_month_date.strftime('%Y-%m-%d'),
        "Assets": assets_list,
        "Monthly Return": round(monthly_return * 100, 2)
    })

# 최종 포트폴리오 DataFrame 생성
portfolio = pd.DataFrame(portfolio_data)

# 최신 36개월(3년) 데이터만 유지
portfolio = portfolio.tail(36)

# 최종 포트폴리오 결과 출력
print("\n최근 3년간 월별 투자 자산 및 수익률:")
print(portfolio)

# CAGR 및 MDD 계산
portfolio['Cumulative'] = (1 + portfolio['Monthly Return'] / 100).cumprod()
years = (datetime.strptime(portfolio['Date'].iloc[-1], '%Y-%m-%d') - datetime.strptime(portfolio['Date'].iloc[0], '%Y-%m-%d')).days / 365.25
cagr = (portfolio['Cumulative'].iloc[-1])**(1/years) - 1

def calculate_mdd(cumulative_returns):
    cumulative_max = cumulative_returns.cummax()
    drawdown = cumulative_returns / cumulative_max - 1
    mdd = drawdown.min()
    return mdd

mdd = calculate_mdd(portfolio['Cumulative'])

# 가장 최신 데이터 날짜
latest_actual_date = portfolio['Date'].iloc[-1]

# 결과 출력
print(f"\n최근 3년간 SRS 전략에 따른 백테스트 결과:")
print(f"CAGR(연평균 성장률): {cagr * 100:.2f}%")
print(f"MDD(최대 낙폭): {mdd * 100:.2f}%")
print(f"\n가장 최신 데이터 날짜: {latest_actual_date}")
