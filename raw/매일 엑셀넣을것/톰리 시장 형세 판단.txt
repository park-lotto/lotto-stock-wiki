# 📌 2. 라이브러리 불러오기
import pandas as pd
import yfinance as yf
import matplotlib.pyplot as plt
from datetime import datetime, timedelta
import pandas_datareader.data as web
from google.colab import files

# 📌 3. 엑셀 파일 업로드
uploaded = files.upload()
file_name = list(uploaded.keys())[0]
df = pd.read_excel(file_name, sheet_name="최종정리")
df['DATE'] = pd.to_datetime(df['DATE'])
latest_row = df.iloc[-1]
sp500_value = latest_row['S&P500']
vix_value = latest_row['VIX']
print(f"✅ S&P500: {sp500_value}, ✅ VIX: {vix_value}")

# 📌 4. Forward PER 수동 입력
forward_per = float(input("📝 Forward PER (예: 21.3): "))

# 📌 5. 하이일드 스프레드 (FRED)
start = df['DATE'].iloc[0]
end = df['DATE'].iloc[-1]
spread_df = web.DataReader('BAMLH0A0HYM2', 'fred', start, end).dropna()
latest_spread = spread_df.iloc[-1].values[0]
print(f"✅ 하이일드 스프레드 (OAS 기준): {latest_spread:.2f}%")

# 📌 6. NASDAQ100 + S&P100 유니버스 수집
def get_nasdaq100_tickers():
    url = 'https://en.wikipedia.org/wiki/NASDAQ-100'
    tables = pd.read_html(url)
    for table in tables:
        if "Ticker" in table.columns:
            return [ticker.replace('.', '-') for ticker in table["Ticker"].tolist()]
    return []

def get_sp100_tickers():
    url = 'https://en.wikipedia.org/wiki/S%26P_100'
    tables = pd.read_html(url)
    for table in tables:
        if "Symbol" in table.columns:
            return [ticker.replace('.', '-') for ticker in table["Symbol"].tolist()]
    return []

nasdaq100 = get_nasdaq100_tickers()
sp100 = get_sp100_tickers()
universe = sorted(list(set(nasdaq100 + sp100)))

# 📌 7. 종가 데이터 수집 (60일, 오늘 제외)
end_date = datetime.today() - timedelta(days=1)
start_date = end_date - timedelta(days=60)

print(f"\n📥 가격 데이터 수집 중... ({len(universe)} 종목)")
raw = yf.download(universe, start=start_date, end=end_date,
                  group_by='ticker', progress=False, auto_adjust=False)
data = raw.xs('Close', axis=1, level=1)
data = data.dropna(axis=1, how='any')
print(f"✅ 유효 종목 수: {data.shape[1]}개")

# 📌 8. ZBT 계산
daily_up_ratio = (data.pct_change() > 0).sum(axis=1) / data.shape[1]
zbt_series = daily_up_ratio.rolling(10).mean()
zbt_today = zbt_series.iloc[-1]
zbt_recent_min = zbt_series.iloc[-10:-1].min()

# 📌 9. ZBT 시각화
plt.figure(figsize=(12, 4))
plt.plot(zbt_series, label="10-day Avg Breadth")
plt.axhline(0.615, color='green', linestyle='--', label='ZBT Signal (61.5%)')
plt.axhline(0.4, color='red', linestyle='--', label='ZBT Base (40%)')
plt.title("Zweig Breadth Thrust (ZBT) - 톰 리 저점 반등 판단 지표")
plt.grid(True)
plt.legend()
plt.tight_layout()
plt.show()

# 📌 10. 진단 출력
print("\n📊 [톰 리 시그널 진단]")
print(f"🔹 S&P500: {sp500_value}")
print(f"🔹 VIX: {vix_value}")
print(f"🔹 하이일드 스프레드: {latest_spread:.2f}%")
print(f"🔹 Forward PER: {forward_per}")
print(f"🔹 ZBT (오늘): {zbt_today:.3f}, (최근 최저): {zbt_recent_min:.3f}")

# 📌 11. 종합 판단 로직 (톰 리 해석 반영)
zbt_msg = "✅ ZBT 바닥 시그널 발생" if (zbt_today > 0.615 and zbt_recent_min < 0.4) else "❌ ZBT 조건 미충족"
vix_msg = "✅ VIX 안정 (공포 완화)" if vix_value < 20 else "⚠️ VIX 높음 (시장 불안정)"
spread_msg = "✅ 하이일드 스프레드 축소 (위험자산 선호)" if latest_spread < 3.5 else "⚠️ 하이일드 스프레드 확대 (리스크 경계)"

# Forward PER 해석 로직
if forward_per >= 20:
    if latest_spread >= 3.5 or vix_value >= 20:
        per_msg = "❌ Forward PER 고평가 + 위험지표 동반 → 고점 경고"
    else:
        per_msg = "🔍 PER 고평가이나, 하이일드·VIX 안정 → 시장 강세 여지"
elif forward_per < 17:
    per_msg = "✅ Forward PER 저평가 → 저점 매력"
else:
    per_msg = "🔍 Forward PER 중립"

# 📌 12. 최종 종합 출력
print("\n🧠 종합 판단 (톰 리 기준):")
print(zbt_msg)
print(vix_msg)
print(spread_msg)
print(per_msg)

# 📌 13. ZBT 설명 주석 추가
print("\n📝 참고: ZBT는 급락장에서 반등이 시작될 때 '반등의 힘'을 평가하기 위한 저점 탐색용 지표입니다.")
print("ZBT는 고점 시그널이 아니라, 단기 저점에서 폭넓은 매수세가 유입되는지를 판별하는 데 사용됩니다.")
