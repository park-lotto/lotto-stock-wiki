> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 카드 미리보기의 멈춘 장면을 구글 렌즈로 역검색해 5개 플랫폼(유튜브/틱톡/인스타/샤오홍슈/도우인) 유사 동영상을 자동 발굴한다.

# lens_discover.py
"""SerpApi Google Lens로 프레임을 역검색해 유사 동영상(5개 플랫폼)만 추린다.

visual_matches에는 동영상 여부 필드가 없어(2026-07-14 실측: link/source/title/
thumbnail만) link 도메인 화이트리스트로 판별한다. product_identify.py와 같은
google_lens 엔진이지만 용도가 달라(제품명 추론 X, 유사영상 발굴 O) 분리."""
import requests
from urllib.parse import urlparse
from shopping_shorts.config import SERPAPI_KEY

_LENS_ENDPOINT = "https://serpapi.com/search"

# 플랫폼 판별 — 도메인 접미사 매칭(서브도메인·www 무관). 순서=표시 우선순위.
_PLATFORM_DOMAINS = [
    ("youtube", ("youtube.com", "youtu.be")),
    ("tiktok", ("tiktok.com",)),
    ("instagram", ("instagram.com",)),
    ("xiaohongshu", ("xiaohongshu.com", "xhslink.com")),
    ("douyin", ("douyin.com", "iesdouyin.com")),
]


def _platform_of(link):
    host = urlparse(link or "").netloc.lower()
    for name, domains in _PLATFORM_DOMAINS:
        if any(host == d or host.endswith("." + d) for d in domains):
            return name
    return None


def search_similar_videos(image_url, api_key=None, timeout=60):
    """공개 이미지 URL → [{platform, url, title, thumbnail}]. 5개 동영상 플랫폼만.
    키 없음·호출 실패 시 []."""
    key = api_key or SERPAPI_KEY
    if not key:
        return []
    params = {"engine": "google_lens", "url": image_url, "api_key": key}
    try:
        r = requests.get(_LENS_ENDPOINT, params=params, timeout=timeout)
        r.raise_for_status()
        data = r.json()
    except requests.RequestException:
        return []
    out = []
    for m in data.get("visual_matches") or []:
        platform = _platform_of(m.get("link"))
        if not platform:
            continue
        out.append({
            "platform": platform,
            "url": m.get("link", ""),
            "title": m.get("title", ""),
            "thumbnail": m.get("thumbnail", ""),
        })
    return out
```

    store = Store(DB_PATH)
    now = datetime.now(timezone.utc)
    month = now.strftime("%Y-%m")
    limit = int(store.get_setting("lens_month_limit", _LENS_MONTH_LIMIT_DEFAULT))
    if store.lens_month_count(month) >= limit:
        return JSONResponse(status_code=429, content={
            "ok": False, "error_code": "lens_limit",
            "error": f"이번 달 렌즈 검색 한도({limit}회)를 다 썼습니다"})

    image_url = f"{PUBLIC_BASE_URL}/api/find/frame/lens/{name}"
    items = search_similar_videos(image_url)
    store.bump_lens(month)
    return {"ok": True, "items": items, "count": len(items)}

# media_download.py 에 추가
import subprocess, sys, json as _json

def resolve_media_url(platform, video_id, timeout=30):
    """유튜브/틱톡 영상ID → 진행형 mp4 direct URL(다운로드 없이). yt-dlp -g로
    재생 가능한 단일 mp4 포맷 URL만 뽑는다. 실패(비공개·지역차단) 시 "".
    캡처(canvas)를 위해 우리 <video>로 same-origin 재생하려는 용도."""
    page = {
        "youtube": f"https://www.youtube.com/watch?v={video_id}",
        "tiktok": f"https://www.tiktok.com/@x/video/{video_id}",
    }.get(platform)
    if not page:
        return ""
    try:
        r = subprocess.run(
            [sys.executable, "-m", "yt_dlp", "-g", "-f",
             "best[ext=mp4][vcodec!=none][acodec!=none]/best[ext=mp4]/best",
             "--no-warnings", page],
            capture_output=True, text=True, encoding="utf-8", timeout=timeout)
    except Exception:
        return ""
    if r.returncode != 0 or not r.stdout.strip():
        return ""
    return r.stdout.strip().splitlines()[0]
```

// 유튜브/틱톡: 우리가 뽑은 mp4 direct URL로 재생(캡처 가능). 실패 시 embed 폴백.
async function _playViaMedia(img, videoId, platform, embedFn){
  if(!videoId) return;
  const box=img.parentElement;
  const badge=box && box.querySelector('.play-badge'); if(badge) badge.remove();
  try{
    const d=await (await fetch(`/api/media?platform=${platform}&id=${encodeURIComponent(videoId)}`)).json();
    if(d.ok && d.url){
      const v=document.createElement('video');
      v.src=d.url; v.controls=true; v.autoplay=true; v.playsInline=true;
      v.setAttribute('playsinline',''); v.setAttribute('crossorigin','anonymous');
      v.className='thumb-video';
      v.onerror=()=>embedFn(v, videoId);   // mp4 재생 실패 시 embed 폴백
      img.replaceWith(v); return;
    }
  }catch(e){}
  embedFn(img, videoId);
}
function _embedYoutube(el, videoId){
  const f=document.createElement('iframe');
  f.src='https://www.youtube.com/embed/'+encodeURIComponent(videoId)+'?autoplay=1&playsinline=1';
  f.style.cssText='width:100%;height:220px;border:0;display:block;background:#000';
  f.allow='autoplay; encrypted-media; picture-in-picture'; f.allowFullscreen=true;
  el.replaceWith(f);
}
function _embedTiktok(el, videoId){
  const f=document.createElement('iframe');
  f.src='https://www.tiktok.com/embed/v2/'+encodeURIComponent(videoId);
  f.style.cssText='width:100%;height:220px;border:0;display:block;background:#000';
  f.allow='autoplay; encrypted-media; picture-in-picture'; f.allowFullscreen=true;
  el.replaceWith(f);
}
function playYoutube(img, videoId){ _playViaMedia(img, videoId, 'youtube', _embedYoutube); }
function playTiktok(img, videoId){ _playViaMedia(img, videoId, 'tiktok', _embedTiktok); }

const LENS_PLATFORMS=[
  {k:'youtube',label:'▶️ 유튜브'},{k:'tiktok',label:'🎵 틱톡'},
  {k:'instagram',label:'📷 인스타'},{k:'xiaohongshu',label:'📕 샤오홍슈'},

// 카드의 현재 <video>(재생중/멈춤) 프레임을 캡처. video 없으면 null.
function _captureFrame(shortcode){
  const box=document.getElementById('lens_'+shortcode);
  const card=box.closest('.card');
  const v=card && card.querySelector('video.thumb-video');
  if(!v || v.readyState<2) return null;
  const cv=document.createElement('canvas');
  cv.width=v.videoWidth; cv.height=v.videoHeight;
  cv.getContext('2d').drawImage(v,0,0);
  return cv;   // toBlob은 호출부에서(async)
}

async function lensSearch(btn, shortcode){
  const box=document.getElementById('lens_'+shortcode);
  box.style.display='block';
  const cv=_captureFrame(shortcode);
  if(!cv){ box.innerHTML='<div style="font-size:12px;color:var(--sub)">▶ 미리보기를 재생하고 원하는 장면에서 멈춘 뒤 눌러주세요.</div>'; return; }
  box.innerHTML='<div style="font-size:12px;color:var(--sub)">🔍 렌즈 검색 중…</div>';
  const blob=await new Promise(r=>cv.toBlob(r,'image/jpeg',0.85));
  const fd=new FormData(); fd.append('frame', blob, 'f.jpg');
  try{
    const r=await fetch('/api/lens/search',{method:'POST',body:fd});
    if(r.status===429){ const d=await r.json(); box.innerHTML=`<div style="font-size:12px;color:#e0623d">💰 ${d.error||'이번 달 렌즈 한도 초과'}</div>`; return; }
    const d=await r.json();
    if(!d.ok || !d.items.length){ box.innerHTML='<div style="font-size:12px;color:var(--sub)">유사 동영상을 찾지 못했습니다.</div>'; return; }
    LENS_STATE[shortcode]={items:d.items, on:new Set(LENS_PLATFORMS.map(p=>p.k))};
    renderLens(shortcode);
  }catch(e){ box.innerHTML='<div style="font-size:12px;color:#e0623d">❌ '+e.message+'</div>'; }
}

function toggleLensPlatform(shortcode, k){
  const st=LENS_STATE[shortcode]; if(!st) return;
  st.on.has(k)?st.on.delete(k):st.on.add(k); renderLens(shortcode);
}

function renderLens(shortcode){
  const box=document.getElementById('lens_'+shortcode);
  const st=LENS_STATE[shortcode]; if(!st) return;
  const chips=LENS_PLATFORMS.map(p=>{
    const active=st.on.has(p.k);
    const n=st.items.filter(i=>i.platform===p.k).length;
    return `<span onclick="toggleLensPlatform('${shortcode}','${p.k}')" style="cursor:pointer;display:inline-block;padding:2px 8px;margin:2px;border-radius:12px;font-size:11px;border:1px solid var(--line);background:${active?'#1f6feb22':'transparent'};opacity:${active?1:0.45}">${p.label} ${n}</span>`;
  }).join('');
  const cards=st.items.filter(i=>st.on.has(i.platform)).map(i=>`
    <a href="${i.url}" target="_blank" style="display:flex;gap:8px;align-items:center;text-decoration:none;color:var(--txt);padding:4px;border-bottom:1px solid var(--line)">
      <img src="${i.thumbnail?('/api/thumb?url='+encodeURIComponent(i.thumbnail)):''}" style="width:56px;height:56px;object-fit:cover;border-radius:6px;flex-shrink:0" onerror="this.style.visibility='hidden'">
      <span style="font-size:12px">${LENS_PLATFORMS.find(p=>p.k===i.platform)?.label||''}<br>${(i.title||'').slice(0,50)}</span>
    </a>`).join('') || '<div style="font-size:12px;color:var(--sub);padding:6px">켠 플랫폼 결과 없음</div>';
  box.innerHTML=`<div style="margin:6px 0">${chips}</div><div>${cards}</div>`;
}
