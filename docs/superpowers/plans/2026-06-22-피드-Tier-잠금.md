```python
# 소스 카테고리 → 요구 Tier (누적 잠금). market은 Tier 무관 항상 노출.
SOURCE_TIER = {
    "news": 1,
    "blog": 2,
    "youtube": 3,
    "telegram": 4,
    "reports": 5, "report": 5,
}
_ALWAYS_VISIBLE = {"market"}


def filter_by_tier(items, feed_tier):
    """feed_tier 이하 소스만 통과. 매핑 없는 카테고리는 노출(누락 방지)."""
    out = []
    for it in items:
        cat = it.get("category", "")
        if cat in _ALWAYS_VISIBLE:
            out.append(it)
            continue
        need = SOURCE_TIER.get(cat)
        if need is None or need <= int(feed_tier or 1):
            out.append(it)
    return out
```
