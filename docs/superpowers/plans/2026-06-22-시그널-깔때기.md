```python
def score_stock(stock: dict) -> int:
    """빈집(2점) 필수. 없으면 0. 있으면 + 7개 플래그 각 1점."""

        return 0
    return 2 + sum(1 for k in _FLAG_KEYS if f.get(k, 0) >= 1)
```

- [ ] **Step 5: build_stocks 구현 — 파서 결합**

`pipeline/build_signal_snapshot.py`에 추가. 각 파서 return의 실제 키는 `scripts/ingest_excel.py`를 읽어 매핑한다(오실레이터 행 = `{name, code, osc, pct, trend}` 확인됨; pct로 빈집등급 산출: ≤10→A, ≤25→B, ≤75→C, else D).

```python
def _vacancy_grade(pct):
    if pct is None: return "?"
    if pct <= 10: return "A"
    if pct <= 25: return "B"
    if pct <= 75: return "C"
    return "D"


def build_stocks(dry_run=True) -> list:
    """오실레이터(빈집) 기준으로 종목 목록을 만들고, RS·수출·컨센·어닝으로 플래그를 채운다."""
    import sys
    sys.path.insert(0, str(ROOT / "scripts"))
    import ingest_excel as ie

    osc = ie.parse_수급오실레이터(dry_run)
    rs = ie.parse_rs(dry_run)
    exp = ie.parse_수출정리(dry_run)
    con = ie.parse_컨센움직임(dry_run)

    # RS·수출·컨센 신고가 종목명 집합 (실제 키는 각 파서 return 확인 후 매핑)
    rs_map = {r["name"]: r for r in _rows(rs)}
    export_red = {r["name"] for r in _rows(exp) if _is_red(r)}
    con_high = {r["name"] for r in _rows(con, key="컨센상향")}

    stocks = []
    for r in _rows(osc):
        name = r.get("name"); pct = r.get("pct")
        grade = _vacancy_grade(pct)
        flags = {
            "빈집": 2 if grade in ("A", "B") else 0,
            "수출": 1 if name in export_red else 0,
            "컨센신고가": 1 if name in con_high else 0,
            "어닝서프": 0,            # 컨센움직임 서프라이즈 시트 매핑 시 1
            "판가": 0, "미국커플링": 0, "정책": 0, "D30": 0,  # 수동 플래그 기본 0
        }
        st = {"name": name, "code": r.get("code"), "sector": ie.get_stock_sector(name) or "?",
              "vacancy": grade, "rs": (rs_map.get(name) or {}).get("rs"), "flags": flags}
        st["score"] = score_stock(st)
        stocks.append(st)
    stocks.sort(key=lambda s: s["score"], reverse=True)
    return stocks


def _rows(parsed, key=None):
    """파서 return에서 행 리스트를 꺼낸다. 구조 불일치 시 빈 리스트."""
    if not isinstance(parsed, dict) or "error" in parsed:
        return []
    res = parsed.get("results", parsed)
    if key and isinstance(res, dict):
        return res.get(key, [])
    if isinstance(res, list):
        return res
    if isinstance(res, dict):
        out = []
        for v in res.values():
            if isinstance(v, list): out += v
        return out
    return []


def _is_red(row) -> bool:
    return "🔴" in str(row.get("signal", "")) or str(row.get("level", "")) == "🔴"
```

> ⚠️ `_rows`/`_is_red`/RS·컨센 키 매핑은 각 파서의 실제 return을 읽고 정확히 맞춘다. 구조가 비면 빈 리스트로 안전 강등(스냅샷은 비어도 유효).

```python
def build_snapshot(macro: dict, lead_sectors: list, stocks: list) -> dict:
    today = datetime.now().strftime("%Y-%m-%d")
    return {
        "date": today,
        "generated_at": datetime.now().isoformat(timespec="seconds"),
        "stage1": macro,
        "stage2": {"sectors": [{"name": s, "score": None,
                                "in_us_strong": True, "in_taerini_lead": True}
                               for s in lead_sectors]},
        "stage3": {"stocks": stocks},
    }


def main():
    # STAGE1 매크로: 자동 소스 미연결 → 수기 입력 파일 우선, 없으면 CAUTION 기본.
    macro_path = ROOT / "output" / "signal" / "macro_today.json"
    if macro_path.exists():
        macro = json.loads(macro_path.read_text(encoding="utf-8"))
    else:
        macro = {"verdict": "CAUTION", "reasons": [{"label": "매크로", "ok": False,
                 "detail": "수기 입력(macro_today.json) 없음 — 기본 경계"}]}
    # STAGE2 주도섹터: 소르티노 미개발 → 수기 lead_sectors.json, 없으면 빈 목록.
    lead_path = ROOT / "output" / "signal" / "lead_sectors.json"
    lead = json.loads(lead_path.read_text(encoding="utf-8")) if lead_path.exists() else []
    stocks = build_stocks(dry_run=True)
    snap = build_snapshot(macro, lead, stocks)
    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text(json.dumps(snap, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"스냅샷 기록: {OUT} (종목 {len(stocks)})")
    return snap


if __name__ == "__main__":
    main()
```

```python
    base = get_config()["server"]["output_dir"]
    # output_dir = .../output/md → 형제 폴더 output/signal
    return os.path.join(os.path.dirname(base), "signal", "signal_snapshot.json")


def load_snapshot() -> dict:
    p = _path()
    if not os.path.exists(p):
        return {"date": None, "stage1": {"verdict": "NO", "reasons": [
            {"label": "데이터", "ok": False, "detail": "스냅샷 없음"}]},
            "stage2": {"sectors": []}, "stage3": {"stocks": []}}
    with open(p, encoding="utf-8") as f:
        return json.load(f)


def find_stock(code: str) -> dict | None:
    for s in load_snapshot().get("stage3", {}).get("stocks", []):
        if str(s.get("code")) == str(code):
            return s
    return None
```

```javascript
function tok(){ return localStorage.getItem("token"); }
async function load(){
  const r = await fetch("/api/signal", {headers:{Authorization:"Bearer "+tok()}});
  if(r.status===403){ document.body.innerHTML="<p>시그널 구독이 필요합니다.</p>"; return; }
  const s = await r.json();
  renderStage1(s.stage1); renderStage2(s.stage2, s.stage3); 
}
function renderStage1(st){
  const v = st.verdict||"NO";
  document.getElementById("stage1").innerHTML =
    `<div class="verdict v-${v}">${v==="GO"?"GO ✅":v==="CAUTION"?"⚠️ 경계":"🔴 NO"}</div>`+
    "<ul>"+(st.reasons||[]).map(x=>`<li>${x.ok?"✅":"❌"} ${x.label}: ${x.detail||""}</li>`).join("")+"</ul>";
}
function renderStage2(st, stage3){
  const sectors = st.sectors||[];
  const box = document.getElementById("stage2"); box.innerHTML="";
  sectors.forEach((sec,i)=>{
    const c=document.createElement("span"); c.className="sector-chip"+(i===0?" active":"");
    c.textContent=sec.name+(sec.score!=null?` (${sec.score})`:"");
    c.onclick=()=>{ document.querySelectorAll(".sector-chip").forEach(e=>e.classList.remove("active"));
      c.classList.add("active"); renderStage3(stage3, sec.name); };
    box.appendChild(c);
  });
  // 펼침 a: 1순위 섹터 자동 펼침
  renderStage3(stage3, sectors[0] ? sectors[0].name : null);
}
function renderStage3(stage3, sectorName){
  const rows=(stage3.stocks||[]).filter(s=>!sectorName||s.sector===sectorName);
  document.getElementById("stage3").innerHTML = rows.length? 
    "<table><tr><th>종목</th><th>빈집</th><th>RS</th><th>점수</th></tr>"+
    rows.map(s=>`<tr><td>${s.name}</td><td>${s.vacancy||"?"}</td><td>${s.rs??"-"}</td><td><b>${s.score}</b>/9</td></tr>`).join("")+
    "</table>" : "<p>해당 섹터 종목 없음</p>";
}
load();
```
