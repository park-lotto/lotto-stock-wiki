```python
"""고객 메시지에서 조사에 필요한 정보를 뽑는다.

★순수 함수만 둔다 — 네트워크·텔레그램을 모른다. 그래야 테스트가 빠르고 정확하다.
"""
import re

# 주소 안의 job / job_id 파라미터. 둘 다 실제로 쓰인다(produce.html은 job_id를 쓰고,
# 사람이 손으로 줄여 job=로 적는 경우도 있다).
_JOB = re.compile(r"[?&]job(?:_id)?=([A-Za-z0-9_-]+)")
_URL = re.compile(r"https?://[^\s<>\"]+")


def extract(text):
    """메시지 → {"job_id": str|None, "urls": [str], "text": str}"""
    s = (text or "").strip()
    m = _JOB.search(s)
    return {
        "job_id": m.group(1) if m else None,
        "urls": _URL.findall(s),
        "text": s,
    }
```

```python
    def __init__(self, code=200, data=None, cookies=None):
```

```python
"""서버를 HTTPS API로 조사한다.

★SSH를 쓰지 않는다 — 포트 22가 막혀도 돌고(2026-09-02 실측), 서버 파일을 안 건드려
  배포가 멈추지 않는다(auto_deploy는 워킹트리가 더러우면 조용히 스킵한다).
★모든 /api/ 는 로그인 없으면 401이다(app.py:10016). 그래서 먼저 로그인해 쿠키를 얻는다.
"""


class ProbeError(Exception):
    """조사 실패. ★사유를 뭉개지 않는다 — 'A 또는 B'로 적으면 원인을 못 찾는다."""


class Prober:
    def __init__(self, base, user, password, *, session=None, timeout=20):
        self.base = base.rstrip("/")
        self.user = user
        self.password = password
        self.timeout = timeout
        self._logged_in = False
        if session is None:
            import requests
            session = requests.Session()
        self.s = session

    def _login(self):
        if self._logged_in:
            return
        r = self.s.post(
            self.base + "/api/login",
            data={"user": self.user, "pass": self.password},
            timeout=self.timeout,
            allow_redirects=False,
        )
        # 성공은 303 리다이렉트다(app.py:9614). 200이어도 쿠키가 붙었으면 통과로 본다.
        if r.status_code not in (200, 302, 303):
            raise ProbeError(f"로그인에 실패했습니다 (HTTP {r.status_code}). 계정 설정을 확인하세요.")
        self._logged_in = True

    def job(self, job_id):
        """작업 상태 조회. 실패하면 ProbeError."""
        self._login()
        r = self.s.get(f"{self.base}/api/mix/status/{job_id}", timeout=self.timeout)
        if r.status_code == 404:
            raise ProbeError(f"작업 {job_id} 을(를) 찾을 수 없습니다. 주소가 맞는지 확인해 주세요.")
        if r.status_code == 401:
            raise ProbeError("로그인 세션이 없습니다. 봇 계정 설정을 확인하세요.")
        if r.status_code != 200:
            raise ProbeError(f"조회가 실패했습니다 (HTTP {r.status_code}).")
        return r.json()
```

```python
"""조사 결과 → 사장님께 보낼 답변 문자열.

★고객 문구를 새로 지어내지 않는다. app.py:3958~3970 의 변환표와 **같은 판단**이라
  문구를 그대로 맞춘다(0순위-B: 같은 결정을 두 곳에 다르게 적으면 언젠가 어긋난다).
  ⚠️ 저쪽 표가 바뀌면 여기도 바꿔야 한다. 테스트가 문구를 붙잡고 있다.
"""

# app.py:3958~3970 과 같은 순서·같은 문구. 위에서부터 먼저 맞는 것을 쓴다.
_CUSTOMER_LINES = (
    (("gemini", "키 소진", "edl 비어", "quota", "permission_denied", "api key"),
     "대본을 만드는 데 실패했습니다. 잠시 후 다시 시도해 주세요. "
     "(계속되면 관리자에게 알려주세요)"),
    (("서버 재시작", "중단되었습니다"),
     "작업이 중단되었습니다. 다시 시도해 주세요."),
    (("payment required", "402", "not enough credits", "[600", "insufficient"),
     "영상 처리 서비스의 사용 한도에 걸렸습니다. 고객님 잘못이 아니에요 — "
     "관리자에게 알려주시면 바로 풀어드립니다."),
)
_FALLBACK = "처리 중 문제가 발생했습니다. 잠시 후 다시 시도해 주세요."

# bot_qa.py:17 과 같은 목록 — 돈 얘기는 잘못 답하면 분쟁이다.
_SENSITIVE = ("환불", "결제", "계좌", "입금", "카드", "청구", "요금제", "해지", "탈퇴")


def customer_line(error):
    """오류 문자열 → 고객에게 그대로 보낼 수 있는 한 줄."""
    low = (error or "").lower()
    for keys, line in _CUSTOMER_LINES:
        if any(k.lower() in low for k in keys):
            return line
    return _FALLBACK


def build(job_id, job, *, question=""):
    """조사 결과를 사장님이 읽을 형태로 조립한다."""
    status = (job or {}).get("status", "")
    error = (job or {}).get("error") or ""
    parts = [f"📋 job {job_id}", ""]

    if status == "failed" or error:
        parts += ["【원인】", f"  {error or '사유가 기록되지 않았습니다'}", ""]
        parts += ["【고객께 보낼 문구】", f"  {customer_line(error)}", ""]
    else:
        parts += [f"【상태】 정상 ({status or '진행 중'})", ""]

    if any(w in (question or "") for w in _SENSITIVE):
        parts += ["⚠️ 돈과 관련된 문의입니다. 사장님이 직접 확인해 주세요.", ""]

    return "\n".join(parts).rstrip()
```

**Files:**
- Create: `tg_bot/poller.py`

발송은 `notify.send_telegram`과 같은 방식(`sendMessage` 직접 호출)을 쓴다. 수신은 `getUpdates`.

- [ ] **Step 1: 구현**

`tg_bot/poller.py`:

```python
"""텔레그램 수신·발송.

★조사 로직을 모른다 — 메시지를 나르기만 한다(폰 스크립트가 '배달만 한다'는 관례와 같다).
★사장님 chat_id 외에는 무시한다 — 봇 주소를 알아낸 남이 말을 걸 수 있다.
"""
import os

_API = "https://api.telegram.org/bot{token}/{method}"


class Telegram:
    def __init__(self, token=None, chat_id=None, *, requests_mod=None):
        self.token = token or os.environ.get("SHORTS_TELEGRAM_TOKEN", "")
        self.chat_id = str(chat_id or os.environ.get("SHORTS_TELEGRAM_CHAT_ID", ""))
        if requests_mod is None:
            import requests as requests_mod
        self.r = requests_mod
        self._offset = None

    def ok(self):
        return bool(self.token and self.chat_id)

    def send(self, text):
        if not self.ok():
            return False
        try:
            resp = self.r.post(
                _API.format(token=self.token, method="sendMessage"),
                json={"chat_id": self.chat_id, "text": text},
                timeout=20,
            )
            return getattr(resp, "status_code", 0) == 200
        except Exception:
            return False        # 알림 실패가 봇을 죽이면 안 된다(notify.py 관례)

    def poll(self, wait=25):
        """새 메시지 목록. 사장님 것만 돌려준다."""
        if not self.ok():
            return []
        params = {"timeout": wait}
        if self._offset is not None:
            params["offset"] = self._offset
        try:
            resp = self.r.get(
                _API.format(token=self.token, method="getUpdates"),
                params=params, timeout=wait + 10,
            )
            data = resp.json()
        except Exception:
            return []
        out = []
        for up in data.get("result", []):
            self._offset = up["update_id"] + 1
            msg = up.get("message") or {}
            if str((msg.get("chat") or {}).get("id")) != self.chat_id:
                continue            # ★남이 건 말은 무시한다
            text = msg.get("text")
            if text:
                out.append(text)
        return out
```

### Task 5: 묶어서 돌리기 (`run.py` + `봇켜기.bat`)

- [ ] **Step 1: `run.py` 구현**

```python
"""작업봇을 돌린다. 사장님 PC에서 실행 — claude CLI를 쓰므로 구독으로 동작(0원).

시작: 봇켜기.bat 더블클릭
멈춤: 창에서 Ctrl+C
"""
import os
import sys
import traceback

from tg_bot.context import extract
from tg_bot.poller import Telegram
from tg_bot.probe import Prober, ProbeError
from tg_bot.reply import build

BASE = os.environ.get("SHORTS_BASE", "https://shoppingshorts.duckdns.org")


def _load_env():
    """.env 를 읽어 환경변수로 올린다(python-dotenv 없이도 되게)."""
    for name in (".env", "../.env"):
        if not os.path.exists(name):
            continue
        with open(name, encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line or line.startswith("#") or "=" not in line:
                    continue
                k, v = line.split("=", 1)
                os.environ.setdefault(k.strip(), v.strip())
        return


def handle(text, prober):
    """메시지 하나를 처리해 답변 문자열을 만든다."""
    info = extract(text)
    if not info["job_id"]:
        return ("주소에서 작업 번호를 찾지 못했습니다.\n"
                "고객이 보낸 주소를 그대로 붙여넣어 주세요 (…/produce?job_id=… 형태).")
    try:
        job = prober.job(info["job_id"])
    except ProbeError as e:
        return f"조사하지 못했습니다.\n{e}"
    return build(info["job_id"], job, question=info["text"])


def main():
    _load_env()
    tg = Telegram()
    if not tg.ok():
        print("중단: SHORTS_TELEGRAM_TOKEN / SHORTS_TELEGRAM_CHAT_ID 가 .env 에 없다.",
              file=sys.stderr)
        return 1

    user = os.environ.get("DASH_USER", "")
    password = os.environ.get("DASH_PASS", "")
    if not user or not password:
        print("중단: DASH_USER / DASH_PASS 가 .env 에 없다(서버 조회에 필요).",
              file=sys.stderr)
        return 1

    prober = Prober(BASE, user, password)
    tg.send("🤖 작업봇을 켰습니다. 고객 질문을 넘겨주세요.")
    print("작업봇 실행 중. 멈추려면 Ctrl+C.")

    try:
        while True:
            for text in tg.poll():
                try:
                    tg.send(handle(text, prober))
                except Exception:               # noqa: BLE001 — 한 건 실패가 봇을 죽이면 안 된다
                    traceback.print_exc()
                    tg.send("처리 중 오류가 났습니다. 창의 기록을 확인해 주세요.")
    except KeyboardInterrupt:
        tg.send("🤖 작업봇을 껐습니다.")
        print("종료.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
