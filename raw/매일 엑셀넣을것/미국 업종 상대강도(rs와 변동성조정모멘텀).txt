# ----------------------------
# 5. Mansfield RS 계산 함수 (다중 윈도우 지원)
# ----------------------------
def compute_mansfield_rs(price_series, benchmark_series, ma_period):
    relative = price_series / benchmark_series
    ma = relative.rolling(window=ma_period, min_periods=ma_period).mean()
    return ((relative / ma) - 1) * 100

def normalize_to_100_scale(x, scale=12):
    return 100 * (1 / (1 + np.exp(-x / scale)))

# ----------------------------
# 6. RS 계산 실행 (60d / 120d / 250d + 평균)
# ----------------------------
RS_WINDOWS = [60, 120, 250]
mansfield_rows = []

for ticker in etf_list:
    if ticker == "SPY":
        continue
    if ticker not in data.columns:
        continue

    try:
        price = data[ticker].dropna()
        aligned = pd.concat([price, spy], axis=1, join='inner')
        aligned.columns = ['etf', 'spy']

        latest = {}
        for win in RS_WINDOWS:
            if len(aligned) >= win:
                rs_series = compute_mansfield_rs(aligned['etf'], aligned['spy'], ma_period=win).dropna()
                if not rs_series.empty:
                    raw = float(rs_series.iloc[-1])
                    latest[f"RS_{win}d"] = round(raw, 2)
                    latest[f"Norm_RS_{win}d"] = round(normalize_to_100_scale(raw), 2)

        raw_keys = [k for k in latest if k.startswith("RS_") and k.endswith("d")]
        if not raw_keys:
            continue

        raw_values = [latest[k] for k in raw_keys]
        rs_avg = round(sum(raw_values) / len(raw_values), 2)
        norm_rs_avg = round(normalize_to_100_scale(rs_avg), 2)

        row = {
            "Ticker": ticker,
            "Industry": industry_etfs.get(ticker, ""),
            "RS_60d": np.nan, "RS_120d": np.nan, "RS_250d": np.nan, "RS_avg": rs_avg,
            "Norm_RS_60d": np.nan, "Norm_RS_120d": np.nan, "Norm_RS_250d": np.nan, "Norm_RS_avg": norm_rs_avg
        }
        for win in RS_WINDOWS:
            if f"RS_{win}d" in latest:
                row[f"RS_{win}d"] = latest[f"RS_{win}d"]
                row[f"Norm_RS_{win}d"] = latest[f"Norm_RS_{win}d"]

        mansfield_rows.append(row)

    except Exception:
        continue

mansfield_df = pd.DataFrame(mansfield_rows).sort_values(by="Norm_RS_avg", ascending=False)

# 7. 변동성 조정 모멘텀 계산 함수 (+ 이동평균선 이탈 필터)
#    ✅ FutureWarning 제거: pct_change(fill_method=None)
#    ✅ vol=0 안전장치
# ----------------------------
def calculate_risk_adjusted_momentum(
    data,
    period_3m=63,
    period_6m=126,
    period_12m=252,
    last_n_days=10,
    ma_filter_days=None,
    top_n=20
):
    """
    3/6/12개월 '일간 평균 수익률'을 각 기간 표준편차로 나눠 평균한 위험조정 모멘텀 점수.
    ma_filter_days가 주어지면 해당 이동평균 하회 종목 제외.
    """
    daily_top_industries = {}
    trading_days = data.index[-last_n_days:]

    for day in trading_days:
        end_idx = data.index.get_loc(day)
        start_3m = max(0, end_idx - period_3m + 1)
        start_6m = max(0, end_idx - period_6m + 1)
        start_12m = max(0, end_idx - period_12m + 1)

        # ✅ FutureWarning 방지
        r3 = data.iloc[start_3m:end_idx + 1].pct_change(fill_method=None).dropna()
        r6 = data.iloc[start_6m:end_idx + 1].pct_change(fill_method=None).dropna()
        r12 = data.iloc[start_12m:end_idx + 1].pct_change(fill_method=None).dropna()

        returns_3m = r3.mean()
        returns_6m = r6.mean()
        returns_12m = r12.mean()

        vol_3m = r3.std()
        vol_6m = r6.std()
        vol_12m = r12.std()

        # ✅ 0 변동성 방지
        vol_3m = vol_3m.replace(0, np.nan)
        vol_6m = vol_6m.replace(0, np.nan)
        vol_12m = vol_12m.replace(0, np.nan)

        risk_adj_3m = returns_3m / vol_3m
        risk_adj_6m = returns_6m / vol_6m
        risk_adj_12m = returns_12m / vol_12m

        avg_risk_adj_momentum = (risk_adj_3m + risk_adj_6m + risk_adj_12m) / 3

        # 이동평균선 이탈 필터
        if ma_filter_days is not None:
            below_ma = []
            for t in data.columns:
                series = data[t].iloc[:end_idx + 1]
                if len(series) < ma_filter_days:
                    below_ma.append(True)
                else:
                    ma_value = series.rolling(window=ma_filter_days).mean().iloc[-1]
                    price_value = series.iloc[-1]
                    below_ma.append(price_value < ma_value)
            below_ma = pd.Series(below_ma, index=data.columns)
            avg_risk_adj_momentum = avg_risk_adj_momentum[~below_ma]

        # ✅ NaN 제거 후 TopN
        avg_risk_adj_momentum = avg_risk_adj_momentum.dropna()
        top_list = avg_risk_adj_momentum.nlargest(top_n).index.tolist()
        top_names = [f"{t} ({industry_etfs.get(t, '')})" for t in top_list]

        # 혹시 부족하면 None 채우기(표 모양 유지)
        top_names += [None] * max(0, top_n - len(top_names))
        daily_top_industries[day] = top_names

    df = pd.DataFrame.from_dict(
        daily_top_industries,
        orient='index',
        columns=[f'Top {i+1}' for i in range(top_n)]
    )
    df.index.name = 'Date'
    return df

# ✅ 9. Sortino / ETR Comfort (모멘텀과 동일한 기간 구조)
#     - pct_change(fill_method=None) 적용
# ============================================================

def mdd_from_returns(ret: pd.Series) -> float:
    cum = (1 + ret.fillna(0)).cumprod()
    peak = cum.cummax()
    dd = (cum - peak) / peak
    return float(dd.min())

def sortino_from_returns(ret: pd.Series, mar: float = 0.0, periods_per_year: int = 252) -> float:
    """
    Sortino (연환산)
    annual_excess / downside_dev(annualized)
    """
    ret = ret.dropna()
    if len(ret) < 5:
        return np.nan

    mar_daily = (1 + mar) ** (1 / periods_per_year) - 1
    excess = ret - mar_daily
    downside = excess[excess < 0]
    if len(downside) < 5:
        return np.nan

    downside_dev = downside.std(ddof=1) * np.sqrt(periods_per_year)
    if downside_dev == 0 or np.isnan(downside_dev):
        return np.nan

    annual_excess = excess.mean() * periods_per_year
    return float(annual_excess / downside_dev)

def cagr_from_prices_window(prices: pd.Series, periods_per_year: int = 252) -> float:
    prices = prices.dropna()
    if len(prices) < 2:
        return np.nan
    n = len(prices)
    return float((prices.iloc[-1] / prices.iloc[0]) ** (periods_per_year / (n - 1)) - 1)

def etr_comfort_from_prices_window(prices: pd.Series, periods_per_year: int = 252) -> float:
    """
    ETR Comfort (윈도우 버전)
    ETR = CAGR / (|MDD| * Vol_annual)
    """
    prices = prices.dropna()
    if len(prices) < 60:
        return np.nan

    # ✅ FutureWarning 방지
    ret = prices.pct_change(fill_method=None).dropna()
    if len(ret) < 5:
        return np.nan

    cagr = cagr_from_prices_window(prices, periods_per_year=periods_per_year)
    vol = float(ret.std(ddof=1) * np.sqrt(periods_per_year))
    mdd = mdd_from_returns(ret)

    denom = abs(mdd) * vol
    if denom == 0 or np.isnan(denom) or np.isnan(cagr):
        return np.nan
    return float(cagr / denom)

def ma_ok_mask(close_df: pd.DataFrame, end_idx: int, ma_days: int) -> pd.Series:
    """
    end_idx 시점에서 '종가 >= MA(ma_days)' 만족 종목만 True
    """
    ok = {}
    for t in close_df.columns:
        s = close_df[t].iloc[:end_idx + 1].dropna()
        if len(s) < ma_days:
            ok[t] = False
        else:
            ma = s.rolling(ma_days).mean().iloc[-1]
            ok[t] = bool(s.iloc[-1] >= ma)
    return pd.Series(ok)

def rank_metric_like_momentum_periods(
    close_df: pd.DataFrame,
    metric: str = "sortino",  # "sortino" or "etr"
    period_3m: int = 63,
    period_6m: int = 126,
    period_12m: int = 252,
    last_n_days: int = 10,
    ma_filter_days: int = 10,
    top_n: int = 20,
    periods_per_year: int = 252,
    mar: float = 0.0,
):
    """
    변동성 모멘텀과 동일한 기간 구조:
    - 각 날짜(day) 기준으로
      3m/6m/12m 윈도우 점수 계산 -> 평균 점수로 랭킹
    - MA 필터는 해당 day 시점의 종가>=MA 만족 종목만 통과
    - 결과 폼: 최근 last_n_days x TopN
    """
    daily_top = {}
    trading_days = close_df.index[-last_n_days:]
    wins = [period_3m, period_6m, period_12m]

    for day in trading_days:
        end_idx = close_df.index.get_loc(day)
        ok = ma_ok_mask(close_df, end_idx=end_idx, ma_days=ma_filter_days)

        scores = {}
        for t in close_df.columns:
            if not ok.get(t, False):
                continue

            vals = []
            for win in wins:
                start_idx = max(0, end_idx - win + 1)
                w_prices = close_df[t].iloc[start_idx:end_idx + 1].dropna()

                # 데이터 부족 방지 (윈도우의 80% 미만이면 스킵)
                if len(w_prices) < max(20, int(win * 0.8)):
                    continue

                if metric == "sortino":
                    # ✅ FutureWarning 방지
                    r = w_prices.pct_change(fill_method=None).dropna()
                    v = sortino_from_returns(r, mar=mar, periods_per_year=periods_per_year)
                elif metric == "etr":
                    v = etr_comfort_from_prices_window(w_prices, periods_per_year=periods_per_year)
                else:
                    raise ValueError("metric은 'sortino' 또는 'etr'만 지원합니다.")

                if pd.notna(v):
                    vals.append(float(v))

            if len(vals) == 0:
                continue

            scores[t] = float(np.mean(vals))

        if len(scores) == 0:
            daily_top[day] = [None] * top_n
            continue

        ranked = pd.Series(scores).sort_values(ascending=False).head(top_n).index.tolist()
        row = [f"{x} ({industry_etfs.get(x, '')})" for x in ranked]
        row += [None] * max(0, top_n - len(row))
        daily_top[day] = row

    out = pd.DataFrame.from_dict(
        daily_top, orient="index",
        columns=[f"Top {i+1}" for i in range(top_n)]
    )
    out.index.name = "Date"
    return out

import numpy as np
import pandas as pd

def ibk_overheat_strength(
    close_df: pd.DataFrame,
    lookback_6m: int = 126,
    min_assets: int = 8,
    weight_mode: str = "linear",   # "linear" 또는 "rank_centered"
):
    """
    close_df: 업종ETF 종가 DataFrame (columns=tickers)
    반환:
      - series_overheat: 날짜별 과열강도 시계열 (daily)
      - latest_detail: (latest date) 업종별 6M return, rank, weight, weighted_return
    """

    df = close_df.copy().dropna(how="all")
    # 6M 수익률
    r6m = df.pct_change(lookback_6m, fill_method=None)

    overheat_vals = []
    details_latest = None

    for dt in r6m.index:
        cs = r6m.loc[dt].dropna()

        # 최소 업종 수 부족하면 스킵
        if len(cs) < min_assets:
            overheat_vals.append(np.nan)
            continue

        # 랭킹 (1이 가장 강함)
        rank = cs.rank(ascending=False, method="first")

        n = len(cs)

        if weight_mode == "linear":
            # 상위 업종일수록 큰 가중치: n, n-1, ..., 1
            w = (n - rank + 1).astype(float)
            w = w / w.sum()
        elif weight_mode == "rank_centered":
            # 상위는 +, 하위는 -가 되도록 중심화 (스프레드 확대에 더 민감)
            # 예: 상위는 양수, 하위는 음수 -> 분포의 표준편차가 "격차"를 더 잘 반영
            w_raw = ( (n + 1)/2 - rank ).astype(float)  # 상위 양수, 하위 음수
            # 스케일만 맞추기 위해 절대합으로 정규화
            denom = np.abs(w_raw).sum()
            if denom == 0:
                overheat_vals.append(np.nan)
                continue
            w = w_raw / denom
        else:
            raise ValueError("weight_mode는 'linear' 또는 'rank_centered'만 지원")

        weighted_return = cs * w

        # ✅ 과열 강도 = 가중수익률 분포의 표준편차
        strength = float(weighted_return.std(ddof=1))

        overheat_vals.append(strength)

        # 최신일 상세 저장
        details_latest = pd.DataFrame({
            "6M_Return": cs,
            "Rank": rank,
            "Weight": w,
            "Weighted_Return": weighted_return
        }).sort_values("Rank")

    series_overheat = pd.Series(overheat_vals, index=r6m.index, name="OverheatStrength")
    return series_overheat, details_latest
