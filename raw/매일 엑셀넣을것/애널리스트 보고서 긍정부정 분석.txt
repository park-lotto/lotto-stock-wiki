# KR-FinBERT로 PDF 감성비율 계산 (Colab)
# - 긴 문서 최적화: 줄→문장 분할, 청크 처리, 배치 추론
# - 표/숫자 라인 필터링, 초장문 안전 분할
# - 출력: 전체(긍/부/중립) + 중립 제외(긍/부 재계산)

import re, math
from pdfminer.high_level import extract_text
from transformers import AutoTokenizer, AutoModelForSequenceClassification, pipeline
from google.colab import files
import torch

# ====== 파라미터 ======
MODEL_NAME = "snunlp/KR-FinBert-SC"
USE_GPU = True
CHUNK_CHARS = 3000          # kss용 텍스트 청크 크기
NUMERIC_SKIP = 0.55         # 숫자 비중 높으면(표/라벨) 라인 제거
MIN_SENT_LEN = 5            # 너무 짧은 문장 제거
MAX_SENT_CHARS = 1800       # 초장문 방어용 최대 길이
BATCH_SIZE_GPU = 96
BATCH_SIZE_CPU = 32

# ====== 파일 업로드 ======
uploaded = files.upload()
pdf_path = list(uploaded.keys())[0]

# ====== PDF → 텍스트 ======
raw_text = extract_text(pdf_path) or ""
if not raw_text.strip():
    raise ValueError("PDF에서 텍스트를 추출하지 못했습니다. (스캔본이면 OCR 필요)")

# ====== 숫자/표 라인 제거 & 정리 ======
def is_numeric_heavy(line, th=NUMERIC_SKIP):
    if not line: return False
    n = sum(ch.isdigit() for ch in line)
    return (n / len(line)) >= th

lines = [ln.strip() for ln in raw_text.split("\n")]
lines = [ln for ln in lines if ln and not is_numeric_heavy(ln)]
text = " ".join(lines)
text = re.sub(r"\s+", " ", text).strip()

# ====== 문장 분리 (kss 우선, 실패 시 정규식) ======
try:
    import kss
    sents = []
    for i in range(0, len(text), CHUNK_CHARS):
        part = text[i:i+CHUNK_CHARS]
        try:
            sents.extend(kss.split_sentences(part))
        except:
            sents.extend(re.split(r'(?<=[.!?])\s+', part))
except Exception:
    sents = re.split(r'(?<=[\.!\?])\s+|\n+', text)

# 공백/짧은 문장/초장문 방어
clean = []
for s in sents:
    s = (s or "").strip()
    if len(s) < MIN_SENT_LEN:
        continue
    if len(s) <= MAX_SENT_CHARS:
        clean.append(s)
    else:
        # 초장문은 구두점 기준 쪼개기 후 잘라서 보관
        parts = re.split(r'(?<=[\.\!\?])\s+', s)
        for p in parts:
            p = p.strip()
            if len(p) >= MIN_SENT_LEN:
                clean.append(p[:MAX_SENT_CHARS])

sentences = clean
print(f"총 문장 수(전처리 후): {len(sentences)}")

if not sentences:
    raise ValueError("문장으로 분리된 텍스트가 없습니다. 전처리 파라미터를 조정해 보세요.")

# ====== 모델 로드 ======
device = 0 if (USE_GPU and torch.cuda.is_available()) else -1
batch_size = BATCH_SIZE_GPU if device == 0 else BATCH_SIZE_CPU

tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME)
# ✅ BERT 최대 길이 명시
tokenizer.model_max_length = 512

model = AutoModelForSequenceClassification.from_pretrained(MODEL_NAME)
clf = pipeline(
    "text-classification",
    model=model,
    tokenizer=tokenizer,
    device=device,
    top_k=None  # 모든 라벨 점수 반환
)

# ====== 배치 추론 (truncation/padding 강제) ======
def infer_batch(sent_list, batch_size):
    pos = neg = neu = 0
    for i in range(0, len(sent_list), batch_size):
        batch = sent_list[i:i+batch_size]
        outs = clf(
            batch,
            truncation=True,
            max_length=512,
            padding=True
        )
        for out in outs:
            # top_k=None → [{"label": "...", "score": ...}, ...] 리스트
            scores = {d['label'].lower(): d['score'] for d in out}
            label = max(scores, key=scores.get)
            if label == "positive":
                pos += 1
            elif label == "negative":
                neg += 1
            else:
                neu += 1
    return pos, neg, neu

pos, neg, neu = infer_batch(sentences, batch_size)
total = pos + neg + neu if (pos + neg + neu) > 0 else 1

# ====== 1) 전체 비율 ======
pos_all = pos / total * 100
neg_all = neg / total * 100
neu_all = neu / total * 100

# ====== 2) 중립 제외 재계산 ======
pn_total = pos + neg
if pn_total > 0:
    pos_pn = pos / pn_total * 100
    neg_pn = neg / pn_total * 100
else:
    pos_pn = neg_pn = float("nan")

# ====== 출력 ======
print("\n📊 결과 (전체 문장 기준)")
print(f"긍정: {pos_all:.2f}%  | 부정: {neg_all:.2f}%  | 중립: {neu_all:.2f}%")

print("\n📊 결과 (중립 제외, 긍/부만 재계산)")
if math.isnan(pos_pn):
    print("분석 가능한 긍/부 문장이 충분치 않습니다.")
else:
    print(f"긍정(중립 제외): {pos_pn:.2f}%  | 부정(중립 제외): {neg_pn:.2f}%")
    print("✅ 긍정>부정" if pos_pn >= 50 else "⚠️ 부정≥긍정")
