import pandas as pd
import yfinance as yf
from datetime import datetime

# ---------------------------
# 1) 티커 설정
# ---------------------------
tickers = [
    "SPY", "IWM", "VEA", "VWO", "VNQ",
    "DBC", "IEF", "TLT", "TIP", "BIL",
    "HYG", "LQD", "EMB", "BNDX", "BWX"
]

# ---------------------------
# 2) 사용자로부터 특정 날짜 입력 (월중 가능)
# ---------------------------
specific_date = input("투자일(YYYY-MM-DD)을 입력하세요: ")
# 예) 2023-11-10

target_dt = datetime.strptime(specific_date, '%Y-%m-%d')

# ---------------------------
# 3) 데이터 다운로드 (과거 10년치) + 특정 날짜까지 필터링
# ---------------------------
end_date = target_dt.strftime('%Y-%m-%d')
start_date = (target_dt - pd.DateOffset(years=10)).strftime('%Y-%m-%d')

print(f"\n[{start_date} ~ {end_date}] 기간에 대해 다운로드합니다.")
data = yf.download(tickers, start=start_date, end=end_date)['Close']

# 해당 날짜보다 이후 데이터가 있을 수 있으므로, 혹시 몰라서 다시 필터링
data = data.loc[:specific_date]

# ---------------------------
# 4) 월말 데이터 추출
# ---------------------------
monthly_data = data.resample('M').last()

# ---------------------------
# 5) 모멘텀 계산 함수
# ---------------------------
def calculate_momentum(series):
    """
    월말 기준 1M/3M/6M/12M 수익률 평균 => 모멘텀
    """
    df = pd.DataFrame(index=series.index)
    df['1M']  = series.pct_change(periods=1)
    df['3M']  = series.pct_change(periods=3)
    df['6M']  = series.pct_change(periods=6)
    df['12M'] = series.pct_change(periods=12)
    return df.mean(axis=1)

# 모든 종목에 대해 (월말 데이터) 모멘텀 계산
momentum_data = monthly_data.apply(calculate_momentum)

# ---------------------------
# 6) TIP (카나리아) 모멘텀 계산
# ---------------------------
def canary_momentum(tip_series):
    tip_mom = calculate_momentum(tip_series)
    return (tip_mom > 0), tip_mom

if 'TIP' not in monthly_data.columns:
    print("\n[오류] TIP 데이터가 충분하지 않습니다.")
    exit()

canary_signal, tip_momentum = canary_momentum(monthly_data['TIP'])

# ---------------------------
# 7) 안전 자산 선택 함수
# ---------------------------
def select_safe_assets(momentum_df, bil_momentum, top_n=3):
    safe_assets = ["BIL", "TLT", "HYG", "LQD", "IEF", "EMB", "TIP", "BNDX", "BWX"]
    # 가장 최근(마지막 row) 모멘텀만 본다
    safe_mom = momentum_df[safe_assets].iloc[-1]
    top_safe = safe_mom.nlargest(top_n)
    
    selected = []
    for asset in top_safe.index:
        if top_safe[asset] > bil_momentum:
            selected.append(asset)
    
    # 아무것도 안 뽑히면 BIL만
    return selected if selected else ['BIL']

# ---------------------------
# 8) 공격 자산 선택 함수
# ---------------------------
def select_top_assets(momentum_df, canary_signal_series, top_n=4):
    """
    - 카나리아 신호 True => 공격 자산 상위 top_n
    - 카나리아 신호 False => 안전 자산
    """
    if canary_signal_series.iloc[-1]:
        top_assets = momentum_df.iloc[-1].nlargest(top_n)
        return top_assets.index.tolist()
    else:
        if 'BIL' not in monthly_data.columns:
            print("[오류] BIL 데이터가 부족합니다.")
            return []
        # BIL 모멘텀
        bil_momentum = calculate_momentum(monthly_data['BIL']).iloc[-1]
        return select_safe_assets(momentum_df, bil_momentum, top_n=3)

# ---------------------------
# 9) 현재(마지막 월말) 기준 모멘텀/카나리아로 자산 선정
# ---------------------------
if len(monthly_data) < 13:
    print("\n[주의] 12개월 이상 월말 데이터가 부족할 수 있습니다 (12M 모멘텀 NaN).")

latest_canary_signal = canary_signal.iloc[-1]  # 최근 TIP 모멘텀 양/음
top_assets = select_top_assets(momentum_data, canary_signal, top_n=4)

# ---------------------------
# 10) 투자 비중 출력
# ---------------------------
if len(top_assets) == 0:
    print("\n[결과] 선택된 자산이 없습니다. (데이터 부족 또는 오류)")
else:
    weight = 1 / len(top_assets)

    print(f"\n[투자일: {specific_date}] (월중 가정, {len(monthly_data)}개월 데이터 기반)")
    print("카나리아 신호 =", latest_canary_signal, "(True면 TIP 모멘텀 0 이상)")

    print("\n[선정된 자산 & 균등 비중]")
    for asset in top_assets:
        if asset not in monthly_data.columns:
            continue
        w_str = f"{weight*100:.2f}%"
        print(f"  - {asset:5s} : {w_str}")

print("\n프로그램 종료.")
