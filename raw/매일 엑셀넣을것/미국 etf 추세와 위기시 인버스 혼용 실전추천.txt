import yfinance as yf
import pandas as pd
import numpy as np
from datetime import datetime, timedelta
import sys

# -----------------------------------------------------------
# 함수 정의: 투자 추천
# -----------------------------------------------------------
def recommend_portfolio(end_date_str):
    """
    지정된 최종 날짜(end_date_str)까지의 월간 데이터를 기반으로 다음 달에 투자할 자산을 추천합니다.

    Parameters:
    - end_date_str (str): 투자 분석의 기준이 되는 최종 날짜 (예: '2024-12-26')

    Returns:
    - 추천 메시지 (str)
    """

    # -----------------------------------------------------------
    # 1. 데이터 다운로드 (월간)
    # -----------------------------------------------------------
    tickers = ["QQQ", "SOXX", "QLD", "PSQ", "SPMO", "TIP"]  # ✅ TIP 추가!

    # 최종 날짜 입력 검증 및 설정
    try:
        end_date = pd.to_datetime(end_date_str)
    except ValueError:
        raise ValueError("최종 날짜 형식이 올바르지 않습니다. 'YYYY-MM-DD' 형식으로 입력해주세요.")

    # 분석 날짜가 미래인지 확인
    today = pd.Timestamp.today()
    if end_date > today:
        raise ValueError("최종 날짜가 현재 날짜보다 미래입니다. 과거의 날짜를 입력해주세요.")

    # 최종 날짜를 월말로 조정
    if end_date != end_date + pd.offsets.MonthEnd(0):
        end_date = end_date + pd.offsets.MonthEnd(0)

    # 시작 날짜 계산 (6개월 전)
    start_date = end_date - pd.DateOffset(months=6)
    start_date_str = start_date.strftime('%Y-%m-%d')

    print(f"데이터 다운로드 기간: {start_date_str} ~ {end_date.strftime('%Y-%m-%d')}")

    # 월간 데이터 다운로드
    try:
        df = yf.download(
            tickers=tickers,
            start=start_date_str,
            end=(end_date + pd.DateOffset(days=1)).strftime('%Y-%m-%d'),  # end 파라미터는 미포함
            interval="1mo",
            progress=False
        )
    except Exception as e:
        raise RuntimeError(f"데이터 다운로드 중 오류가 발생했습니다: {e}")

    # -----------------------------------------------------------
    # 2. 종가만 추출 및 멀티 인덱스 처리
    # -----------------------------------------------------------
    if 'Close' not in df.columns:
        raise ValueError("종가 데이터가 없습니다.")

    df_close = df["Close"].dropna(how="all")

    if isinstance(df_close.columns, pd.MultiIndex):
        df_close.columns = df_close.columns.droplevel(0)

    print("종가 데이터가 성공적으로 로드되었습니다.")
    print("df_close.columns =>", df_close.columns)

    # -----------------------------------------------------------
    # 3. 데이터 검증: 충분한 데이터 확보 여부 확인
    # -----------------------------------------------------------
    required_months = 6
    if len(df_close) < required_months:
        raise ValueError(f"충분한 데이터가 없습니다. 최소 {required_months}개월 이상의 데이터가 필요합니다.")

    # -----------------------------------------------------------
    # 4. 모멘텀(추세) 판별:
    #    - QQQ와 TIP 모두 종가 > 6개월 이동평균이면 상승 추세
    # -----------------------------------------------------------
    required_tickers = ["QQQ", "TIP"]
    for t in required_tickers:
        if t not in df_close.columns:
            raise ValueError(f"{t} 컬럼이 없습니다. 야후 파이낸스에서 {t} 데이터를 불러오지 못했습니다.")

    qqq = df_close["QQQ"].dropna()
    tip = df_close["TIP"].dropna()

    # 이동평균 6개월
    window_months = 6
    qqq_ma6 = qqq.rolling(window=window_months).mean()
    tip_ma6 = tip.rolling(window=window_months).mean()

    # 분석 날짜에 해당하는 데이터 확인
    if qqq_ma6.index[-1] != end_date or tip_ma6.index[-1] != end_date:
        latest_date = qqq_ma6.index[-1]
    else:
        latest_date = end_date

    # 최신 종가와 이동평균 가져오기
    latest_close_qqq = qqq.loc[latest_date]
    latest_ma6_qqq = qqq_ma6.loc[latest_date]

    latest_close_tip = tip.loc[latest_date]
    latest_ma6_tip = tip_ma6.loc[latest_date]

    # 디버깅용 출력
    print(f"최신 종가 (QQQ): {latest_close_qqq:.2f}")
    print(f"6개월 이동평균 (QQQ): {latest_ma6_qqq:.2f}")
    print(f"최신 종가 (TIP): {latest_close_tip:.2f}")
    print(f"6개월 이동평균 (TIP): {latest_ma6_tip:.2f}")

    # 추세 판단: 둘 다 6개월 이동평균선 위
    uptrend = (latest_close_qqq > latest_ma6_qqq) and (latest_close_tip > latest_ma6_tip)

    print(f"\n=== 추세 판단 (QQQ + TIP) ===")
    if uptrend:
        recommendation = (
            "🚀 상승추세 → 다음 달에는 QQQ 60%, QLD 15%, SOXX 25% 투자"
        )
    else:
        recommendation = (
            "📉 하락추세 → 다음 달에는 SPMO 50%, PSQ 50% 투자"
        )

    print(recommendation)

    return recommendation

# -----------------------------------------------------------
# 사용자 입력 및 함수 실행
# -----------------------------------------------------------
if __name__ == "__main__":
    try:
        end_date_input = input("날짜는 미국 종가 날짜를 넣으세요 (예: '2024-12-26'): ").strip()
        recommendation_message = recommend_portfolio(end_date_input)
        print("\n===== 투자 추천 =====")
        print(recommendation_message)

    except Exception as e:
        print(f"오류 발생: {e}")
        sys.exit(1)
