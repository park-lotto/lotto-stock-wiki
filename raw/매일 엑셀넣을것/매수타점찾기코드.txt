import yfinance as yf
import pandas as pd
import numpy as np

# 티커 리스트 입력
tickers = input("분석할 티커들을 쉼표로 구분하여 입력하세요 (예: AAPL, MSFT, GOOG): ").split(',')

for ticker in tickers:
    ticker = ticker.strip().upper()  # 티커를 대문자로 변환

    # 1년 동안의 주가 데이터를 가져옴
    stock_data = yf.download(ticker, period='1y')

    # 전일 종가 대비 당일 저가 괴리율 계산 (저가 - 하루 전 종가)
    stock_data['저가_종가_괴리'] = (stock_data['Low'] - stock_data['Close'].shift(1)) / stock_data['Close'].shift(1) * 100

    # 값이 있는 날만 필터링
    괴리율_있는_날 = stock_data['저가_종가_괴리'].dropna()

    # 저가-종가 괴리율의 평균 계산
    평균_저가_종가_괴리율 = 괴리율_있는_날.mean()

    # 당일 고가 대비 종가의 평균 하락율 계산
    stock_data['고가_종가_하락율'] = (stock_data['Close'] - stock_data['High']) / stock_data['High'] * 100
    고가_종가_평균_하락율 = stock_data['고가_종가_하락율'].mean()

    # 시가와 저가의 괴리율 계산 (시가 - 저가)
    stock_data['시가_저가_괴리'] = (stock_data['Low'] - stock_data['Open']) / stock_data['Open'] * 100
    평균_시가_저가_괴리율 = stock_data['시가_저가_괴리'].mean()

    # 전일 종가 대비 당일 고가 괴리율 계산
    stock_data['전일_종가_고가_괴리'] = (stock_data['High'] - stock_data['Close'].shift(1)) / stock_data['Close'].shift(1) * 100
    평균_전일_종가_고가_괴리율 = stock_data['전일_종가_고가_괴리'].mean()

    # 당일 시가 대비 당일 고가 괴리율 계산
    stock_data['시가_고가_괴리'] = (stock_data['High'] - stock_data['Open']) / stock_data['Open'] * 100
    평균_시가_고가_괴리율 = stock_data['시가_고가_괴리'].mean()

    # 티커의 이름 가져오기
    company_name = yf.Ticker(ticker).info.get('longName', 'Company Name Not Found')

    # 결과 출력
    print(f"1 of 1 completed")
    print(f"[티커: {ticker}] ({company_name})")
    print("1년 간 데이터 분석 결과:")
    print(f"당일 고가 대비 종가 평균 하락율: {고가_종가_평균_하락율:.2f}%")
    print(f"당일 저가가 하루 전 종가와의 평균 괴리율: {평균_저가_종가_괴리율:.2f}%")
    print(f"시가 대비 저가의 평균 괴리율: {평균_시가_저가_괴리율:.2f}%")
    print(f"전일 종가 대비 당일 고가의 평균 괴리율: {평균_전일_종가_고가_괴리율:.2f}%")
    print(f"당일 시가 대비 당일 고가의 평균 괴리율: {평균_시가_고가_괴리율:.2f}%")
