import yfinance as yf
import pandas as pd
import numpy as np

# =========================================================
# 1. 티커 (논문 ETF 유니버스)
# =========================================================
tickers = [
    "SPY", "QQQ", "IWM", "VEA", "VWO",
    "TLT", "IEF", "LQD", "EMB", "TIP",
    "GLD", "GSG", "VNQ"
]

start_date = "2020-01-01"

# =========================================================
# 2. 데이터 다운로드 (Close 사용)
# =========================================================
data = yf.download(tickers, start=start_date)["Close"]

# 데이터 없는 컬럼 제거
data = data.dropna(axis=1)

# 수익률
returns = data.pct_change().dropna()

# =========================================================
# 3. 모멘텀 (6, 9, 12개월 평균)
# =========================================================
mom_6 = data.pct_change(126)
mom_9 = data.pct_change(189)
mom_12 = data.pct_change(252)

momentum = (mom_6 + mom_9 + mom_12) / 3

# 최신 모멘텀
mu = momentum.iloc[-1].dropna()

# =========================================================
# 4. 공분산 (최근 2년)
# =========================================================
lookback = min(504, len(returns))

hist_returns = returns.iloc[-lookback:]
cov = hist_returns.cov()

# 공분산 정렬
cov = cov.loc[mu.index, mu.index]

# =========================================================
# 5. 평균분산 최적화
# =========================================================
inv_cov = np.linalg.inv(cov)

weights = pd.Series(inv_cov @ mu, index=mu.index)

# 정규화
weights = weights / weights.sum()

# =========================================================
# 6. 롱온리 (음수 제거) 🔥 핵심
# =========================================================
weights[weights < 0] = 0

# =========================================================
# 7. 절대 모멘텀
# =========================================================
weights[mu < 0] = 0

# =========================================================
# 8. 상대 모멘텀 (상위 5개)
# =========================================================
top_n = 5
top_assets = mu.sort_values(ascending=False).head(top_n).index

weights = weights[weights.index.isin(top_assets)]

# =========================================================
# 9. 재정규화
# =========================================================
if weights.sum() > 0:
    weights = weights / weights.sum()
else:
    print("⚠️ 모든 자산 모멘텀이 음수 → 투자 없음")

# =========================================================
# 10. 결과 출력
# =========================================================
result = pd.DataFrame({
    "Momentum": mu,
    "Weight": weights
}).dropna().sort_values(by="Weight", ascending=False)

print("\n📊 현재 기준 포트폴리오 비중 (롱온리)\n")
print(result)

print("\n총합:", result["Weight"].sum())
