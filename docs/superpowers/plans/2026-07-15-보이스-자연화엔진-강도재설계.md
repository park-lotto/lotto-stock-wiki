# 보이스 자연화엔진 강도 재설계 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 사장님이 노브를 돌리면 텍스트와 소리가 실제로 단계적으로 변하고, 한 번 맞춘 말투규칙이 모든 성우(앞으로 추가할 성우 포함)에 자동 적용된다.

**Architecture:** 순수 엔진(`narration_naturalize.py`)의 각 스테이지를 "후보 목록 × 비율"의 비례형으로 바꾸고, 원노브(`naturalness` 0~1)가 5단계 매핑표로 8스테이지 값을 한꺼번에 세팅한다. 말투규칙은 `settings` 테이블에 전역 1개로 저장하고 성우별로는 발음 예외만 병합한다. 엔진이 스테이지별 적용 횟수(`applied`)와 경고를 반환해 작업대가 "무엇이 실제로 적용됐는지"를 보여준다.

**Tech Stack:** Python 3.12 · pytest · FastAPI · SQLite(`store.py`) · 바닐라 JS(`voice_tune.html`) · ElevenLabs(eleven_v3) · GROQ Whisper

```python
_ROLE_ALIASES = {
    "훅": "훅", "hook": "훅",
    "페인포인트": "페인포인트", "painpoint": "페인포인트", "pain point": "페인포인트",

    "cta": "CTA",

def normalize_role(role):
    """실제 role(한글/영어/대소문자/동의어) → 정본. 미지면 None."""
    if not role:
        return None
    return _ROLE_ALIASES.get(str(role).strip().lower())


# 정본 role별 감정 태그. **알려진 v3 태그만 사용**(새 태그를 지어내면 그대로 읽힐 위험).

    "훅": "[curious]",
    "페인포인트": None,        # 문제 제기 구간 — 무태그(태그 도배 금지)

    "CTA": "[excited]",
```

```python
def _emotion_arc(text, cfg, ctx):
    intensity = cfg.get("intensity", 0.3)
    if intensity < 0.15:
        return text
    if ctx["caps"].get("max_tags_per_beat", 1) <= 0:
        return text
    tag = _tag_for(ctx)
    return f"{tag} {text}" if tag else text


def _tag_for(ctx):
    """정본 role → 태그. 미지 role은 경고 후 위치기반 폴백."""
    raw = ctx.get("beat_role")
    canon = normalize_role(raw)
    if canon:
        return _ARC_BY_ROLE[canon]
    if raw:
        ctx["warnings"].append(f"미지 role '{raw}' — 위치기반으로 폴백함(별칭표 추가 검토)")
    if ctx.get("beat_index") is None or not ctx.get("beat_total"):
        return None
    n = max(1, ctx["beat_total"] - 1)
    raw_pos = round((ctx["beat_index"] / n) * (len(_ARC_BY_POS) - 1))
    pos = min(len(_ARC_BY_POS) - 1, max(0, raw_pos))
    return _ARC_BY_POS[pos]
```

```python
    """naturalize와 같지만 {"text", "applied", "warnings"}를 반환.

    applied = 스테이지별 실제 적용 횟수. 작업대가 이걸 보여줘서 "슬라이더를 돌렸는데

    out = _enforce_total_tag_cap(out, p.get("caps", {}).get("max_tags_total", 3))

    return naturalize_detail(text, profile, beat_role=beat_role,
```

```python
def normalize_reading(text):
    """숫자·단위·기호를 한국어 읽기로 바꾼다 → (변환텍스트, 적용횟수).

    정규화 스테이지의 순수 알맹이. **오독경보(asr_check)도 이 함수를 재사용한다** —
    Whisper가 '삼 점 오 킬로그램'을 다시 '3.5kg'로 표기해버려서, 양쪽을 같은 표기로
    맞추지 않으면 성우가 제대로 읽었는데도 전부 오독으로 뜬다(2026-07-15 실측).
    """
    def num_repl(m):
        return _num_to_words(m.group(0))
    def numunit(m):
        return f"{_num_to_words(m.group(1))} {_UNIT_MAP[m.group(2)]}"
    unit_pat = r"(\d+(?:\.\d+)?)(" + "|".join(sorted(_UNIT_MAP, key=len, reverse=True)) + r")"
    text, n1 = re.subn(unit_pat, numunit, text)
    text, n2 = re.subn(r"\d+(?:\.\d+)?", num_repl, text)
    n3 = 0
    for sym, word in _SYMBOL_MAP.items():
        n3 += text.count(sym)
        text = text.replace(sym, " " + word)
    text = re.sub(r" {2,}", " ", text)
    return text, n1 + n2 + n3


def _normalize(text, cfg, ctx):
    text, n = normalize_reading(text)
    _bump(ctx, "normalize", n)
    return text
```

```python
def _take_count(n_candidates, intensity):
    """후보 n개 중 강도 비율만큼 앞에서부터 적용할 개수(결정적).
```

```python
#    애매한 단음절은 버린다(놓치는 건 안전, 오탐은 성우가 엉뚱한 곳에서 쉰다).
#    lookahead `(?=\s|$)`는 **완화 금지** — 풀면 dot 후보와 tail 후보가 겹쳐 이중 삽입된다.

def _endings(text, cfg, ctx):

    if intensity <= 0:
        return text
    # 후보: ① 마침표 ② 마침표 없는 종결어미. 위치 순으로 앞에서부터 강도 비율만.
    cands = [(m.start(), m.end(), "dot") for m in re.finditer(r"\.(?=\s|$)", text)]
    cands += [(m.start(), m.end(), "tail") for m in _ENDING_TAIL.finditer(text)]
    cands.sort()
    take = _take_count(len(cands), intensity)
    if take <= 0:
        return text
    # 오프셋이 밀리지 않도록 뒤에서부터 적용
    for start, end, kind in sorted(cands[:take], key=lambda c: c[0], reverse=True):
        if kind == "dot":
            text = text[:start] + "…" + text[end:]
        else:
            text = text[:end] + "…" + text[end:]
    _bump(ctx, "endings", take)
    return text
```

```python
def _sentence_starts(text):
    """문장 시작 오프셋들. 추임새를 문장 앞에 붙이기 위한 후보 목록."""
    starts = [0]
    for m in re.finditer(r"(?<=[.!?…])\s+", text):
        if m.end() < len(text):
            starts.append(m.end())
    return starts


def _fillers(text, cfg, ctx):
    intensity = cfg.get("intensity", 0.2)
    bank = cfg.get("bank") or ["음"]
    cap = ctx["caps"].get("max_fillers_per_text", 1)
    if intensity <= 0 or cap <= 0:
        return text
    # 비트 빈도: 낮은 강도 = 'N비트마다 1번'(도배 방지). 1.0이면 매 비트.
    every = 1 if intensity >= 1.0 else max(1, round(1.0 / intensity))
    bi = ctx.get("beat_index") or 0
    if bi % every != 0:
        return text
    # 텍스트 내 개수: 문장 수 × 강도(올림), cap 이하.
    starts = _sentence_starts(text)
    n = min(cap, _take_count(len(starts), intensity))
    if n <= 0:
        return text
    for k, pos in enumerate(sorted(starts[:n], reverse=True)):
        filler = bank[(bi + (n - 1 - k)) % len(bank)]   # 비트·문장별 결정적 순환
        text = text[:pos] + f"{filler}, " + text[pos:]
    _bump(ctx, "fillers", n)
    return text
```

```python
def _tag_priority(total):
    """태그를 붙일 비트 우선순위: 훅(첫) → 마지막(CTA) → 나머지 앞에서부터.
    강도가 낮을 때 어디에 태그를 쓸지의 예산 배분이다."""
    if total <= 0:
        return []
    order = [0]
    if total > 1:
        order.append(total - 1)
    order += [i for i in range(1, total - 1)]
    return order

def _emotion_arc(text, cfg, ctx):

    if intensity <= 0 or ctx["caps"].get("max_tags_per_beat", 1) <= 0:

    # ★ 여기에 현재 파일의 `max_tags_total <= 0` 게이트를 **반드시 유지**할 것(Task 2 수정분)

    n_tagged = _take_count(total, intensity)

    tag = _tag_for(ctx)
    if not tag:
```

```python
# v3 태그를 더 쓰지 않고 구두점만으로 만든다 — 태그 캡을 감정곡선이 이미 쓰고 있고,
# 미지원 태그를 지어내면 그대로 읽힐 위험이 있기 때문.
_EMPHASIS_WORDS = ["진짜", "완전", "역대급", "훨씬", "절대", "딱"]


def _intonation(text, cfg, ctx):
    intensity = cfg.get("intensity", 0.2)
    if intensity <= 0:
        return text
    pat = r"(?<=[^\s,.!?…])(\s+)(" + "|".join(_EMPHASIS_WORDS) + r")"
    cands = [m.start(1) for m in re.finditer(pat, text)]
    take = _take_count(len(cands), intensity)
    if take <= 0:
        return text
    for pos in sorted(cands[:take], reverse=True):
        text = text[:pos] + "," + text[pos:]
    _bump(ctx, "intonation", take)
    return text
```

```python
def resolve_profile(profile):
    """naturalness(0~1) 노브를 5단계 매핑으로 펼쳐 8스테이지 값을 세팅한 프로파일 반환.

    """
    p = dict(profile or {})
    if "naturalness" not in p or p.get("manual"):
        return p
    try:
        knob = float(p["naturalness"])
    except (TypeError, ValueError):
        return p
    knob = min(1.0, max(0.0, knob))
    level = int(round(knob * (len(NATURALNESS_LEVELS) - 1)))
    out = copy.deepcopy(NATURALNESS_LEVELS[level])
    out["_level"] = level
    # 노브가 만들지 않는 값(seed·n_best·발음사전 등)은 원본 유지
    for k, v in p.items():
        if k not in out:
            out[k] = v
    return out
```

```python
def merge_profile(profile):
    """유저 프로파일을 DEFAULT_PROFILE 위에 1단계 병합(빈 값은 기본으로 채움).

    naturalness 노브가 있으면 먼저 5단계 매핑으로 펼친다(resolve_profile).

    최상위 dict 값(스테이지·caps)은 `.update`로 얕게 합쳐지므로 그 내부에 중첩된
    dict(예: pronunciation.dict)를 부분 지정하면 통째로 교체된다(재귀 병합 아님)."""
    profile = resolve_profile(profile)
    out = copy.deepcopy(DEFAULT_PROFILE)
    for k, v in (profile or {}).items():
        if isinstance(v, dict) and isinstance(out.get(k), dict):
            out[k].update(v)
        else:
            out[k] = v
    return out
```

```python
def _canon(text):
    """비교 전용 표준형. 표시는 원문 기준이고 이 변환은 오직 비교에만 쓴다.

    Whisper는 우리가 한글로 풀어 읽힌 숫자를 다시 아라비아 숫자로 표기한다
    (성우는 정확히 읽었는데 '삼 점 오 킬로그램' vs '3.5kg'로 전부 sub가 됐다 —
    2026-07-15 실측). 그래서 hyp에도 엔진과 **같은 규칙**(normalize_reading)을 태워
    표기를 맞춘다. 한국어 띄어쓰기 차이도 오독이 아니므로 공백을 지운다.
    """
    t = re.sub(r"\[[^\]]+\]", " ", text or "")     # 감정태그는 발화되지 않음
    t = re.sub(r"(?<=\d),(?=\d)", "", t)           # 2,000 → 2000 (숫자 패턴 매칭용)
    t, _ = normalize_reading(t)                    # 3.5kg → 삼 점 오 킬로그램 (양쪽 동일 규칙)
    t = re.sub(r"[^\w가-힣]", "", t)               # 문장부호·공백 제거(띄어쓰기 오탐 방지)
    return t


def _tokens(text):
    """표시용 토큰: 감정태그·문장부호 제거, 공백분리."""
    text = re.sub(r"\[[^\]]+\]", " ", text or "")
    text = re.sub(r"[^\w가-힣\s]", " ", text)
    return [t for t in text.split() if t]
```

```python
    if _canon(ref_text) == _canon(hyp_text):
        return {"ok": True, "words": [{"kind": "equal", "ref": t, "hyp": t}
                                      for t in _tokens(ref_text)]}
    ref, hyp = _tokens(ref_text), _tokens(hyp_text)
    ...  # 이하 기존 코드 그대로
```
