# 다중 PC 관제·분배 Implementation Plan (카드 046)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 텔레그램 관제창이 `py tools/dispatch.py <트랙> "<지시>"` 한 줄로 트랙 폴더에 이름 붙은 원격 조종 창을 열고, 그 창이 끝나면 텔레그램으로 결과를 보고하며, 같은 구성을 회사 PC에 스크립트 한 번으로 깐다.

**Architecture:** 창 여는 절차는 `tools/track_open.py` 한 곳(계정 환경·이름·설정 조립 = 순수 함수 `build_launch`, 실제 실행 = `launch_window`). `dispatch.py` 는 살아 있는 창 검사 + 카드 연결 + `launch_window` 호출만. `tg_notify.py` 는 Bot API sendMessage(수신 poll 과 안 겹침). PC 이름은 `pc_label.py` 한 곳. PC 설치물 원본은 `tools/pc/` 에 두고 `pc_setup.ps1` 이 사용자 폴더로 복사한다.

**Tech Stack:** Python 3 (표준 라이브러리만 — psutil 없음, `tasklist` 사용), pytest, Windows PowerShell 5.1, Claude Code CLI (`--remote-control`, `--continue`, `--settings`), Codex CLI.

스펙: `docs/superpowers/specs/2026-09-30-다중PC관제분배-design.md`. 모든 명령은 트랙 폴더 `.tracks/관제분배` 에서 실행.

```python
"""이 PC 이름(집/회사) — 창 이름·보고 머리말이 쓰는 한 곳 (카드 046).

`%USERPROFILE%\\.claude-pc.json` 의 {"label": "집"} 을 읽는다. 없거나 깨졌으면 컴퓨터 이름.

def label(home=None, computer=None):
    home = Path(home or Path.home())
    computer = computer or os.environ.get("COMPUTERNAME", "PC")
    try:
        data = json.loads((home / ".claude-pc.json").read_text(encoding="utf-8"))

        return computer
```

```python
#: 계정 번호 → CLAUDE_CONFIG_DIR. 1번(본계정)은 변수를 **지운다** — 걸면 ~\.claude\.claude.json 을
#: 찾아 빈 설정을 새로 만든다(프로필 주석, 2026-09-21 실측).
ACCOUNT_DIRS = {"1": None, "2": ".claude-sub", "3": ".claude-sub3"}


def build_launch(cwd, *, account="1", agent="claude", model="", prompt="", pc="", tag="",
                 card=None, resume=True, home=None):
    """새 창에서 돌릴 명령·환경·이름을 조립한다(실행 없음 — 테스트 가능한 한 곳, 카드 046).

    이름 = "PC·계정·용도[·카드N]" — Claude 앱 Code 목록에서 두 PC 창을 가른다.
    Claude 창은 텔레그램 플러그인을 끈 설정으로 띄운다: 봇 수신자는 관제창 하나뿐이어야 한다(409).
    """
    account = str(account)
    if account not in ACCOUNT_DIRS:
        raise ValueError(f"없는 계정: {account} (1/2/3)")
    home = Path(home or Path.home())
    name = "·".join([pc, account, tag] + ([f"카드{card}"] if card else []))
    env, unset = {}, []
    if ACCOUNT_DIRS[account]:
        env["CLAUDE_CONFIG_DIR"] = str(home / ACCOUNT_DIRS[account])
    else:
        unset.append("CLAUDE_CONFIG_DIR")
    if agent == "codex":
        argv = list(AGENTS["codex"])
    else:
        argv = ["claude", "--remote-control", name,
                "--settings", str(home / "claude-remote-settings.json")]
        if resume:
            argv.append("--continue")
        if model:
            argv += ["--model", model]
    if prompt:
        argv.append(prompt)
    return {"argv": argv, "env": env, "env_unset": unset, "name": name, "cwd": str(cwd)}
```

```python
import json
import os
import tempfile

SESSION_FILE = ".session.json"


def _ps_quote(s):
    return "'" + str(s).replace("'", "''") + "'"


def launch_script(L):
    """새 PowerShell 창이 -File 로 돌릴 스크립트. 인라인 -Command 는 Start-Process 가 안쪽
    큰따옴표를 벗겨 ParserError 가 난다(2026-09-30 실측) — 그래서 파일로 넘긴다."""
    lines = ['$env:PATH = "$env:LOCALAPPDATA\\Microsoft\\WinGet\\Links;$env:PATH"']
    lines += [f"Remove-Item Env:\\{k} -ErrorAction SilentlyContinue" for k in L["env_unset"]]
    lines += [f"$env:{k} = {_ps_quote(v)}" for k, v in L["env"].items()]
    lines.append(f"Set-Location -LiteralPath {_ps_quote(L['cwd'])}")
    lines.append(f"$Host.UI.RawUI.WindowTitle = {_ps_quote(L['name'])}")
    lines.append("& " + " ".join(_ps_quote(a) for a in L["argv"]))
    return "\r\n".join(lines) + "\r\n"


def launch_window(L):
    """보이는 새 PowerShell 창에서 L 을 실행하고, 트랙 폴더에 창 pid 를 적는다."""
    fd, path = tempfile.mkstemp(suffix=".ps1", prefix="track_open_")
    with os.fdopen(fd, "w", encoding="utf-8-sig") as f:       # PS 5.1 은 BOM 없으면 한글이 깨진다
        f.write(launch_script(L))
    p = subprocess.Popen(
        ["powershell.exe", "-NoProfile", "-NoExit", "-ExecutionPolicy", "Bypass", "-File", path],
        creationflags=subprocess.CREATE_NEW_CONSOLE,
    )
    Path(L["cwd"], SESSION_FILE).write_text(
        json.dumps({"pid": p.pid, "name": L["name"]}, ensure_ascii=False), encoding="utf-8")
    return p.pid


def _image_name(pid):
    out = subprocess.run(["tasklist", "/FI", f"PID eq {pid}", "/FO", "CSV", "/NH"],
                         capture_output=True, text=True).stdout.strip()
    return out.split(",")[0].strip('"').lower() if out.startswith('"') else ""


def session_alive(cwd):
    """트랙 폴더에 기록된 창이 아직 떠 있으면 그 기록, 아니면 None(낡은 기록 포함)."""
    try:
        rec = json.loads(Path(cwd, SESSION_FILE).read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return None
    return rec if _image_name(rec.get("pid", 0)) == "powershell.exe" else None
```

```python
# tools/dispatch.py
"""관제 → 트랙 분배 (카드 046). 관제창(텔레그램)이 부르는 한 줄.

    py tools/dispatch.py <트랙> "<지시>" [--account 2] [--agent codex] [--model fable] [--card N]

exit 0 = 새 창 열림 · 3 = 그 트랙 창이 이미 작업 중(두 번째 창 금지 — 같은 대화를 두 창이 잡으면 꼬인다)
· 1 = 오류. 새 창은 그 트랙의 지난 대화를 이어받고(--continue), 끝나면 tg_notify 로 보고한다.
"""
import argparse
import subprocess
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))

import merge_gate
import pc_label
import track
import track_open

REPORT_RULE = ("\n\n(관제 분배 작업이다. 끝나면 결과를 한 단락으로 "
               "`py tools/tg_notify.py \"<결과>\"` 로 보고하라. 막히면 막힌 이유를 같은 방법으로 보고하라.)")


def _link_card(number, track_name):
    subprocess.run([sys.executable, str(Path(__file__).with_name("control.py")),
                    "link", str(number), track_name], check=False)


def main(argv=None):
    merge_gate.make_output_safe()
    ap = argparse.ArgumentParser(description="관제 → 트랙 분배")
    ap.add_argument("track")
    ap.add_argument("order")
    ap.add_argument("--account", default="1")
    ap.add_argument("--agent", default="claude", choices=["claude", "codex"])
    ap.add_argument("--model", default="")
    ap.add_argument("--card", type=int)
    a = ap.parse_args(argv)

    wt = track.worktree_path(a.track)
    if not wt.exists():
        print(f"없는 트랙: {a.track} — 만들려면 카드 등록 뒤 py tools/track.py start {a.track} --card <번호>")
        return 1
    busy = track_open.session_alive(wt)
    if busy:
        print(f"작업 중: {busy['name']} — 앱에서 이어 지시하거나 끝난 뒤 다시 분배")
        return 3
    if a.card:
        _link_card(a.card, a.track)
    L = track_open.build_launch(wt, account=a.account, agent=a.agent, model=a.model,
                                prompt=a.order + REPORT_RULE, pc=pc_label.label(),
                                tag=a.track, card=a.card)
    pid = track_open.launch_window(L)
    print(f"열림: {L['name']} (pid {pid})")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""트랙 창 → 텔레그램 보고 (카드 046).  py tools/tg_notify.py "<결과>"

보내기(sendMessage)만 한다 — 받기(getUpdates)는 관제창 하나만 해야 하므로(409) 절대 부르지 않는다.

from pathlib import Path

import pc_label

STATE = Path.home() / ".claude" / "channels" / "telegram"


def _token(state_dir):
    try:
        for line in (state_dir / ".env").read_text(encoding="utf-8").splitlines():
            if line.startswith("TELEGRAM_BOT_TOKEN="):
                return line.split("=", 1)[1].strip()
    except OSError:
        pass
    return ""


def _post(url, data):
    body = urllib.parse.urlencode(data).encode()
    with urllib.request.urlopen(url, body, timeout=15) as r:
        return r.status == 200


def main(argv=None, state_dir=STATE, pc=None, cwd_name=None):
    text = " ".join(argv if argv is not None else sys.argv[1:]).strip()
    token = _token(state_dir)
    try:
        chat = str(json.loads((state_dir / "access.json").read_text(encoding="utf-8"))["allowFrom"][0])
    except (OSError, ValueError, KeyError, IndexError):
        chat = ""
    if not (token and chat and text):
        print("보고 실패: 토큰·받는 사람·내용 중 하나가 없다 (이 화면에 결과를 남겨라)")
        return 1
    head = f"[{pc or pc_label.label()}·{cwd_name or Path.cwd().name}] "
    try:
        ok = _post(f"https://api.telegram.org/bot{token}/sendMessage", {"chat_id": chat, "text": head + text})
    except OSError as e:
        print(f"보고 실패: {type(e).__name__}")
        return 1
    print("보고 완료" if ok else "보고 실패")
    return 0 if ok else 1


if __name__ == "__main__":
    sys.exit(main())
```
