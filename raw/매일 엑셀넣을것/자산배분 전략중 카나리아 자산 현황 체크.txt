import time
import yfinance as yf
import pandas as pd
from datetime import datetime

# 단순 모멘텀 계산 함수 (일간 기준 1M,3M,6M,12M 평균)
def calculate_simple_momentum(asset_data):
    momentum = pd.DataFrame(index=asset_data.index)
    for column in asset_data.columns:
        m1 = asset_data[column] / asset_data[column].shift(21)  - 1
        m3 = asset_data[column] / asset_data[column].shift(63)  - 1
        m6 = asset_data[column] / asset_data[column].shift(126) - 1
        m12= asset_data[column] / asset_data[column].shift(252) - 1
        momentum[column] = (m1+m3+m6+m12)/4
    # 데이터가 너무 짧아 비면 None 리턴
    if len(momentum.dropna()) == 0:
        return None
    return momentum.iloc[-1]

def safe_download(ticker, start_date, end_date, retries=3, sleep_sec=2):
    last_err = None
    for i in range(retries):
        try:
            df = yf.download(ticker, start=start_date, end=end_date, auto_adjust=False, threads=False, progress=False)
            close = df['Close'].dropna()
            if isinstance(close, pd.Series):
                close = close.to_frame(name=ticker)
            else:
                close = close[[ticker]].dropna()
            if len(close) > 300:  # 대충 252일+ 확보 체크
                return close
        except Exception as e:
            last_err = e
        time.sleep(sleep_sec * (i+1))  # 2s,4s,6s 백오프
    raise RuntimeError(f"{ticker} 다운로드 실패: {last_err}")

# 기간
end_date = datetime.today().strftime('%Y-%m-%d')
start_date = (datetime.today() - pd.DateOffset(years=3)).strftime('%Y-%m-%d')

print("\n전략: QQQ&TIP\n")

# ✅ QQQ / TIP 분리 다운로드
qqq = safe_download("QQQ", start_date, end_date)
tip = safe_download("TIP", start_date, end_date)

data = qqq.join(tip, how="inner").dropna()

last_date = data.index[-1]
print(f"마지막 데이터 날짜: {last_date.strftime('%Y-%m-%d')}")

latest_momentum = calculate_simple_momentum(data)
if latest_momentum is None:
    raise RuntimeError("모멘텀 계산에 필요한 기간(최소 252거래일)이 부족합니다.")

print("모멘텀 값:")
print(latest_momentum)

qqq_mom = latest_momentum["QQQ"]
tip_mom = latest_momentum["TIP"]

if qqq_mom > 0 and tip_mom > 0:
    print("\n🚀 카나리아 자산 신호: 공격 모드 유지 가능")
    print("   → QQQ 및 TIP 모두 모멘텀이 양수입니다.")
    print("   → 위험 자산 100% 투자")
else:
    print("\n📉 카나리아 자산 신호: 방어 모드 활성화 필요")
    if qqq_mom <= 0:
        print(f"   - QQQ 모멘텀이 음수입니다: {qqq_mom:.2%}")
    if tip_mom <= 0:
        print(f"   - TIP 모멘텀이 음수입니다: {tip_mom:.2%}")
    print("   → 안전 자산 100% 투자")
