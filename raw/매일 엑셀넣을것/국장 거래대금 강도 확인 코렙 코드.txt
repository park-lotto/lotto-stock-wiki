import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from google.colab import files

import matplotlib
matplotlib.rcParams["font.family"] = "DejaVu Sans"
matplotlib.rcParams["axes.unicode_minus"] = False

# ----------------------------
# 0) Params
# ----------------------------
PLOT_DAYS_D   = 126      # Daily plot window (6M)
PLOT_WEEKS_W  = 52       # Weekly plot window (1Y)
BUFFER_DAYS   = 120      # indicator warmup buffer

VOL_MA        = 20
ZSCORE_WIN    = 60
ACC_WIN       = 7

W_VOL_SPIKE   = 0.50
W_TURNOVER    = 0.30
W_GAP_TREND   = 0.20

# ----------------------------
# 1) Helper functions
# ----------------------------
def rolling_zscore(series, window=60, min_periods=20):
    mean = series.rolling(window, min_periods=min_periods).mean()
    std  = series.rolling(window, min_periods=min_periods).std(ddof=0)
    return (series - mean) / (std.replace(0, np.nan))

def minmax_0_100(series: pd.Series):
    s = series.dropna()
    if s.empty:
        return series * np.nan
    vmin, vmax = s.min(), s.max()
    if vmin == vmax:
        return series * np.nan
    return (series - vmin) * 100.0 / (vmax - vmin)

def plot_price_and_osc(close_s, osc_s, ticker, title_suffix):
    """Price + raw osc + SMA3/EMA5/Signal EMA7 on shared axes"""
    plot_close = close_s.copy()
    plot_osc   = osc_s.copy()

    sma3 = plot_osc.rolling(3, min_periods=1).mean()
    ema5 = plot_osc.ewm(span=5, adjust=False).mean()
    sig7 = plot_osc.ewm(span=7, adjust=False).mean()

    fig, ax1 = plt.subplots(figsize=(12, 5))
    ax1.plot(plot_close.index, plot_close.values, lw=1.6, color="black", label="Price")
    ax1.set_xlabel("Date")
    ax1.set_ylabel("Price", color="black")
    ax1.tick_params(axis="y", labelcolor="black")

    ax2 = ax1.twinx()
    ax2.plot(plot_osc.index, plot_osc.values, ls="--", lw=1.3, color="black", label="7D Osc (raw)")
    ax2.plot(sma3.index,    sma3.values,      ls="-.", lw=1.5, label="SMA(3)")
    ax2.plot(ema5.index,    ema5.values,      ls="-",  lw=1.6, label="EMA(5)")
    ax2.plot(sig7.index,    sig7.values,      ls=":",  lw=1.6, label="Signal EMA(7)")
    ax2.axhline(50, ls="--", lw=1, alpha=0.4, color="gray")
    ax2.set_ylabel("Oscillator (0-100)")

    l1, lb1 = ax1.get_legend_handles_labels()
    l2, lb2 = ax2.get_legend_handles_labels()
    ax1.legend(l1 + l2, lb1 + lb2, loc="upper left")

    plt.title(f"{ticker} - Price & 7D Oscillator ({title_suffix})")
    plt.tight_layout()
    plt.show()

def build_trading_intensity_from_excel(df_ohlcv: pd.DataFrame):
    """
    엑셀 기반 OHLCV + 회전율로 TI 계산.
    RotationRate는 HTS에서 흔한 '회전율(%) * 100' 가정:
      - fraction(0~1)로 변환: Turnover = RotationRate / 10000
    """
    df = df_ohlcv.copy()
    df["PrevClose"] = df["Close"].shift(1)

    # Volume spike vs 20D MA
    df["VolMA"] = df["Volume"].rolling(VOL_MA, min_periods=max(5, VOL_MA//2)).mean()
    df["VolSpike"] = df["Volume"] / df["VolMA"]

    # Turnover: 엑셀에서 회전율 가져옴 (단위 보정)
    df["Turnover"] = df["RotationRate"] / 10000.0

    # Gap + intraday trend
    df["Gap"] = (df["Open"] / df["PrevClose"] - 1.0)
    df["IntradayTrend"] = (df["Close"] / df["Open"] - 1.0)
    df["GapTrend"] = df["Gap"] + df["IntradayTrend"]

    # Z-scores
    df["z_VolSpike"] = rolling_zscore(df["VolSpike"], window=ZSCORE_WIN)
    df["z_Turnover"] = rolling_zscore(df["Turnover"], window=ZSCORE_WIN)
    df["z_GapTrend"] = rolling_zscore(df["GapTrend"], window=ZSCORE_WIN)

    # Composite daily score
    df["TI_raw"] = (
        W_VOL_SPIKE * df["z_VolSpike"].fillna(0) +
        W_TURNOVER  * df["z_Turnover"].fillna(0) +
        W_GAP_TREND * df["z_GapTrend"].fillna(0)
    )

    # 7D rolling sum -> normalize to 0-100
    df["TI_acc7"] = df["TI_raw"].rolling(ACC_WIN, min_periods=ACC_WIN).sum()
    df["TI_acc7_0_100"] = minmax_0_100(df["TI_acc7"])

    return df

def get_ticker_name(raw_df: pd.DataFrame, file_name: str, prefer_col: str = "종목명") -> str:
    """
    1) 엑셀에 '종목명' 컬럼이 있으면 첫 값 사용
    2) 없으면 파일명에서 확장자 제거한 값 사용
    """
    if prefer_col in raw_df.columns:
        v = raw_df[prefer_col].dropna()
        if not v.empty:
            return str(v.iloc[0]).strip()

    base = os.path.splitext(os.path.basename(file_name))[0]
    # 파일명이 "원익IPS_주가데이터"면 앞부분만 쓰고 싶으면 아래 줄 사용:
    # base = base.split("_")[0]
    return base

# ----------------------------
# 2) 업로드 & 로드
# ----------------------------
uploaded = files.upload()
file_name = list(uploaded.keys())[0]
print("Loaded:", file_name)

xls = pd.ExcelFile(file_name)
sheet_name = xls.sheet_names[0]
print("Sheet:", sheet_name)

raw = pd.read_excel(file_name, sheet_name=sheet_name)

# 종목명 자동 설정
ticker_name = get_ticker_name(raw, file_name)
print("TickerName:", ticker_name)

# ----------------------------
# 3) 컬럼명 매핑 (네 파일 기준)
# ----------------------------
need_map = {
    "Date": "일자",
    "Open": "시가",
    "High": "고가",
    "Low": "저가",
    "Close": "종가",
    "Volume": "거래량",
    "RotationRate": "_RotationRate_",
}

for k, v in need_map.items():
    if v not in raw.columns:
        raise ValueError(f"Missing column: {v} (need for {k}). 현재 컬럼: {list(raw.columns)}")

df = raw.rename(columns={v:k for k, v in need_map.items()}).copy()

# Date 처리
df["Date"] = pd.to_datetime(df["Date"])
for c in ["Open","High","Low","Close","Volume","RotationRate"]:
    df[c] = pd.to_numeric(df[c], errors="coerce")

df = df.dropna(subset=["Date","Open","High","Low","Close","Volume","RotationRate"])
df = df.sort_values("Date").set_index("Date")

print("\n[Data range]", df.index.min().date(), "~", df.index.max().date())
print("[RotationRate median(raw)]", float(df["RotationRate"].median()))

# ----------------------------
# 4) TI 계산
# ----------------------------
feat_full = build_trading_intensity_from_excel(df[["Open","High","Low","Close","Volume","RotationRate"]])

# ----------------------------
# 5) DAILY view (최근 6M)
# ----------------------------
daily_view = feat_full.tail(PLOT_DAYS_D + BUFFER_DAYS).copy()
daily_close = daily_view["Close"].tail(PLOT_DAYS_D)
daily_osc   = daily_view["TI_acc7_0_100"].tail(PLOT_DAYS_D)

plot_price_and_osc(daily_close, daily_osc, ticker_name, "Daily (6M)")

# ----------------------------
# 6) WEEKLY view (최근 1Y) - W-FRI last
# ----------------------------
wk = pd.DataFrame({
    "Close": feat_full["Close"],
    "Osc":   feat_full["TI_acc7_0_100"]
}).resample("W-FRI").last().dropna()

weekly_close = wk["Close"].tail(PLOT_WEEKS_W)
weekly_osc   = wk["Osc"].tail(PLOT_WEEKS_W)

plot_price_and_osc(weekly_close, weekly_osc, ticker_name, "Weekly (1Y)")

print("\nDone.")
