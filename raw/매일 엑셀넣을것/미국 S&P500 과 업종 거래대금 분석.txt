def parse_ticker_lines(raw: str):
    tickers = []
    for line in raw.splitlines():
        t = line.strip().upper()
        if not t:
            continue
        t = t.replace(".", "-")
        if t not in tickers:
            tickers.append(t)
    return tickers

# [1] 한경 → S&P500 티커 추출/검증

HK_URL = "https://www.hankyung.com/globalmarket/usa-stock-sp500"
HEADERS = {"User-Agent": "Mozilla/5.0 (compatible; ColabScraper/1.0)"}

def clean_to_yahoo_symbol(sym: str) -> str:
    s = sym.strip().upper().replace(".", "-")
    if sym.upper() == "BRK.B": return "BRK-B"
    if sym.upper() == "BF.B" : return "BF-B"
    return s

def has_history(ticker: str, period="5d") -> bool:
    try:
        df = yf.download(ticker, period=period, interval="1d", progress=False)
        if df is None or df.empty:
            return False
        ser = df.get("Adj Close", df.get("Close", pd.Series(dtype=float))).dropna()
        return not ser.empty
    except Exception:
        return False

def extract_sp500_from_hankyung_valid() -> list:
    html = requests.get(HK_URL, headers=HEADERS, timeout=30).text
    soup = BeautifulSoup(html, "lxml")

    tickers = []
    pat_pair = re.compile(r"\(([A-Z]{1,6}(?:-[A-Z]{1,2})?)\)")
    has_hangul = re.compile(r"[가-힣]")

    # 1) '이름(티커)' 패턴
    for node in soup.find_all(string=True):
        txt = str(node)
        if not has_hangul.search(txt):
            continue
        for m in pat_pair.finditer(txt):
            tickers.append(m.group(1))

    # 2) 보조: 한글 포함 링크 텍스트에서 대문자 티커 토큰
    ticker_pat = re.compile(r"\b[A-Z]{1,5}(?:-[A-Z]{1,2})?\b")
    blacklist = {
        "USA","US","USD","S","P","SP","SNP","CPI","PPI","EPS","PE","YOY","QOQ",
        "ETF","AI","EV","GDP","FOMC","FED","ECB","BOJ","SEC","IRS","NY","EU",
        "BOOK","POLL","MAP","TESAT","AT","DR","WR","SBA","CH","JM","AO","JB","BF","RSS"
    }
    for a in soup.find_all("a"):
        text = a.get_text(" ", strip=True)
        if not text or len(text) > 300:
            continue
        if not has_hangul.search(text):
            continue
        for c in ticker_pat.findall(text):
            if c not in blacklist:
                tickers.append(c)

    # 정규화 + 중복 제거(순서 보존)
    cleaned = []
    for t in (clean_to_yahoo_symbol(x) for x in tickers):
        if 1 <= len(t) <= 8 and re.fullmatch(r"[A-Z\-]+", t):
            if t not in cleaned:
                cleaned.append(t)

    # yfinance로 실제 데이터 존재하는 심볼만
    valid = []
    for t in cleaned:
        if has_history(t, period="5d"):
            valid.append(t)

    return valid[:500]

# [2] 공통: 달러 거래대금 스파이크 계산
# ============================
def fetch_prices_in_chunks(tickers, period="15d", interval="1d", chunk_size=120, pause=0.8):
    frames = []
    for i in range(0, len(tickers), chunk_size):
        batch = tickers[i:i+chunk_size]
        df = yf.download(
            tickers=batch,
            period=period,
            interval=interval,
            group_by="ticker",
            auto_adjust=False,
            progress=False,
            threads=True,
        )
        frames.append(df)
        time.sleep(pause)
    return frames

def _pick_subframe_from_chunks(chunks, ticker):
    for df in chunks:
        if df is None or df.empty:
            continue
        if isinstance(df.columns, pd.MultiIndex):
            if ticker in df.columns.get_level_values(0):
                return df[ticker].copy()
        else:
            return df.copy()
    return None

def get_dollar_volume_spike(tickers, label, period="15d"):
    """
    종목별 달러 거래대금 = Close * Volume
    스파이크 = (가장 최근 거래일 거래대금) - (직전 5거래일 평균 거래대금)
    """
    chunks = fetch_prices_in_chunks(tickers, period=period, chunk_size=120, pause=0.8)

    rows = []
    for t in tickers:
        sub = _pick_subframe_from_chunks(chunks, t)
        if sub is None or sub.empty:
            continue
        if "Close" not in sub.columns or "Volume" not in sub.columns:
            continue

        sub = sub.dropna(subset=["Close","Volume"]).copy()
        if len(sub) < 6:
            continue

        dv = (sub["Close"] * sub["Volume"]).dropna()
        if len(dv) < 6:
            continue

        dv6 = dv.iloc[-6:]
        latest = float(dv6.iloc[-1])
        avg_prev5 = float(dv6.iloc[:-1].mean())
        spike = latest - avg_prev5
        spike_pct = (spike / avg_prev5) if avg_prev5 != 0 else np.nan

        rows.append({
            "Group": label,
            "Ticker": t,
            "Latest_DollarVolume": latest,
            "Prev5Avg_DollarVolume": avg_prev5,
            "Spike": spike,
            "SpikePct": spike_pct
        })

    out = pd.DataFrame(rows)
    if out.empty:
        return out
    out = out.sort_values("Spike", ascending=False).reset_index(drop=True)
    return out

def get_basic_info(ticker: str):
    info = {"Name": "", "Sector_or_Category": ""}
    try:
        d = yf.Ticker(ticker).get_info()
        info["Name"] = d.get("longName") or d.get("shortName") or ""
        info["Sector_or_Category"] = d.get("sector") or d.get("category") or d.get("quoteType") or ""
    except Exception:
        pass
    return info

def format_top(df_already_topn, add_info=True, sleep_each=0.35):
    out = df_already_topn.copy()
    if out.empty:
        return out

    if add_info:
        names, cats = [], []
        for t in out["Ticker"].tolist():
            meta = get_basic_info(t)
            names.append(meta.get("Name",""))
            cats.append(meta.get("Sector_or_Category",""))
            time.sleep(sleep_each)
        out.insert(2, "Name", names)
        out.insert(3, "Sector/Category", cats)

    def fmt_money(x):
        if pd.isna(x): return ""
        return f"${x:,.0f}"

    out["Latest_DollarVolume"] = out["Latest_DollarVolume"].map(fmt_money)
    out["Prev5Avg_DollarVolume"] = out["Prev5Avg_DollarVolume"].map(fmt_money)
    out["Spike"] = out["Spike"].map(fmt_money)
    out["SpikePct"] = (out["SpikePct"] * 100).round(2).astype(str) + "%"

    return out
