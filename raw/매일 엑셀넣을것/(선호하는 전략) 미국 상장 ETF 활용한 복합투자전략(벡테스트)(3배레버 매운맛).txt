import pandas as pd
import yfinance as yf
from datetime import datetime

# -------------------- 1. 데이터 다운로드 --------------------
tickers = [
    "SPXL", "TQQQ", "UWM", "EFO", "EET", "DRN",   # 공격 자산
    "PDBC", "IEF", "TMF", "TIP", "BIL",           # 안전 자산 (UST → IEF로 교체)
    "HYG", "LQD", "EMB", "BNDX", "BWX", "GLD",    # 안전 자산
    "QYLD", "QQQ"                                 # 고정 및 카나리아 자산
]

end_date = datetime.today().strftime('%Y-%m-%d')
start_date = (datetime.today() - pd.DateOffset(years=10)).strftime('%Y-%m-%d')

data = yf.download(tickers, start=start_date, end=end_date)['Close']
monthly_data = data.resample('M').last()

# -------------------- 2. 모멘텀 계산 함수 --------------------
def calculate_momentum(price_series):
    df = pd.DataFrame(index=price_series.index)
    df['1M']  = price_series.pct_change(periods=1)
    df['3M']  = price_series.pct_change(periods=3)
    df['6M']  = price_series.pct_change(periods=6)
    df['12M'] = price_series.pct_change(periods=12)
    return df.mean(axis=1)

# -------------------- 3. 카나리아 신호 (TIP & QQQ) --------------------
def canary_signal(qqq_series, tip_series):
    qqq_mom = calculate_momentum(qqq_series)
    tip_mom = calculate_momentum(tip_series)
    signal = (qqq_mom > 0) & (tip_mom > 0)  # 둘 다 양수여야 True
    return signal, qqq_mom, tip_mom

# -------------------- 4. 안전 자산 선택 --------------------
def select_safe_assets(momentum_data, bil_momentum, top_n=3):
    safe_assets = ["BIL", "TMF", "HYG", "LQD", "IEF", "EMB", "TIP", "BNDX", "BWX", "GLD"]
    safe_mom = momentum_data[safe_assets].iloc[-1]
    top_safe = safe_mom.nlargest(top_n)
    selected = [asset for asset in top_safe.index if top_safe[asset] > bil_momentum]
    return selected if selected else ['BIL']

# -------------------- 5. 공격 or 방어 자산 선택 --------------------
def select_top_assets(momentum_data, canary_signal_series, top_n=4):
    if canary_signal_series.iloc[-1]:
        return momentum_data.iloc[-1].nlargest(top_n).index.tolist()
    else:
        bil_mom = calculate_momentum(monthly_data['BIL']).iloc[-1]
        return select_safe_assets(momentum_data, bil_mom, top_n=3)

# -------------------- 6. 백테스트 실행 --------------------
fixed_assets = {"QYLD": 0.1, "GLD": 0.05, "PDBC": 0.05}
fixed_weight_sum = sum(fixed_assets.values())

dates = monthly_data.index
portfolio_data = []
last_recorded_date = None

for i in range(12, len(dates)):
    decision_date = dates[i]

    if i < len(dates) - 1:
        invest_date = dates[i + 1]
        current_data = monthly_data.loc[:decision_date]
        current_momentum = current_data.apply(calculate_momentum)

        # ✅ QQQ와 TIP 각각 모멘텀 계산
        canary_sig, qqq_mom, tip_mom = canary_signal(current_data['QQQ'], current_data['TIP'])

        # 자산 선택
        top_assets = select_top_assets(current_momentum, canary_sig, top_n=4)

        # 비중 계산
        remaining_weight = 1 - fixed_weight_sum
        haa_weights = {asset: remaining_weight / len(top_assets) for asset in top_assets}

        combined_weights = fixed_assets.copy()
        for asset, weight in haa_weights.items():
            combined_weights[asset] = combined_weights.get(asset, 0) + weight

        # 비중 정규화
        total_weight = sum(combined_weights.values())
        for k in combined_weights:
            combined_weights[k] /= total_weight

        # 수익 계산
        monthly_return = sum(
            combined_weights[asset] * monthly_data[asset].pct_change().loc[invest_date]
            for asset in combined_weights.keys() if asset in monthly_data.columns
        )

        record_date_str = invest_date.strftime('%Y-%m-%d')
        if record_date_str != last_recorded_date:
            portfolio_data.append({
                "Date": record_date_str,
                "Assets": ', '.join(combined_weights.keys()),
                "Weights": ', '.join([f"{k}: {v:.2%}" for k, v in combined_weights.items()]),
                "Monthly Return": round(monthly_return * 100, 2),
                "QQQ Mom": round(qqq_mom.loc[decision_date] * 100, 2),
                "TIP Mom": round(tip_mom.loc[decision_date] * 100, 2),
            })
            last_recorded_date = record_date_str

# -------------------- 7. 백테스트 결과 정리 --------------------
portfolio = pd.DataFrame(portfolio_data)
portfolio['Cumulative'] = (1 + portfolio['Monthly Return'].fillna(0) / 100).cumprod()

years = (datetime.strptime(portfolio['Date'].iloc[-1], '%Y-%m-%d') -
         datetime.strptime(portfolio['Date'].iloc[0], '%Y-%m-%d')).days / 365.25
cagr = (portfolio['Cumulative'].iloc[-1]) ** (1 / years) - 1

def calculate_mdd(cum_returns):
    return (cum_returns / cum_returns.cummax() - 1).min()

mdd = calculate_mdd(portfolio['Cumulative'])

# -------------------- 8. 결과 출력 --------------------
print(portfolio[['Date', 'Weights']].to_string(index=False))  # 요약 출력
print("\n백테스트 결과:")
print(f"CAGR (연평균 수익률): {cagr * 100:.2f}%")
print(f"MDD (최대 낙폭): {mdd * 100:.2f}%")

# 저장
portfolio.to_csv('portfolio.csv', index=False, encoding='utf-8-sig')
