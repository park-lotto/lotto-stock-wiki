import pandas as pd
import yfinance as yf
from datetime import datetime

# ---------------------------
# 1) 티커/자산 설정
# ---------------------------
risky_assets = {
    "133690.KS": "TIGER 미국나스닥100",
    "143850.KS": "TIGER 미국S&P500선물(H)",
    "280930.KS": "KODEX 미국러셀2000(H)",
    "195980.KS": "ARIRANG 신흥국MSCI(합성 H)",
    "251350.KS": "KODEX 선진국MSCI World",
    "261220.KS": "KODEX WTI원유선물(H)",
    "182480.KS": "TIGER 미국MSCI리츠(합성 H)",
    "102110.KS": "TIGER 200"
}

safe_assets = {
    "304660.KS": "KODEX 미국채울트라30년선물(H)",
    "305080.KS": "TIGER 미국채10년선물",
    "148070.KS": "KOSEF 국고채10년",
    "261240.KS": "KODEX 미국달러선물",
    "130730.KS": "KOSEF 단기자금"
}

canary_asset = {"TIP": "TIP 물가연동채"}

# 모든 티커 합치기
tickers = list(risky_assets.keys()) + list(safe_assets.keys()) + list(canary_asset.keys())

# ---------------------------
# 2) 사용자로부터 특정 투자일 입력받기
# ---------------------------
specific_date = input("투자일(YYYY-MM-DD)을 입력하세요: ")
# 예: 2023-11-10

# 날짜 파싱 (유효성 검사는 생략)
target_dt = datetime.strptime(specific_date, "%Y-%m-%d")

# ---------------------------
# 3) 데이터 다운로드 (과거 3년치) + 특정 날짜까지 필터링
# ---------------------------
end_date   = target_dt.strftime('%Y-%m-%d')
start_date = (target_dt - pd.DateOffset(years=3)).strftime('%Y-%m-%d')

data = yf.download(tickers, start=start_date, end=end_date)['Close']

# daily_data: 특정일(specific_date)까지의 일별 종가
daily_data = data.loc[:specific_date]

# 만약 daily_data가 너무 짧다면 에러처리를 할 수도 있음(생략)

# ---------------------------
# 4) 월말 데이터 생성
# ---------------------------
# "월말" 기준으로 last() => 마지막 영업일 종가
monthly_data = daily_data.resample('M').last()

# ---------------------------
# 5) 모멘텀 계산 함수 (1/3/6/12개월 평균)
# ---------------------------
def calculate_momentum(series):
    momentum_df = pd.DataFrame(index=series.index)
    momentum_df['1M']  = series.pct_change(periods=1)
    momentum_df['3M']  = series.pct_change(periods=3)
    momentum_df['6M']  = series.pct_change(periods=6)
    momentum_df['12M'] = series.pct_change(periods=12)
    return momentum_df.mean(axis=1)

# 전체 종목의 (월말 기반) 모멘텀
momentum_data = monthly_data.apply(calculate_momentum)

# ---------------------------
# 6) 카나리아 신호 (TIP 모멘텀) 계산
# ---------------------------
def canary_momentum(tip_series):
    tip_mom = calculate_momentum(tip_series)
    return (tip_mom > 0), tip_mom

canary_signal, tip_momentum = canary_momentum(monthly_data['TIP'])

# ---------------------------
# 7) 안전/공격 자산 선택 함수
# ---------------------------
def select_safe_assets(momentum_df, bil_momentum, top_n=2):
    safe_mom = momentum_df[safe_assets.keys()].iloc[-1]
    top_safe_assets = safe_mom.nlargest(top_n)
    selected_assets = []
    for asset in top_safe_assets.index:
        if top_safe_assets[asset] > bil_momentum:
            selected_assets.append(asset)
    # 최소 1개는 '130730.KS'
    if not selected_assets:
        return ['130730.KS']
    if '130730.KS' not in selected_assets:
        selected_assets.append('130730.KS')
    return selected_assets[:top_n]

def select_top_assets(momentum_df, canary_signal_series, top_n=4):
    if canary_signal_series.iloc[-1]:  # TIP 모멘텀이 양수면 공격
        top_assets = momentum_df.iloc[-1].nlargest(top_n)
        return top_assets.index.tolist()
    else:
        # bil_momentum(단기자금)
        bil_momentum = calculate_momentum(daily_data['130730.KS']).iloc[-1]
        return select_safe_assets(momentum_df, bil_momentum, top_n=2)

# ---------------------------
# 8) 현재(특정 날짜) 기준으로 모멘텀 & 카나리아 신호
# ---------------------------
# 모멘텀/카나리아는 monthly_data 전체로 계산했지만,
# 마지막 행(=가장 최근 월말)까지의 값만 유효
if len(monthly_data) < 13:
    print("경고: 12개월 이상 데이터가 충분하지 않아, 12M 모멘텀이 NaN일 수 있습니다.")

# 카나리아 신호 (True/False)
latest_canary_signal = canary_signal.iloc[-1] if len(canary_signal) > 0 else False

# 자산 선택 (공격 or 안전)
top_assets = select_top_assets(momentum_data, canary_signal, top_n=4)

# ---------------------------
# 9) 균등 비중 계산 및 결과 출력
# ---------------------------
if len(top_assets) == 0:
    print("\n선택된 자산이 없습니다(데이터 부족 혹은 오류).")
else:
    weight = 1.0 / len(top_assets)

    print(f"\n[투자일: {specific_date} (월중 가정)]")
    print(f"카나리아 신호: {latest_canary_signal} (TIP 모멘텀이 0 이상이면 True)")
    print("\n■ 선정된 자산 리스트:")
    for asset in top_assets:
        # 자산명(한글) 가져오기
        if asset in risky_assets:
            asset_name = risky_assets[asset]
        elif asset in safe_assets:
            asset_name = safe_assets[asset]
        elif asset in canary_asset:
            asset_name = canary_asset[asset]
        else:
            asset_name = asset  # 미등록 시 티커 그대로

        print(f"  - {asset}: {asset_name}")

    print("\n■ 각 자산 비중 (균등):")
    for asset in top_assets:
        print(f"  {asset} : {weight:.2%}")
