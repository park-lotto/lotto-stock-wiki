# 4. Mansfield RS 계산 함수
# ----------------------------
def compute_mansfield_rs(etf, benchmark, ma_period=52):
    """
    etf와 benchmark(코스피)의 비율(relative)을 ma_period 이동평균으로 나눈 편차(%)를 Mansfield RS로 계산
    Note: 일간 데이터 기준 ma_period=52면 52일 이동평균
    """
    relative = etf / benchmark
    ma = relative.rolling(window=ma_period, min_periods=ma_period).mean()
    rs = ((relative / ma) - 1) * 100
    return rs


def normalize_to_100_scale(x, scale=12):
    """
    Mansfield RS(%)를 0~100 스케일로 변환 (시그모이드)
    scale이 작을수록 민감도 증가
    """
    return 100 * (1 / (1 + np.exp(-x / scale)))

# ----------------------------
# 5. RS 계산 실행 (60d / 120d / 250d + 평균)
# ----------------------------
RS_WINDOWS = [60, 120, 250]

mansfield_rows = []

for column in etf_data.columns:
    series = etf_data[column].dropna()
    common_idx = series.index.intersection(kospi.index)
    etf_series = series.loc[common_idx]
    kospi_series = kospi.loc[common_idx]

    if len(etf_series) < min(RS_WINDOWS):
        continue

    latest = {}

    for win in RS_WINDOWS:
        if len(etf_series) >= win:
            rs_series = compute_mansfield_rs(etf_series, kospi_series, ma_period=win).dropna()
            if not rs_series.empty:
                raw = float(rs_series.iloc[-1])
                latest[f"RS_{win}d"] = round(raw, 2)
                latest[f"Norm_RS_{win}d"] = round(normalize_to_100_scale(raw), 2)

    raw_keys = [k for k in latest.keys() if k.startswith("RS_") and k.endswith("d")]
    if len(raw_keys) == 0:
        continue

    raw_values = [latest[k] for k in raw_keys]
    rs_avg = round(sum(raw_values) / len(raw_values), 2)
    norm_rs_avg = round(normalize_to_100_scale(rs_avg), 2)

    row = {
        "ETF": column,
        "RS_60d": np.nan,
        "RS_120d": np.nan,
        "RS_250d": np.nan,
        "RS_avg": rs_avg,
        "Norm_RS_60d": np.nan,
        "Norm_RS_120d": np.nan,
        "Norm_RS_250d": np.nan,
        "Norm_RS_avg": norm_rs_avg,
    }

    for win in RS_WINDOWS:
        if f"RS_{win}d" in latest:
            row[f"RS_{win}d"] = latest[f"RS_{win}d"]
            row[f"Norm_RS_{win}d"] = latest[f"Norm_RS_{win}d"]

    mansfield_rows.append(row)

mansfield_df = pd.DataFrame(mansfield_rows).sort_values(by="Norm_RS_avg", ascending=False)

# 6. 변동성 조정 모멘텀 계산 함수
# ----------------------------
def calculate_risk_adjusted_momentum(data, last_n_days=10, ma_filter_days=None, top_n=20):
    """
    3/6/12개월 수익률을 각 기간 표준편차로 나눠 평균한 위험조정 모멘텀 점수.
    ma_filter_days가 주어지면 해당 이동평균 하회 종목 제외.
    """
    daily_top = {}
    trading_days = data.dropna(how='all').index[-last_n_days:]

    for day in trading_days:
        if day not in data.index:
            continue

        end_idx = data.index.get_indexer_for([day])[0]
        start_3m = max(0, end_idx - 62)
        start_6m = max(0, end_idx - 125)
        start_12m = max(0, end_idx - 251)

        window_3m = data.iloc[start_3m:end_idx + 1]
        window_6m = data.iloc[start_6m:end_idx + 1]
        window_12m = data.iloc[start_12m:end_idx + 1]

        returns_3m = window_3m.pct_change(fill_method=None).mean(skipna=True)
        returns_6m = window_6m.pct_change(fill_method=None).mean(skipna=True)
        returns_12m = window_12m.pct_change(fill_method=None).mean(skipna=True)

        vol_3m = window_3m.pct_change(fill_method=None).std(skipna=True)
        vol_6m = window_6m.pct_change(fill_method=None).std(skipna=True)
        vol_12m = window_12m.pct_change(fill_method=None).std(skipna=True)

        valid = (vol_3m > 0) & (vol_6m > 0) & (vol_12m > 0)

        if ma_filter_days is not None:
            ma_ok = []
            for col in data.columns:
                series = data[col].iloc[:end_idx + 1].dropna()
                if len(series) < ma_filter_days:
                    ma_ok.append(False)
                else:
                    ma_value = series.rolling(window=ma_filter_days).mean().iloc[-1]
                    price_value = series.iloc[-1]
                    ma_ok.append(price_value >= ma_value)
            ma_ok = pd.Series(ma_ok, index=data.columns)
            valid &= ma_ok

        risk_adj = (
            returns_3m[valid] / vol_3m[valid]
            + returns_6m[valid] / vol_6m[valid]
            + returns_12m[valid] / vol_12m[valid]
        ) / 3

        top_list = risk_adj.nlargest(top_n).index.tolist()
        while len(top_list) < top_n:
            top_list.append(None)

        daily_top[day] = top_list

    df_top = pd.DataFrame.from_dict(
        daily_top,
        orient="index",
        columns=[f"Top {i+1}" for i in range(top_n)]
    )
    df_top.index.name = "Date"
    return df_top

# 7. Sortino / ETR Comfort 계산 함수
# ============================================================
def _mdd_from_returns(ret: pd.Series) -> float:
    cum = (1 + ret.fillna(0)).cumprod()
    peak = cum.cummax()
    dd = (cum - peak) / peak
    return float(dd.min())


def _sortino_from_returns(ret: pd.Series, mar: float = 0.0, periods_per_year: int = 252) -> float:
    ret = ret.dropna()
    if len(ret) < 5:
        return np.nan

    mar_daily = (1 + mar) ** (1 / periods_per_year) - 1
    excess = ret - mar_daily
    downside = excess[excess < 0]

    if len(downside) < 5:
        return np.nan

    downside_dev = downside.std(ddof=1) * np.sqrt(periods_per_year)
    if downside_dev == 0 or np.isnan(downside_dev):
        return np.nan

    annual_excess = excess.mean() * periods_per_year
    return float(annual_excess / downside_dev)


def _cagr_from_prices_window(prices: pd.Series, periods_per_year: int = 252) -> float:
    prices = prices.dropna()
    if len(prices) < 2:
        return np.nan

    n = len(prices)
    return float((prices.iloc[-1] / prices.iloc[0]) ** (periods_per_year / (n - 1)) - 1)


def _etr_comfort_from_prices_window(prices: pd.Series, periods_per_year: int = 252) -> float:
    prices = prices.dropna()
    if len(prices) < 60:
        return np.nan

    ret = prices.pct_change(fill_method=None).dropna()
    if len(ret) < 5:
        return np.nan

    cagr = _cagr_from_prices_window(prices, periods_per_year=periods_per_year)
    vol = float(ret.std(ddof=1) * np.sqrt(periods_per_year))
    mdd = _mdd_from_returns(ret)
    denom = abs(mdd) * vol

    if denom == 0 or np.isnan(denom) or np.isnan(cagr):
        return np.nan

    return float(cagr / denom)


def _ma_ok_mask(close_df: pd.DataFrame, end_idx: int, ma_days: int) -> pd.Series:
    ok = {}
    for col in close_df.columns:
        s = close_df[col].iloc[:end_idx + 1].dropna()
        if len(s) < ma_days:
            ok[col] = False
        else:
            ma = s.rolling(ma_days).mean().iloc[-1]
            ok[col] = bool(s.iloc[-1] >= ma)
    return pd.Series(ok)


def rank_sortino_or_etr_like_momentum(
    data: pd.DataFrame,
    metric: str = "sortino",
    last_n_days: int = 10,
    ma_filter_days: int = 10,
    top_n: int = 20,
    mar: float = 0.0,
    periods_per_year: int = 252,
    period_3m: int = 63,
    period_6m: int = 126,
    period_12m: int = 252,
) -> pd.DataFrame:

    daily_top = {}
    trading_days = data.dropna(how="all").index[-last_n_days:]
    wins = [period_3m, period_6m, period_12m]

    for day in trading_days:
        end_idx = data.index.get_indexer_for([day])[0]
        ok = _ma_ok_mask(data, end_idx=end_idx, ma_days=ma_filter_days)

        scores = {}
        for col in data.columns:
            if not ok.get(col, False):
                continue

            vals = []
            for win in wins:
                start_idx = max(0, end_idx - (win - 1))
                w_prices = data[col].iloc[start_idx:end_idx + 1].dropna()

                if len(w_prices) < max(20, int(win * 0.8)):
                    continue

                if metric == "sortino":
                    r = w_prices.pct_change(fill_method=None).dropna()
                    v = _sortino_from_returns(r, mar=mar, periods_per_year=periods_per_year)
                elif metric == "etr":
                    v = _etr_comfort_from_prices_window(w_prices, periods_per_year=periods_per_year)
                else:
                    raise ValueError("metric은 'sortino' 또는 'etr'만 지원합니다.")

                if pd.notna(v):
                    vals.append(float(v))

            if len(vals) == 0:
                continue

            scores[col] = float(np.mean(vals))

        if len(scores) == 0:
            daily_top[day] = [None] * top_n
            continue

        ranked = pd.Series(scores).sort_values(ascending=False).head(top_n).index.tolist()
        while len(ranked) < top_n:
            ranked.append(None)

        daily_top[day] = ranked

    out = pd.DataFrame.from_dict(
        daily_top,
        orient="index",
        columns=[f"Top {i+1}" for i in range(top_n)]
    )
    out.index.name = "Date"
    return out

def compute_crowding_index(
    price_df: pd.DataFrame,
    lookback_days: int = 126,
    min_universe: int = 10,
    score_mode: str = "linear",
    scale: float = 100.0
):
    ret6m = price_df.pct_change(lookback_days, fill_method=None)

    crowding_vals = []
    valid_dates = ret6m.index

    for dt in valid_dates:
        r = ret6m.loc[dt].dropna()
        n = len(r)
        if n < min_universe:
            crowding_vals.append(np.nan)
            continue

        rank = r.rank(ascending=False, method="average")

        if score_mode == "linear":
            score = (n - rank) / (n - 1)
        elif score_mode == "centered":
            score = ((n + 1) / 2 - rank) / ((n - 1) / 2)
        else:
            raise ValueError("score_mode는 'linear' 또는 'centered'만 지원합니다.")

        weighted = score * r
        crowd = float(weighted.std(ddof=1) * scale)
        crowding_vals.append(crowd)

    crowding_series = pd.Series(crowding_vals, index=valid_dates, name="CrowdingIndex").dropna()
    return crowding_series, ret6m

#     구성:
#     1) 먼저 전체 요약표 1개 출력
#     2) 그 다음 섹터별 가로형 그래프 출력
#     3) 그래프에는 한글을 최소화하여 폰트 깨짐 방지
# ============================================================

LOOKBACK_DAYS = 247    # 최근 247거래일 기준
GAP_MA_DAYS = 30      # 30일 이격도
GRAPH_YLIM = (90, 160)
BAR_ALPHA = 0.15


def make_sortino_10w_gap30_summary(price_df: pd.DataFrame, sortino_top_df: pd.DataFrame) -> pd.DataFrame:
    """
    Sortino 10주선(50일선) 이상 섹터만 대상으로
    30일 이격도와 과거 247거래일 97% 초과 이격도 수준을 계산합니다.

    표시용 컬럼은 % 기호를 붙여 눈으로 보기 쉽게 만들고,
    그래프 계산용 숫자 컬럼은 앞에 '_'를 붙여 내부용으로 보관합니다.
    """
    latest_sortino_list = sortino_top_df.iloc[-1].dropna().tolist()

    summary_rows = []

    for rank_no, sector in enumerate(latest_sortino_list, start=1):
        if sector not in price_df.columns:
            continue

        price = price_df[sector].dropna()
        if len(price) < GAP_MA_DAYS + 5:
            continue

        ma30 = price.rolling(GAP_MA_DAYS).mean()
        gap30 = price / ma30 * 100
        gap30 = gap30.dropna()

        if gap30.empty:
            continue

        recent_gap = gap30.tail(LOOKBACK_DAYS)

        current_price = float(price.loc[gap30.index[-1]])
        current_ma30 = float(ma30.loc[gap30.index[-1]])
        current_gap = float(gap30.iloc[-1])
        current_gap_excess = current_gap - 100

        gap97 = float(recent_gap.quantile(0.97))
        gap97_excess = gap97 - 100

        # 현재 위치: 30일선 초과분 기준으로 비교
        # 예: 현재 초과분 12, 247거래일 97% 초과분 10이면 120%
        excess_heat_ratio = (
            current_gap_excess / gap97_excess * 100
            if gap97_excess != 0
            else np.nan
        )

        기준일 = gap30.index[-1].strftime("%Y-%m-%d")

        summary_rows.append({
            "Sortino순위": rank_no,
            "섹터": sector,
            "종가": f"{current_price:,.0f}",
            "MA30": f"{current_ma30:,.0f}",
            "현재30일선초과이격도": round(current_gap_excess, 2),
            "247거래일97%초과이격도": round(gap97_excess, 2),
            "97%구간대비현재위치": f"{excess_heat_ratio:.1f}%",
            "기준일": 기준일,
            # 그래프 제목용 내부 숫자 컬럼
            "_현재30일선초과이격도": round(current_gap_excess, 2),
            "_247거래일97%초과이격도": round(gap97_excess, 2),
            "_97%구간대비현재위치": round(excess_heat_ratio, 1),
        })

    summary_df = pd.DataFrame(summary_rows)
    return summary_df


def plot_gap30_charts_after_summary(price_df: pd.DataFrame, summary_df: pd.DataFrame):
    """
    표는 한글로 출력하고, 그래프 내부 텍스트는 영문/숫자 중심으로 표시합니다.
    그래서 NanumGothic 폰트가 없어도 경고가 나오지 않습니다.
    """
    for _, row in summary_df.iterrows():
        sector = row["섹터"]
        rank_no = int(row["Sortino순위"])

        if sector not in price_df.columns:
            continue

        price = price_df[sector].dropna()
        ma30 = price.rolling(GAP_MA_DAYS).mean()
        gap30 = price / ma30 * 100

        plot_df = pd.DataFrame({
            "Price": price,
            "MA30": ma30,
            "Gap30": gap30,
        }).dropna()

        if plot_df.empty:
            continue

        fig, ax1 = plt.subplots(figsize=(16, 4))
        ax2 = ax1.twinx()

        # 이격도 막대: 뒤쪽에 연하게 표시
        ax2.bar(
            plot_df.index,
            plot_df["Gap30"],
            color="#FFC000",
            alpha=BAR_ALPHA,
            width=1.5,
            label="Gap30",
            zorder=1,
        )
        ax2.set_ylim(GRAPH_YLIM)

        # 왼쪽 축을 앞으로 가져오기
        ax1.set_zorder(ax2.get_zorder() + 1)
        ax1.patch.set_visible(False)

        # 가격과 MA30
        ax1.plot(
            plot_df.index,
            plot_df["Price"],
            color="black",
            linewidth=2.8,
            label="Price",
            zorder=10,
        )

        ax1.plot(
            plot_df.index,
            plot_df["MA30"],
            color="#8FA3BD",
            linewidth=3.0,
            label="MA30",
            zorder=9,
        )

        ax1.yaxis.set_major_formatter(mticker.StrMethodFormatter("{x:,.0f}"))
        ax1.grid(True, axis="both", alpha=0.25, zorder=0)

        # 그래프 제목은 영문/숫자 중심으로 표시해서 한글 폰트 깨짐 및 font warning 방지
        ax1.set_title(
            f"Sortino Rank #{rank_no} | Gap30 Excess={row['_현재30일선초과이격도']:.1f} | P97 Excess={row['_247거래일97%초과이격도']:.1f} | Heat={row['_97%구간대비현재위치']:.1f}%",
            fontsize=13,
            fontweight="bold",
        )

        lines1, labels1 = ax1.get_legend_handles_labels()
        lines2, labels2 = ax2.get_legend_handles_labels()

        ax1.legend(
            lines2 + lines1,
            labels2 + labels1,
            loc="upper center",
            bbox_to_anchor=(0.5, 1.18),
            ncol=3,
            frameon=False,
            fontsize=11,
        )

        plt.tight_layout()
        plt.show()
