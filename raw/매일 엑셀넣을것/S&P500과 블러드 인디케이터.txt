# 라이브러리 임포트
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import yfinance as yf
from datetime import datetime

# -----------------------------------------------------------
# 1. 데이터 다운로드
# -----------------------------------------------------------
start_date = "2007-01-01"
today = datetime.today().strftime('%Y-%m-%d')

# HYG 가격
hyg = yf.download("HYG", start=start_date, end=today)['Close']

# IRX (3M T-bill 금리)
irx = yf.download("^IRX", start=start_date, end=today)['Close'] / 100
irx = irx.resample('W-FRI').last()

# 10년물 금리 (Yahoo)
t10y = yf.download("^TNX", start=start_date, end=today)['Close'] / 10
t10y = t10y.resample('W-FRI').last()

# SPY 가격
spy = yf.download("SPY", start=start_date, end=today)['Close']
spy = spy.resample('W-FRI').last()

# -----------------------------------------------------------
# 2. HYG 배당수익률 가져오기
# -----------------------------------------------------------
hyg_info = yf.Ticker("HYG").info
hyg_dividend_yield = hyg_info.get('dividendYield', None)

if hyg_dividend_yield is None:
    raise ValueError("HYG dividend yield 정보를 가져올 수 없습니다.")

hyg_dividend_yield = hyg_dividend_yield * 100
print(f"현재 HYG Dividend Yield: {hyg_dividend_yield:.2f}%")

# 시계열로 확장
hyg_yield_series = pd.Series(hyg_dividend_yield, index=hyg.index)
hyg_yield_series = hyg_yield_series.resample('W-FRI').last()

# -----------------------------------------------------------
# 3. 데이터 정렬
# -----------------------------------------------------------
df = pd.concat([irx, t10y, spy, hyg_yield_series], axis=1)

df.columns = [
    "IRX",
    "T10Y",
    "SPY",
    "HYG_YIELD"
]

df.dropna(inplace=True)

# -----------------------------------------------------------
# 4. High Yield Spread Proxy 계산
# -----------------------------------------------------------
df["HY_SPREAD_PROXY"] = df["HYG_YIELD"] - df["T10Y"]

# -----------------------------------------------------------
# 5. BLOOD Proxy 계산
# -----------------------------------------------------------
df["BLOOD_PROXY"] = df["IRX"] / df["HY_SPREAD_PROXY"]

# 이동평균
df["20MA"] = df["BLOOD_PROXY"].rolling(20).mean()
df["60MA"] = df["BLOOD_PROXY"].rolling(60).mean()
df["120MA"] = df["BLOOD_PROXY"].rolling(120).mean()

# -----------------------------------------------------------
# 6. BLOOD Proxy + SPY 그래프
# -----------------------------------------------------------
fig, ax1 = plt.subplots(figsize=(14, 7))

# BLOOD Proxy
ax1.set_xlabel('Date')
ax1.set_ylabel('BLOOD PROXY', color='tab:blue')

ax1.plot(df.index, df["BLOOD_PROXY"], color='tab:blue', label='BLOOD_PROXY')
ax1.plot(df.index, df["20MA"], '--', label='20MA')
ax1.plot(df.index, df["60MA"], '--', label='60MA')
ax1.plot(df.index, df["120MA"], '--', label='120MA')

ax1.tick_params(axis='y', labelcolor='tab:blue')
ax1.legend(loc='upper left')

# SPY
ax2 = ax1.twinx()

ax2.set_ylabel('SPY Price', color='tab:red')
ax2.plot(df.index, df["SPY"], color='tab:red', alpha=0.5, label='SPY')

ax2.tick_params(axis='y', labelcolor='tab:red')
ax2.legend(loc='upper right')

plt.title("BLOOD PROXY (IRX / (HYG Yield - 10Y Yield)) + SPY Price")
plt.grid()
plt.show()
