# 핵심 입력들만 우선 결측 제거(금리/변동성/채권ETF 중 하나라도 없으면 파생지표 불가)
data = data.dropna(subset=['S&P500', 'NASDAQ', 'VIX', '10Y', '5Y', 'HYG', 'IEF', 'QQQ', 'SPY'])

# Risk Appetite (HYG / IEF)
data['Risk_Appetite'] = data['HYG'] / data['IEF']


# ---------------------------------------
# 1. RSI 함수
# ---------------------------------------
def calculate_rsi(df, column, window=10):
    delta = df[column].diff()
    gain = (delta.where(delta > 0, 0)).rolling(window).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(window).mean()

    # 0으로 나누는 케이스 방지 (loss=0이면 rs가 inf -> rsi가 100에 수렴)
    rs = gain / loss.replace(0, np.nan)
    rsi = 100 - (100 / (1 + rs))

    return rsi


# ---------------------------------------
# 2. MACD 오실레이터 (MACD - Signal)
# ---------------------------------------
def calculate_macd(df, column):
    short_ema = df[column].ewm(span=12, adjust=False).mean()
    long_ema = df[column].ewm(span=26, adjust=False).mean()
    macd = short_ema - long_ema
    signal = macd.ewm(span=9, adjust=False).mean()
    return macd - signal  # Histogram


# ---------------------------------------
# 3. Fear & Greed 계산 함수 (S&P500 / NASDAQ)
#    ✅ 스케일링 전에 NaN 제거해서 RuntimeWarning 제거
# ---------------------------------------
def calculate_fear_greed(df, index_col, label):
    # 125일 이동평균 기준 모멘텀
    df[f'{label}_125MA'] = df[index_col].rolling(125).mean()
    df[f'{label}_Momentum'] = (df[index_col] - df[f'{label}_125MA']) / df[f'{label}_125MA'] * 100

    # RSI
    df[f'{label}_RSI'] = calculate_rsi(df, index_col)

    # 금리 스프레드 (10Y - 5Y)
    df[f'{label}_BondSpread'] = df['10Y'] - df['5Y']

    # VIX, Risk Appetite
    df[f'{label}_VIX'] = df['VIX']
    df[f'{label}_RiskAppetite'] = df['Risk_Appetite']

    # 정규화 대상 컬럼
    cols = [
        f'{label}_Momentum',
        f'{label}_RSI',
        f'{label}_BondSpread',
        f'{label}_VIX',
        f'{label}_RiskAppetite'
    ]

    # ✅ NaN 구간 제거 후 그 구간에만 스케일링 결과를 다시 주입
    valid = df[cols].dropna()
    scaler = MinMaxScaler()
    scaled = scaler.fit_transform(valid)
    df.loc[valid.index, cols] = scaled

    # Fear & Greed Index (0~1)
    # - RiskAppetite: 높을수록 긍정
    # - VIX: 낮을수록 긍정 => (1 - VIX)
    df[f'{label}_FGI'] = (
        df[f'{label}_Momentum'] * 0.2 +
        df[f'{label}_RiskAppetite'] * 0.2 +
        (1 - df[f'{label}_VIX']) * 0.2 +
        df[f'{label}_BondSpread'] * 0.2 +
        df[f'{label}_RSI'] * 0.2
    )

    # F&G 오실레이터 (MACD 방식)
    df[f'{label}_Oscillator'] = calculate_macd(df, f'{label}_FGI')
    return df


# S&P500, NASDAQ F&G 계산
data = calculate_fear_greed(data, 'S&P500', 'SPX')
data = calculate_fear_greed(data, 'NASDAQ', 'NDX')


# ---------------------------------------
# 4. QQQ / SPY 합성이평(SuperMA) + 괴리율
# ---------------------------------------
def add_super_ma_gap(df, price_col, label, windows=[20, 60, 120, 200]):
    for w in windows:
        df[f'{label}_MA{w}'] = df[price_col].rolling(w).mean()
    ma_cols = [f'{label}_MA{w}' for w in windows]
    df[f'{label}_SuperMA'] = df[ma_cols].mean(axis=1)
    df[f'{label}_GapPct'] = (df[price_col] - df[f'{label}_SuperMA']) / df[f'{label}_SuperMA'] * 100
    return df

data = add_super_ma_gap(data, 'QQQ', 'QQQ')
data = add_super_ma_gap(data, 'SPY', 'SPY')


# ---------------------------------------
# 5. 임펄스 시스템 (EMA13 + MACD Histogram)
# ---------------------------------------
def add_impulse_components(df, price_col, label):
    df[f'{label}_EMA13'] = df[price_col].ewm(span=13, adjust=False).mean()
    df[f'{label}_MACD_Hist'] = calculate_macd(df, price_col)
    return df


def get_impulse_colors(df, ema_col, macd_col):
    colors = []
    ema = df[ema_col]
    macd = df[macd_col]

    for i in range(len(df)):
        if i == 0:
            colors.append('gray')
            continue

        ema_up = ema.iloc[i] > ema.iloc[i-1]
        ema_down = ema.iloc[i] < ema.iloc[i-1]
        macd_up = macd.iloc[i] > macd.iloc[i-1]
        macd_down = macd.iloc[i] < macd.iloc[i-1]

        if ema_up and macd_up:
            colors.append('green')
        elif ema_down and macd_down:
            colors.append('red')
        else:
            colors.append('blue')
    return colors

data = add_impulse_components(data, 'QQQ', 'QQQ')
data = add_impulse_components(data, 'SPY', 'SPY')


# ---------------------------------------
# 6. DeMark TD Setup 카운트 (Sell/Buy)
#    Sell = 4일, Buy = 2일 기준
# ---------------------------------------
def add_td_setup_counts(df, price_col, label):
    n = len(df)
    sell = np.zeros(n)
    buy = np.zeros(n)

    prices = df[price_col].values

    for i in range(n):
        # Sell Setup (4일 전보다 계속 위면 카운트)
        if i >= 4 and prices[i] > prices[i-4]:
            sell[i] = sell[i-1] + 1
        else:
            sell[i] = 0

        # Buy Setup (2일 전보다 계속 아래면 카운트)
        if i >= 2 and prices[i] < prices[i-2]:
            buy[i] = buy[i-1] + 1
        else:
            buy[i] = 0

    df[f'{label}_TD_SellSetup'] = sell
    df[f'{label}_TD_BuySetup'] = buy
    return df

data = add_td_setup_counts(data, 'SPY', 'SPY')
data = add_td_setup_counts(data, 'QQQ', 'QQQ')
