```python
@app.post("/api/lens/yt")
async def api_lens_yt(request: Request, frame: UploadFile = File(None),
                       source_caption: str = Form(""), max_results: int = Form(40)):
    """프레임+캡션으로 유튜브를 키워드 검색해 렌즈 결과에 합류할 항목. YouTube Data API라
    무료(쿼터 내) → 월 호출가드 없음. 검색어는 cn_search_keyword_vision의 product
    (프롬프트가 '한국어 제품명'을 명시) → 캡션 앞토큰(_cn_keyword) 폴백."""
    keyword = ""
    if frame is not None:
        try:
            raw = await frame.read()
            v = cn_search_keyword_vision(raw, source_caption)
            keyword = (v.get("product") or "").strip()
        except Exception:
            keyword = ""
    if not keyword:
        keyword = _cn_keyword(source_caption)
    if not keyword:
        return {"ok": True, "items": [], "count": 0, "note": "검색어를 만들지 못했습니다"}
    n = max(1, min(int(max_results or 40), 60))
    try:
        rows = youtube_search.search(keyword, max_results=n)
    except Exception:
        rows = []
    for r in rows:
        r["platform"] = "youtube"
        r["match"] = None
    return {"ok": True, "items": rows, "count": len(rows), "keyword": keyword}
```

```javascript
async function fetchLensYt(shortcode, caption, blob){
  try{
    const fd=new FormData();
    if(blob) fd.append('frame', blob, 'f.jpg');   // 비전으로 화면 제품명 추출
    if(caption) fd.append('source_caption', caption);
    fd.append('max_results', '40');
    const d=await (await fetch('/api/lens/yt',{method:'POST',body:fd})).json();
    const st=LENS_STATE[shortcode]; if(!st) return;   // 사용자가 닫았을 수 있음
    st.keyword=d.keyword||'';
    const add=(d.items||[]);
    add.forEach(i=>{ i.savesc='lens_'+i.platform+'_'+_lensHash(i.url); });
    const have=new Set(st.items.map(i=>i.savesc));
    st.items.push(...add.filter(i=>!have.has(i.savesc)));   // 중복 savesc 제외
    st.ytLoading=false;
  }catch(e){ const st=LENS_STATE[shortcode]; if(st) st.ytLoading=false; }
  const root=document.getElementById('lensRoot');
  if(root && root.dataset.sc===shortcode) renderLens(shortcode);
}
```

```javascript
// 플랫폼별 검색 페이지 URL(search_links.py의 _URL_BUILDERS와 동일). 바로가기+담기용.
function _lensSearchUrl(platform, kw){
  const q=encodeURIComponent(kw||'');
  const B={
    tiktok:`https://www.tiktok.com/search?q=${q}`,
    instagram:`https://www.instagram.com/explore/search/keyword/?q=${q}`,
    xiaohongshu:`https://www.xiaohongshu.com/search_result?keyword=${q}`,
    douyin:`https://www.douyin.com/search/${q}`,
  };
  return B[platform]||'#';
}
```
