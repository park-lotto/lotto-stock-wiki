# 텔레그램 인제스트 시스템 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 17개 텔레그램 채널의 하루치 메시지를 타입별 질문지로 추출해 2층(사실/스탠스/방법론) 원자로 atoms DB에 누적한다.

**Architecture:** 채널 레지스트리로 타입·신뢰등급을 고정 매핑 → Gemini flash-lite가 타입별 질문지를 채움 → 결정론적 코드가 fan-out(2층·외국주 분기·종목명 정규화·이벤트 병합) → quote 원문대조 검증 → atoms DB 저장. 리포트 인제스트(`questionnaire.py`/`report_ingest.py`/`verify_questionnaire.py`) 패턴을 차용한다.

**Tech Stack:** Python 3.14, google-genai(gemini-3.1-flash-lite), sqlite3, pytest. 기존 모듈: `codemap.py`, `db.py`, `vector_db.py`, `verify_questionnaire.py`.

## Global Constraints

- 설계 출처: `docs/superpowers/specs/2026-06-21-텔레그램-인제스트-design.md`. 모든 슬롯·규칙은 이 스펙을 따른다.
- **2층 모델**: L-FACT(append, validity=event) / L-STANCE(같은 `stance_key`로 갱신, 과거 is_active=0) / L-METHOD(append, asset_level="method", permanent).
- **quote는 원문 축자** = 강한 신뢰. `comment`/`reason`은 재서술 허용 = 약한 신뢰(검증 안 함).
- **종목 분기**: `is_korean_stock(name)` True만 종목 원자 생성. False(외국주)는 섹터 원자 본문에 보존. 매칭 실패 한국주는 `raw/telegram_unmatched.log`에 기록.
- **종목명 정규화**: `stock_aliases.json` 통과 후 `codemap.code_for()`.
- **이벤트 병합**: `event_key`로 그룹핑, `mention_channels`/`mention_count` 누적.
- **strength**: base(trust B=3, C=2) × 층가중(FACT 1.0 / STANCE 0.8 / METHOD 0.7) + (mention_count-1), 상한 4.
- 제외 채널: `리포트요약`, `투경현황`, 이미지/스크린샷 파일.
- 기존 `insert_atom` 호환 유지(리포트 파이프라인 깨지면 안 됨): 신규 필드는 `setdefault`로 기본값.
- 토큰 절약: Gemini 호출은 테스트하지 않는다(결정론 로직만 테스트). 골든셋=스파이크 4종.

```python
# pipeline/atoms/telegram_registry.py
"""텔레그램 채널 레지스트리 + 종목명 별칭."""
import json
from pathlib import Path

_DIR = Path(__file__).parent
_CHANNELS = json.loads((_DIR / "telegram_channels.json").read_text(encoding="utf-8"))
_ALIASES_RAW = json.loads((_DIR / "stock_aliases.json").read_text(encoding="utf-8"))
_ALIASES = {k.lower(): v for k, v in _ALIASES_RAW.items()}

_EXCLUDED = {"리포트요약", "투경현황"}


def channel_info(name: str) -> dict | None:
    return _CHANNELS.get((name or "").strip())


def is_excluded(name: str) -> bool:
    return (name or "").strip() in _EXCLUDED


def resolve_alias(name: str) -> str:
    if not name:
        return name
    return _ALIASES.get(name.strip().lower(), name.strip())
```

```python
# pipeline/atoms/stock_resolve.py
"""종목명 정규화: 별칭 → codemap. 한국주 추정 매칭 실패는 로그."""
import re
from pathlib import Path

from .telegram_registry import resolve_alias
from .codemap import code_for, is_korean_stock

UNMATCHED_LOG = Path(__file__).parent.parent.parent / "raw" / "telegram_unmatched.log"

# 한글이 포함되면 한국 종목일 가능성 — 외국주(영문/숫자)와 구분
_HANGUL = re.compile(r"[가-힣]")


def resolve_stock(name: str, *, date: str, channel: str) -> dict:
    norm = resolve_alias(name)
    code = code_for(norm)
    if code:
        return {"name": norm, "code": code, "is_korean": True, "matched": True}

    # 코드 없음: 한글 포함 = 한국주 추정인데 미매칭 → 로그
    looks_korean = bool(_HANGUL.search(norm))
    if looks_korean:
        UNMATCHED_LOG.parent.mkdir(parents=True, exist_ok=True)
        with UNMATCHED_LOG.open("a", encoding="utf-8") as f:
            f.write(f"{date}\t{channel}\t{norm}\n")
    return {"name": norm, "code": None,
            "is_korean": is_korean_stock(norm), "matched": False}
```

```python
import hashlib
import re as _re

from .stock_resolve import resolve_stock
from .telegram_registry import channel_info  # noqa: F401 (호출자 편의)

_TRUST_BASE = {"A": 4, "B": 3, "C": 2, "D": 1}
_LAYER_W = {"fact": 1.0, "stance": 0.8, "method": 0.7}
_SECTOR_KEYS = ["반도체", "조선", "로봇", "방산", "바이오", "전력", "2차전지",
                "자동차", "통신", "철강", "중국", "기타"]


def _mk_id(channel: str, date: str, tag: str, i: int) -> str:
    raw = f"{channel}_{date}_{tag}_{i}"
    return "atom_tg_" + hashlib.md5(raw.encode()).hexdigest()[:12]


def _strength(trust: str, layer_tag: str, mention_count: int) -> int:
    base = _TRUST_BASE.get(trust, 1) * _LAYER_W.get(layer_tag, 1.0)
    return max(1, min(4, round(base + (mention_count - 1))))


def _event_key(fact: str) -> str:
    norm = _re.sub(r"[^가-힣A-Za-z0-9]", "", fact or "")
    return hashlib.md5(norm.encode()).hexdigest()[:10]


def _norm_sector(s: str) -> str:
    if not s:
        return "기타"
    for k in _SECTOR_KEYS:
        if k in s:
            return k
    return "기타"


def _base(meta: dict, **kw) -> dict:
    d = {
        "id": "", "date": meta["date"], "source_type": "telegram",
        "source_name": meta["channel"], "source_trust": meta.get("trust", "C"),
        "raw_file": meta.get("raw_file", ""), "layer": "L5",
        "sector": meta.get("sector") or "기타", "asset": "", "asset_level": "sector",
        "signal": "neutral", "event_type": "report", "magnitude": "minor",
        "content_type": "fact", "strength_score": 1, "validity_type": "permanent",
        "validity_until": None, "is_active": 1, "content": "", "relations": [],
        "stance_key": None, "mention_channels": [meta["channel"]],
        "mention_count": 1, "msg_ts": None,
    }
    d.update(kw)
    return d


def _stock_atom(meta, name_info, content, *, ts, i, layer_tag="fact"):
    return _base(
        meta,
        id=_mk_id(meta["channel"], meta["date"], "stk", i),
        asset=name_info["name"], asset_level="stock",
        content_type="fact", msg_ts=ts,
        strength_score=_strength(meta.get("trust", "C"), layer_tag, 1),
        content=content,
    )


def questionnaire_to_atoms_tg(q: dict, meta: dict) -> list[dict]:
    ctype = meta["type"]
    atoms: list[dict] = []
    trust = meta.get("trust", "C")

            if not raw:
                continue
            info = resolve_stock(raw, date=meta["date"], channel=meta["channel"])
            quote = s.get("quote") or s.get(key_text) or ""
            if info["is_korean"] and info["matched"]:

                                         ts=s.get("ts"), i=len(atoms)))

    if ctype == "sector":

            meta, id=_mk_id(meta["channel"], meta["date"], "mkt", 0),
            asset="시장", asset_level="market", event_type="macro",
            content=q.get("market_direction") or "",
            strength_score=_strength(trust, "fact", 1),
        ))

            atoms.append(_base(
                meta, id=_mk_id(meta["channel"], meta["date"], "macro", i),

                msg_ts=nw.get("ts"), content=nw.get("fact") or "",
                strength_score=_strength(trust, "fact", 1),
            ))

    elif ctype == "insight":
        for i, st in enumerate(q.get("stance") or []):
            target = (st.get("target") or "").strip()
            if not target:
                continue
            atoms.append(_base(
                meta, id=_mk_id(meta["channel"], meta["date"], "stance", i),
                asset=target, asset_level="stance",
                stance_key=f"{meta['channel']}|{target}",

                    ts=rp.get("ts"), i=len(atoms)))

    return atoms
```

```python
"""스탠스 만료: 같은 (채널|대상)의 옛 판단을 비활성화. 과거는 삭제하지 않음(changelog)."""
from datetime import datetime
from .db import get_conn


def deactivate_prior_stance(stance_key: str, keep_id: str) -> int:
    if not stance_key:
        return 0
    conn = get_conn()
    cur = conn.execute(
        "UPDATE atoms SET is_active=0, updated_at=? "
        "WHERE stance_key=? AND id<>? AND is_active=1",
        [datetime.now().isoformat(), stance_key, keep_id],
    )
    n = cur.rowcount
    conn.commit()
    conn.close()
    return n
```

```python
    out.mkdir(parents=True, exist_ok=True)
    (out / f"{channel}.json").write_text(
        json.dumps(q, ensure_ascii=False, indent=2), encoding="utf-8")


def ingest_telegram(md_path: Path) -> int:
    date, channel = _parse_filename(md_path.name)
    if _is_image_file(md_path.name) or is_excluded(channel) or "_digest" in md_path.name:
        return 0
    info = channel_info(channel)
    if not info:
        print(f"  [SKIP] 미등록 채널: {channel}")
        return 0

    q = extract_telegram(md_path, info["type"])
    if not q:
        print(f"  [WARN] 빈 질문지: {channel}")
        return 0

    _save_artifact(q, date, channel)
    meta = {"date": date, "channel": channel, "type": info["type"],
            "sector": info.get("sector"), "trust": info["trust"],
            "raw_file": str(md_path)}
    atoms = questionnaire_to_atoms_tg(q, meta)
    for a in atoms:
        if a.get("stance_key"):
            deactivate_prior_stance(a["stance_key"], keep_id=a["id"])
        insert_atom(a)
        try:
            embed_and_store(a)
        except Exception:
            pass
    return len(atoms)


def get_pending_telegram(date_filter: str = None) -> list[Path]:
    files = []
    for f in sorted(_TG_DIR.glob("*.md")):
        if "_digest" in f.name:
            continue
        date, channel = _parse_filename(f.name)
        if date_filter and date != date_filter:
            continue
        if is_excluded(channel) or not channel_info(channel):
            continue
        files.append(f)
    return sorted(files, reverse=True)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("file", nargs="?")
    ap.add_argument("--all", action="store_true")
    ap.add_argument("--date", default=None)
    ap.add_argument("--limit", type=int, default=30)
    ap.add_argument("--dry-run", action="store_true")
    args = ap.parse_args()
    init_db(); migrate_db()

    if args.file:
        targets = [Path(args.file)]
    elif args.all:
        targets = get_pending_telegram(args.date)[: args.limit]
        print(f"미처리 텔레그램: {len(targets)}개\n")
    else:
        print("file 또는 --all 필요")
        return

    if args.dry_run:
        for f in targets:
            d, c = _parse_filename(f.name)
            info = channel_info(c)
            print(f"  {f.name} → type={info['type'] if info else '?'}")
        return

    total = 0
    for i, f in enumerate(targets, 1):
        print(f"[{i}/{len(targets)}] {f.name}")
        n = ingest_telegram(f)
        total += n
        print(f"  → {n}개 원자")
    print(f"\n완료: {len(targets)}개 채널, {total}개 원자")


if __name__ == "__main__":
    main()
```

```python
# pipeline/atoms/verify_telegram.py
"""텔레그램 quote 원문 대조 (정답지 없음 → 유일한 1차 검증)."""


def _all_quotes(q: dict) -> list[str]:
    out = []

    def walk(v):
        if isinstance(v, dict):
            for k, val in v.items():
                if k == "quote" and isinstance(val, str) and val.strip():
                    out.append(val.strip())
                else:
                    walk(val)
        elif isinstance(v, list):
            for x in v:
                walk(x)

    walk(q)
    return out


def verify_telegram_quotes(q: dict, md_text: str) -> list[dict]:
    flags = []
    for quote in _all_quotes(q):
        snippet = quote[:15].strip()
        if snippet and snippet not in md_text:
            flags.append({"code": "TG_QUOTE_NOT_FOUND",
                          "msg": f"인용 미발견: '{snippet}...'"})
    return flags
```
