import yfinance as yf
import pandas as pd
import numpy as np

# =========================================================
# 1. 사용자 입력
# =========================================================
# 네가 이미 상대강도 상위 업종으로 뽑아둔 ETF만 넣으면 됨
selected_tickers = [
    "487240.KS", "421320.KS", "367760.KS", "091160.KS", "449450.KS",
    "210780.KS", "305540.KS", "244580.KS"
]

start_date = "2024-01-01"

# 옵션
use_absolute_momentum_filter = True   # 절대모멘텀(음수 제외)
max_weight = 0.35                     # 종목당 최대 비중 제한
min_data_days = 200                   # 최소 데이터 일수
vol_lookback = 126                    # 변동성 계산 기간(약 6개월)
momentum_mode = "13612"               # "13612" 또는 "6912"

# =========================================================
# 2. 티커 이름 가져오기 (야후 이름 그대로)
# =========================================================
def get_ticker_names(tickers):
    """
    Yahoo Finance의 longName -> shortName -> ticker 순서로 이름 반환
    """
    name_dict = {}
    for t in tickers:
        try:
            info = yf.Ticker(t).info
            name = info.get("longName") or info.get("shortName") or t
            name_dict[t] = name
        except Exception:
            name_dict[t] = t
    return name_dict

# =========================================================
# 3. 데이터 다운로드
# =========================================================
raw = yf.download(
    selected_tickers,
    start=start_date,
    auto_adjust=True,
    progress=False
)

# yfinance 반환 형태 대응
if isinstance(raw.columns, pd.MultiIndex):
    # 보통 auto_adjust=True여도 Close 사용 가능
    if "Close" in raw.columns.get_level_values(0):
        data = raw["Close"].copy()
    else:
        first_field = raw.columns.get_level_values(0)[0]
        data = raw[first_field].copy()
else:
    data = raw.copy()

# 단일 종목일 때 Series -> DataFrame 변환
if isinstance(data, pd.Series):
    data = data.to_frame(name=selected_tickers[0])

# 인덱스 정렬
data = data.sort_index()

# =========================================================
# 4. 최소 데이터 길이 충족 종목만 사용
# =========================================================
valid_cols = [col for col in data.columns if data[col].dropna().shape[0] >= min_data_days]
data = data[valid_cols].copy()

if data.shape[1] == 0:
    raise ValueError("사용 가능한 종목이 없습니다. start_date를 더 과거로 늘리거나 티커를 확인하세요.")

print("사용 가능 종목:", list(data.columns))
print("종목 수:", len(data.columns))

# =========================================================
# 5. 수익률 계산
# =========================================================
returns = data.pct_change()

# =========================================================
# 6. 모멘텀 계산
# =========================================================
if momentum_mode == "13612":
    # 1, 3, 6, 12개월 모멘텀 합산
    mom_1 = data.pct_change(21)
    mom_3 = data.pct_change(63)
    mom_6 = data.pct_change(126)
    mom_12 = data.pct_change(252)

    momentum = mom_1 + mom_3 + mom_6 + mom_12

elif momentum_mode == "6912":
    # 6, 9, 12개월 평균
    mom_6 = data.pct_change(126)
    mom_9 = data.pct_change(189)
    mom_12 = data.pct_change(252)

    momentum = (mom_6 + mom_9 + mom_12) / 3

else:
    raise ValueError("momentum_mode는 '13612' 또는 '6912' 중 하나여야 합니다.")

# 최신 모멘텀
mu = momentum.iloc[-1].dropna().copy()

if len(mu) == 0:
    raise ValueError("최신 모멘텀을 계산할 수 없습니다. start_date를 더 과거로 설정하세요.")

# =========================================================
# 7. 변동성 계산
# =========================================================
hist_returns = returns.iloc[-vol_lookback:].copy()
hist_returns = hist_returns.dropna(axis=1, how="all")

common = mu.index.intersection(hist_returns.columns)
mu = mu.loc[common]
hist_returns = hist_returns[common]

vol = hist_returns.std().replace(0, np.nan).dropna()

common = mu.index.intersection(vol.index)
mu = mu.loc[common]
vol = vol.loc[common]

if len(mu) == 0:
    raise ValueError("모멘텀/변동성 공통 종목이 없습니다.")

# =========================================================
# 8. 절대모멘텀 필터
# =========================================================
if use_absolute_momentum_filter:
    mu = mu[mu > 0]
    vol = vol.loc[mu.index]

if len(mu) == 0:
    print("⚠️ 절대모멘텀 필터 적용 후 남은 종목이 없습니다.")
    result = pd.DataFrame(columns=["Name", "Momentum", "Volatility", "Score", "Weight", "Weight_%"])
else:
    # =====================================================
    # 9. 점수 계산: Momentum / Volatility
    # =====================================================
    score = (mu / vol).replace([np.inf, -np.inf], np.nan).dropna()
    score = score[score > 0]

    if len(score) == 0:
        print("⚠️ 양수 점수를 가진 종목이 없습니다.")
        result = pd.DataFrame(columns=["Name", "Momentum", "Volatility", "Score", "Weight", "Weight_%"])
    else:
        # =================================================
        # 10. 1차 비중
        # =================================================
        weights = score / score.sum()

        # =================================================
        # 11. 최대 비중 제한 함수
        # =================================================
        def cap_weights(weight_series: pd.Series, cap: float) -> pd.Series:
            """
            비중 상한(cap)을 넘는 종목은 cap으로 제한하고,
            남는 비중은 cap 미만 종목에 비례 재분배
            """
            w = weight_series.copy().astype(float)

            if cap >= 1.0:
                return w / w.sum()

            for _ in range(100):
                over = w > cap
                if not over.any():
                    break

                excess = (w[over] - cap).sum()
                w[over] = cap

                under = w < cap
                if under.sum() == 0:
                    break

                redistribute_base = w[under].sum()
                if redistribute_base <= 0:
                    break

                w[under] += (w[under] / redistribute_base) * excess

            return w / w.sum()

        weights = cap_weights(weights, max_weight)

        # =================================================
        # 12. 이름 매핑
        # =================================================
        name_map = get_ticker_names(weights.index.tolist())

        # =================================================
        # 13. 결과 정리
        # =================================================
        result = pd.DataFrame({
            "Name": [name_map[t] for t in weights.index],
            "Momentum": mu.loc[weights.index],
            "Volatility": vol.loc[weights.index],
            "Score": score.loc[weights.index],
            "Weight": weights
        }, index=weights.index)

        result["Weight_%"] = result["Weight"] * 100
        result = result.sort_values("Weight", ascending=False)

        print("\n📊 주도 업종 ETF 비중 산출 결과\n")
        print(result)
        print("\n총합:", result["Weight"].sum())

# =========================================================
# 14. 최종 비중만 깔끔하게 출력
# =========================================================
if len(result) > 0:
    final_weights = result[["Name", "Weight"]].copy()
    final_weights["Weight_%"] = (final_weights["Weight"] * 100).round(2).astype(str) + "%"

    print("\n✅ 최종 투자 비중\n")
    print(final_weights[["Name", "Weight_%"]])
