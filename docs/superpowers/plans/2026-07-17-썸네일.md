```python
def _probe_duration(video_path):
    """영상 길이(초). 못 구하면 None."""
    cmd = [
        "ffprobe", "-v", "error", "-show_entries", "format=duration",
        "-of", "default=noprint_wrappers=1:nokey=1", str(video_path),
    ]
    result = subprocess.run(cmd, capture_output=True, check=False,
                            stdin=subprocess.DEVNULL)
    if result.returncode != 0:
        return None
    try:
        return float(result.stdout.decode().strip())
    except (ValueError, AttributeError):
        return None


def extract_grid_frames(video_path, dest_dir, n=10):
    """영상을 n등분해 각 구간 중앙의 프레임을 뽑는다 → [(Path, ts), ...].

    장면전환 감지(extract_frames)와 갈리는 지점(2026-07-17 썸네일 설계):
    장면전환은 ①감지가 적으면 빈 리스트고 ②ts를 안 주며 ③위 extract_frame_at의
    주석대로 "얼굴·손·배경 같은 프레임이 섞"인다. 썸네일 후보 그리드는 개수가
    보장되고 시각을 표시할 수 있어야 하므로 등분이 맞다.

    구간 중앙(i+0.5)을 쓰는 이유: 0초는 검은 첫 프레임이 잡히고, duration 정각은
    범위 밖이라 추출이 실패한다.

    ⚠️ -ss에 초를 넘기는 건 결함이 아니다 — ffmpeg엔 -start_frame이 없어 seek은
    본질적으로 시간 기반이고, 길이는 extract_frame_at이 -frames:v 1로 잡는다.
    """
    dest_dir = Path(dest_dir)
    dest_dir.mkdir(parents=True, exist_ok=True)
    duration = _probe_duration(video_path)
    if not duration or duration <= 0:
        raise RuntimeError(f"영상 길이를 구할 수 없다: {video_path}")
    out = []
    for i in range(n):
        ts = duration * (i + 0.5) / n
        path = extract_frame_at(video_path, dest_dir, ts, filename=f"grid_{i:02d}.jpg")
        if path is not None:          # 실패는 조용히 건너뛴다(extract_frame_at 기존 계약)
            out.append((path, ts))
    return out
```

def _thumb_dir(job_id: str):
    """job_id 검증 후 그 job의 썸네일 폴더. 경로순회를 여기서 한 번에 막는다.
    부적합하면 None — 호출부가 400으로 돌려준다(이 파일은 HTTPException을 안 쓴다)."""
    safe = os.path.basename(job_id)
    if not safe or safe != job_id or safe in (".", ".."):

@app.post("/api/produce/thumb/frames")
def api_thumb_frames(body: dict):
    """5단계 썸네일 — 믹스 결과 영상을 등분해 후보 프레임 10장.

    자막이 박히지 않은 2단계 결과(video_path)에서 뽑는다(설계 Q1) — 썸네일 텍스트는
    위에 새로 얹으므로 배경 자막은 방해다. 이미 뽑아뒀으면 재추출하지 않는다.
    """
    job_id = str(body.get("job_id") or "")
    job = Store(DB_PATH).get_mix_job(job_id)
    if not job:
        return JSONResponse(status_code=404, content={"ok": False, "error": "job 없음"})
    video = job.get("video_path")
    if not video or not Path(video).exists():
        return JSONResponse(status_code=404, content={"ok": False, "error": "믹스 영상 없음"})

    out_dir = _thumb_dir(job_id)
    if out_dir is None:
        return JSONResponse(status_code=400, content={"ok": False, "error": "bad job_id"})
    existing = sorted(out_dir.glob("grid_*.jpg")) if out_dir.exists() else []
    if existing:
        meta = (job.get("thumbnail") or {}).get("frames") or []
        if len(meta) == len(existing):
            return {"ok": True, "frames": meta}

    try:
        pairs = extract_grid_frames(video, out_dir, n=10)
    except RuntimeError as e:
        return JSONResponse(status_code=502, content={"ok": False, "error": str(e)})

    frames = [{"url": f"/api/produce/thumb/file/{job_id}/{p.name}", "ts": round(ts, 2)}
              for p, ts in pairs]
    thumb = job.get("thumbnail") or {}
    thumb["frames"] = frames
    Store(DB_PATH).update_mix_job(job_id, thumbnail=thumb)
    return {"ok": True, "frames": frames}


@app.get("/api/produce/thumb/file/{job_id}/{name}")

// ── 6단계 썸네일 (2026-07-17) ──────────────────────────────────
// 합성은 전적으로 여기(canvas)서 한다. 서버는 완성된 PNG를 받아 저장만 한다(설계 Q3).
// 프리뷰 캔버스와 업로드 캔버스가 **같은 하나**라서 "보이는 그대로"가 구조적으로 보장된다.
// 좌표(x·y)는 0~1 비율, size는 1080 기준 px — 그래야 270px 프리뷰와 1080px 결과가 일치한다.
const THUMB_W = 1080, THUMB_H = 1920;

function thumbPresetToLayer(p){
  return {
    text: '', font: p.font, size: p.size, color: p.color,
    outline: p.outline ? {color: p.outline_color || '#000000', w: p.outline_w || 0} : null,
    box: p.box ? {color: p.box_color || '#000000', pad: p.box_pad != null ? p.box_pad : 16,
                  opacity: p.box_opacity != null ? p.box_opacity : 80} : null,
    rot: 0, x: 0.5, y: (p.y != null ? p.y : 12) / 100,
  };
}

function thumbFontCss(file){
  const f = (typeof HC_FONTS !== 'undefined' ? HC_FONTS : []).find(x => x.file === file);
  return f ? f.css : 'sans-serif';
}

function drawThumb(ctx, img, layers, W, H){
  const s = W / THUMB_W;                    // 캔버스 배율 — 프리뷰든 실측이든 같은 식
  if (img) ctx.drawImage(img, 0, 0, W, H);
  (layers || []).forEach(L => {
    if (!L.text) return;
    const size = L.size * s;
    ctx.save();
    ctx.translate(L.x * W, L.y * H);
    if (L.rot) ctx.rotate(L.rot * Math.PI / 180);
    ctx.font = `900 ${size}px ${thumbFontCss(L.font)}`;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    if (L.box){
      const w = ctx.measureText(L.text).width, pad = L.box.pad * s;
      ctx.globalAlpha = (L.box.opacity != null ? L.box.opacity : 80) / 100;
      ctx.fillStyle = L.box.color;
      ctx.fillRect(-w/2 - pad, -size/2 - pad*0.6, w + pad*2, size + pad*1.2);
      ctx.globalAlpha = 1;
    }
    if (L.outline && L.outline.w){
      ctx.lineWidth = L.outline.w * s;
      ctx.strokeStyle = L.outline.color;
      ctx.lineJoin = 'round';
      ctx.strokeText(L.text, 0, 0);
    }
    ctx.fillStyle = L.color;
    ctx.fillText(L.text, 0, 0);
    ctx.restore();
  });
}

  THUMB_STATE.layers[THUMB_STATE.sel] = styled;
  renderThumbLayers(); renderThumbCanvas();
}

function setThumbField(key, val){
  const L = THUMB_STATE.layers[THUMB_STATE.sel]; if (!L) return;
  if (key === 'outline_w'){ L.outline = L.outline || {color: '#000000', w: 0}; L.outline.w = +val; }
  else if (key === 'outline_color'){ L.outline = L.outline || {color: '#000000', w: 4}; L.outline.color = val; }
  else if (key === 'box'){ L.box = val ? {color: '#000000', pad: 16, opacity: 80} : null; }
  else if (key === 'size' || key === 'rot'){ L[key] = +val; }
  else L[key] = val;
  renderThumbCanvas();
}

// 드래그 = 비율 좌표를 바꾼다(픽셀이 아니라) — 그래야 1080 결과가 같은 자리에 찍힌다.
function initThumbDrag(){
  const cv = document.getElementById('thumbCanvas'); if (!cv || !cv.addEventListener) return;
  let on = false;
  const to = e => {
    const r = cv.getBoundingClientRect();
    return {x: (e.clientX - r.left) / r.width, y: (e.clientY - r.top) / r.height};
  };
  cv.addEventListener('mousedown', () => { on = true; });
  window.addEventListener('mouseup', () => { on = false; });
  cv.addEventListener('mousemove', e => {
    if (!on) return;
    const L = THUMB_STATE.layers[THUMB_STATE.sel]; if (!L) return;
    const p = to(e);
    L.x = Math.min(1, Math.max(0, p.x)); L.y = Math.min(1, Math.max(0, p.y));
    renderThumbCanvas();
  });
}

async function generateThumb(){
  const st = document.getElementById('thumbSaveStatus');
  st.textContent = '생성 중…';
  renderThumbCanvas();
  const cv = document.getElementById('thumbCanvas');
  const blob = await new Promise(res => cv.toBlob(res, 'image/png'));
  const fd = new FormData();
  fd.append('job_id', STATE.job_id);
  fd.append('meta', JSON.stringify({frame_ts: THUMB_STATE.frame_ts,
                                    frame_url: THUMB_STATE.frame_url,
                                    layers: THUMB_STATE.layers}));
  fd.append('file', blob, 'thumb.png');   // 이 이름은 서버가 무시한다(경로순회 방지)
  const r = await fetch('/api/produce/thumb/save', {method: 'POST', body: fd});
  const d = await r.json();
  if (!d.ok){ st.textContent = d.error || '저장 실패'; return; }
  st.textContent = '';
  THUMB_STATE.results.push({name: d.name, url: d.url});
  renderThumbGallery();
  selectThumbResult(d.name);
}

function renderThumbGallery(){
  document.getElementById('thumbGallery').innerHTML = THUMB_STATE.results.map(r => `
    <div onclick="selectThumbResult('${r.name}')" data-name="${r.name}"
         style="cursor:pointer;border-radius:6px;overflow:hidden;border:2px solid
         ${THUMB_STATE.selected === r.name ? 'var(--accent2)' : 'transparent'}">
      <img src="${r.url}" style="width:100%;display:block">
    </div>`).join('');
}

async function selectThumbResult(name){
  THUMB_STATE.selected = name;
  renderThumbGallery();
  await fetch('/api/produce/thumb/select', {
    method: 'POST', headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({job_id: STATE.job_id, name}),
  });
}
