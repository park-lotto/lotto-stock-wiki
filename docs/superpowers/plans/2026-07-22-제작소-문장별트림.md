# 제작소 문장별 트림 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 제작소 검토 페이지에서 문장(비트)마다 끝/앞의 조용한 부분(TTS 무음 꼬리)을 버튼으로 잘라, 영상·자막·음성이 한 번에 짧아지게 한다(캡컷 없이).

**Architecture:** 비트 dict에 `head_trim`/`tail_trim`(초)만 비파괴로 저장한다. 렌더 길이의 실질 단일 출처인 `_effective_dur(probe, head, tail)`를 `_beat_timeline`과 `_render_mix` 두 곳에서 공통으로 써, 트림값을 빼면 영상 conform·자막·다음 비트 t0·캡컷 export가 전부 자동으로 따라 줄어든다. 원본 TTS mp3는 안 건드려 되돌리기 = 값 0.

**Tech Stack:** Python 3, FastAPI, ffmpeg/ffprobe(`_run_ffmpeg`), pytest, 바닐라 JS(produce.html).

## Global Constraints

- 작업 폴더: **`.tracks/제작소트림`**. 모든 편집·git은 이 폴더 절대경로/`git -C`로. main 폴더 코드 금지.
- 시니어 UX: **슬라이더 금지, 버튼만**(메모리 `feedback_ttalkkak_senior_northstar`).
- 트림 하한 상수: `_TRIM_FLOOR = 0.4`(초). 미세조정 기본 스텝 `0.3`(초).
- 무음 감지 임계: ffmpeg `silencedetect=noise=-40dB:d=0.2`.
- 비파괴 원칙: 원본 `tts_path` mp3를 절대 덮어쓰지 않는다. 트림은 비트 dict의 `head_trim`/`tail_trim`만.
- 렌더 중(`status in {"rendering","removing_subtitles"}`) 트림 요청은 409 거부(regen 규율과 동일).
- 로컬 TTS 무음 mock 함정(메모리 `reference_local_tts_silent_mock_trap`): 실제 무음길이 검증은 config 키 확인 필요.

def _effective_dur(probe, head_trim=0.0, tail_trim=0.0, floor=_TRIM_FLOOR):
    """probe(원본 TTS 길이)에서 앞/뒤 트림을 뺀 실질 비트 길이. 하한 가드 포함.
    음수 트림은 0으로 취급. 결과가 floor 밑이면 floor로 고정한다."""
    d = probe - max(0.0, head_trim) - max(0.0, tail_trim)
    return max(floor, d)


def _beat_effective_dur(beat, tts_path):
    """비트 dict의 head_trim/tail_trim을 반영한 실질 길이(단일 출처)."""
    return _effective_dur(_probe_duration(tts_path),
                          beat.get("head_trim", 0.0), beat.get("tail_trim", 0.0))

```python
import re as _re
import subprocess as _subprocess


def _parse_silence_edges(stderr, total_dur):
    """silencedetect stderr → (앞무음초, 뒤무음초).
    앞무음 = silence_start≈0에서 시작한 구간의 end.
    뒤무음 = silence_end≈total_dur에서 끝난 구간의 duration(끝에 닿는 것만)."""
    starts = [float(m) for m in _re.findall(r"silence_start:\s*([0-9.]+)", stderr)]
    ends = _re.findall(r"silence_end:\s*([0-9.]+)\s*\|\s*silence_duration:\s*([0-9.]+)", stderr)
    head = 0.0
    tail = 0.0
    for s in starts:
        if s <= 0.05:            # 0에서 시작 = 앞무음
            # 짝지는 end 찾기(첫 end)
            if ends:
                head = float(ends[0][0])
            break
    for end_t, dur in ends:
        if abs(float(end_t) - total_dur) <= 0.05:   # 끝에 닿음 = 뒤무음
            tail = float(dur)
    return head, tail


def detect_edge_silence(path, edge):
    """path의 앞/뒤 무음 길이(초). edge in {"head","tail"}. 감지 실패 시 0.0."""
    from shopping_shorts.video_assemble import _probe_duration
    total = _probe_duration(path)
    if total <= 0:
        return 0.0
    proc = _subprocess.run(
        ["ffmpeg", "-i", str(path), "-af", "silencedetect=noise=-40dB:d=0.2",
         "-f", "null", "-"],
        capture_output=True, text=True)
    head, tail = _parse_silence_edges(proc.stderr or "", total)
    return head if edge == "head" else tail
```

```python
@app.post("/api/produce/mix/{job_id}/trim")
def api_produce_mix_trim(job_id: str, body: dict):
    """비트의 앞/뒤 조용한 부분을 자른다(비파괴 — head_trim/tail_trim만 저장).
    mode=auto: silencedetect로 그 엣지 무음 길이 산출. nudge: step 누적. reset: 0.
    렌더 중이면 409(regen 규율). 하한 가드: 트림 합이 (probe−_TRIM_FLOOR)를 못 넘는다."""
    store = Store(DB_PATH)
    job = store.get_mix_job(job_id)
    if not job:
        return JSONResponse(status_code=404, content={"ok": False, "error": "작업 없음"})
    if job.get("status") in ("rendering", "removing_subtitles"):
        return JSONResponse(status_code=409, content={"ok": False, "error": "렌더 중에는 자를 수 없어요"})
    plan = job.get("edit_plan") or {}
    beats = plan.get("beats") or []
    try:
        bi = int(body.get("beat_idx"))
    except (TypeError, ValueError):
        return JSONResponse(status_code=422, content={"ok": False, "error": "beat_idx 필요"})
    edge = body.get("edge")
    if edge not in ("tail", "head"):
        return JSONResponse(status_code=422, content={"ok": False, "error": "edge는 tail/head"})
    hit = next((b for b in beats if b.get("beat_idx") == bi), None)
    if hit is None:
        return JSONResponse(status_code=422, content={"ok": False, "error": "beat_idx 범위 밖"})
    key = "tail_trim" if edge == "tail" else "head_trim"
    mode = body.get("mode", "nudge")
    tts = hit.get("tts_path")
    if mode == "reset":
        hit[key] = 0.0
    elif mode == "auto":
        hit[key] = float(detect_edge_silence(tts, edge)) if tts else 0.0
    else:  # nudge
        step = float(body.get("step", 0.3))
        hit[key] = round(float(hit.get(key, 0.0)) + step, 3)
    # 하한 가드: head+tail이 probe−floor를 넘으면 이번 엣지를 되돌려 막는다.
    probe = _probe_duration(tts) if tts else 0.0
    if probe > 0:
        head, tail = hit.get("head_trim", 0.0), hit.get("tail_trim", 0.0)
        if _effective_dur(probe, head, tail) <= _TRIM_FLOOR and (head + tail) > (probe - _TRIM_FLOOR):
            hit[key] = max(0.0, round(probe - _TRIM_FLOOR - (head + tail - hit.get(key, 0.0)), 3))
    store.update_mix_job(job_id, edit_plan=plan)
    return {"ok": True, "head_trim": hit.get("head_trim", 0.0),
            "tail_trim": hit.get("tail_trim", 0.0), "trimmed": hit.get(key, 0.0)}
```
