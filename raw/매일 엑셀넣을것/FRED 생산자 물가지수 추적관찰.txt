import pandas as pd
import matplotlib.pyplot as plt
from pandas_datareader import data as pdr
import datetime

# 시작 및 종료 날짜 설정
start = datetime.datetime(2000, 1, 1)
end = datetime.datetime(2025, 1, 1)

# FRED에서 가져올 산업별 PPI 코드
ppi_codes = {
    # 반도체
    "Semiconductors & Related Devices": "PCU334413334413",
    
    # 자동차
    "Automobile Manufacturing": "PCU336111336111",
    "U.S. New Vehicles CPI": "CUSR0000SETA01",
    
    # 배터리 & 전기차
    "Primary Batteries (Dry & Wet)": "WPU1176",
    
    # 철강 & 원자재
    "Iron & Steel Mills & Ferroalloys": "PCU331110331110",
    
    # 변압기 및 네트워크 장비
    "Power & Specialty Transformers": "PCU335311335311P",
    "Power & Distribution Transformers": "WPU117409",
    "Other Transformers & Parts": "WPU117411",
    "Transformers & Power Regulators": "WPU1174",
    "Communications Equipment": "PCU334220334220"
}

# 데이터 가져오기
ppi_data = {}
for label, code in ppi_codes.items():
    try:
        ppi_data[label] = pdr.get_data_fred(code, start, end)
    except Exception as e:
        print(f"데이터 가져오기 실패: {label} ({code}) - {e}")

# 개별 그래프 및 변화율 출력
for label, df in ppi_data.items():
    if df.empty:
        continue

    # 최근 데이터와 직전달 데이터 가져오기
    latest_date = df.dropna().index[-1]
    prev_date = df.dropna().index[-2]
    latest_value = df.iloc[-1, 0]
    prev_value = df.iloc[-2, 0]

    # 변화율 계산
    change_rate = ((latest_value - prev_value) / prev_value) * 100

    # 그래프 그리기
    plt.figure(figsize=(10, 5))
    plt.plot(df.index, df[df.columns[0]], label=label, color='blue')

    plt.xlabel("Year")
    plt.ylabel("PPI Index")
    plt.title(f"PPI Trend: {label}")
    plt.legend()
    plt.grid()
    plt.show()

    # 변화율 출력
    print(f"📊 {label} PPI 변화율")
    print(f"  - 최근 데이터 ({latest_date.date()}): {latest_value:.2f}")
    print(f"  - 직전달 데이터 ({prev_date.date()}): {prev_value:.2f}")
    print(f"  - 변화율: {change_rate:.2f}%\n")
