import yfinance as yf
import pandas as pd
import numpy as np

# =========================================================
# 0-1. 현금 비중 입력 🔥
# =========================================================
cash_weight = float(input("현금 비중 입력 (예: 0.3 = 30%): "))
stock_weight = 1 - cash_weight

# =========================================================
# 1. 티커
# =========================================================
tickers = [
    "000150.KS", "101490.KQ", "353200.KS", "425420.KQ",
    "050890.KQ", "230240.KQ", "138080.KQ", "032500.KQ",
    "010170.KQ", "100790.KQ", "347700.KQ", "189300.KQ",
    "001430.KS", "295310.KQ", "375500.KS", "226950.KQ",
    "475830.KQ", "174900.KQ", "298380.KQ", "115180.KQ",
]
start_date = "2020-01-01"

# =========================================================
# 2. 데이터
# =========================================================
data = yf.download(tickers, start=start_date)["Close"]

data = data.ffill()
data = data.dropna()

returns = data.pct_change().dropna()

# =========================================================
# 3. 멀티 모멘텀
# =========================================================
mom_3 = data.pct_change(63)
mom_6 = data.pct_change(126)
mom_12 = data.pct_change(252)

momentum = (mom_3*0.4 + mom_6*0.3 + mom_12*0.3)
mu = momentum.iloc[-1].dropna()

mu = np.tanh(mu)

# =========================================================
# 4. 변동성
# =========================================================
vol = returns.rolling(60).std().iloc[-1]
vol = np.sqrt(vol)

# =========================================================
# 5. 점수 계산
# =========================================================
score = mu / vol
score = score - score.min()
score = score.replace([np.inf, -np.inf], 0)

# =========================================================
# 6. 순위 기반 점수
# =========================================================
rank = score.rank(ascending=False)
rank_score = (len(rank) - rank + 1)

# =========================================================
# 7. 최소 비중
# =========================================================
min_weight = 0.03
n = len(rank_score)

base_weight = pd.Series(min_weight, index=rank_score.index)
remaining = 1 - (min_weight * n)

# =========================================================
# 8. 추가 비중
# =========================================================
rank_score = rank_score / rank_score.sum()
extra_weight = rank_score * remaining

weights = base_weight + extra_weight

# =========================================================
# 9. 최대 비중 제한
# =========================================================
max_weight = 0.10
exception = ["005930.KS", "000660.KS"]

weights_adj = weights.copy()

for t in weights_adj.index:
    if t not in exception:
        weights_adj[t] = min(weights_adj[t], max_weight)

remaining = 1 - weights_adj.sum()

alloc_targets = [t for t in weights_adj.index if t in exception]

if len(alloc_targets) == 0:
    alloc_targets = weights_adj.index.tolist()

if remaining > 0:
    base = weights_adj[alloc_targets]

    if base.sum() > 0:
        base = base / base.sum()
    else:
        base = pd.Series(1/len(alloc_targets), index=alloc_targets)

    weights_adj[alloc_targets] += base * remaining

weights = weights_adj

# =========================================================
# 10. 재정규화
# =========================================================
weights = weights / weights.sum()

# =========================================================
# 11. 현금 반영 🔥
# =========================================================
weights = weights * stock_weight

# =========================================================
# 12. 결과
# =========================================================
result = pd.DataFrame({
    "Momentum": mu,
    "Volatility": vol,
    "Score": score,
    "Rank": rank,
    "Weight": weights
}).sort_values(by="Weight", ascending=False)

# =========================================================
# 13. 종목명 추가
# =========================================================
names = []
for t in result.index:
    try:
        ticker_obj = yf.Ticker(t)
        name = ticker_obj.fast_info.get("shortName", "")
        if not name:
            name = ticker_obj.info.get("shortName", t)
    except:
        name = t
    names.append(name)

result["Name"] = names

# 퍼센트 변환
result["Weight_%"] = (result["Weight"] * 100).round(1)

final = result[["Name", "Weight_%"]]

# =========================================================
# 14. 출력
# =========================================================
print("\n📊 최종 포트폴리오 (현금 포함)\n")
print(final.sort_values(by="Weight_%", ascending=False))

print(f"\n현금 비중: {round(cash_weight*100,1)}%")
print(f"주식 비중: {round(stock_weight*100,1)}%")

print("\n총합:", round(final["Weight_%"].sum() + cash_weight*100, 1), "%")
