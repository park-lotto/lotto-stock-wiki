# 라이브러리 불러오기
import pandas as pd
import yfinance as yf
from datetime import datetime

# pandas 출력 옵션
pd.set_option('display.max_rows', None)
pd.set_option('display.max_columns', None)
pd.set_option('display.width', 2000)
pd.set_option('display.max_colwidth', None)

# ----------------------------
# 1) 티커/자산 설정
# ----------------------------
risky_assets = {
    "418660.KS": "TIGER 미국나스닥100레버리지(합성)",
    "225040.KS": "TIGER 미국S&P500레버리지(합성 H)",
    "280930.KS": "KODEX 미국러셀2000(H)",
    "241180.KS": "TIGER 일본니케이225",
    "251350.KS": "KODEX 선진국MSCI World",
    "261220.KS": "KODEX WTI원유선물(H)",
    "182480.KS": "TIGER 미국MSCI리츠(합성 H)",
    "229200.KS": "KODEX 코스닥 150",
    "069500.KS": "KODEX 200",
}

safe_assets = {
    "304660.KS": "KODEX 미국채울트라30년선물(H)",
    "305080.KS": "TIGER 미국채10년선물",
    "148070.KS": "KOSEF 국고채10년",
    "261240.KS": "KODEX 미국달러선물",
    "130730.KS": "KOSEF 단기자금"
}

# TIP (카나리아 자산)
canary_asset = {"TIP": "TIP 물가연동채"}

# 고정 자산 (예: 총 35%)
fixed_assets = {
    "225040.KS": 10,  # TIGER 미국S&P500레버리지(합성 H)
    "418660.KS": 10,  # TIGER 미국나스닥100레버리지(합성)
    "411060.KS": 5,   # ACE KRX금현물
    "261220.KS": 10,  # KODEX WTI원유선물(H)
}

# ----------------------------
# 2) 사용자 입력
# ----------------------------
specific_date = input("투자일(YYYY-MM-DD)을 입력하세요: ")
# 예) 2023-11-10 (월중 가능)
try:
    target_dt = datetime.strptime(specific_date, '%Y-%m-%d')
except ValueError:
    print("[오류] 날짜 형식이 잘못되었습니다. (YYYY-MM-DD)")
    exit()

# ----------------------------
# 3) 데이터 다운로드 (과거 3년) + 그 날짜까지 필터링
# ----------------------------
end_date = target_dt.strftime('%Y-%m-%d')
start_date = (target_dt - pd.DateOffset(years=3)).strftime('%Y-%m-%d')

tickers = (
    list(risky_assets.keys())
    + list(safe_assets.keys())
    + list(canary_asset.keys())
    + list(fixed_assets.keys())
)

print(f"\n[INFO] {start_date} ~ {end_date} 기간 데이터 다운로드 중...")
data = yf.download(tickers, start=start_date, end=end_date)['Close']

# 혹시 중복열 제거
data = data.loc[:, ~data.columns.duplicated()].copy()

# 사용자가 입력한 날짜를 초과하는 데이터가 있을 수 있으므로
data = data.loc[:specific_date]

# ----------------------------
# 4) 월말 데이터로 리샘플링
# ----------------------------
monthly_data = data.resample('M').last()

# ----------------------------
# 5) 모멘텀 계산 함수
# ----------------------------
def calculate_momentum(series):
    """
    월말 종가에 대해 1M, 3M, 6M, 12M 전 대비 수익률을 평균
    """
    df = pd.DataFrame(index=series.index)
    df['1M']  = series.pct_change(periods=1)
    df['3M']  = series.pct_change(periods=3)
    df['6M']  = series.pct_change(periods=6)
    df['12M'] = series.pct_change(periods=12)
    return df.mean(axis=1)

# ----------------------------
# 6) 카나리아 신호
# ----------------------------
def canary_momentum(tip_series):
    tip_mom = calculate_momentum(tip_series)
    return (tip_mom > 0), tip_mom

# ----------------------------
# 7) 자산 선택 함수
# ----------------------------
def select_safe_assets(momentum_df, bil_momentum, top_n=2):
    safe_keys = list(safe_assets.keys())
    recent_mom = momentum_df[safe_keys].iloc[-1]  # 최근(마지막) 월말
    top_safe = recent_mom.nlargest(top_n)

    selected = []
    for asset in top_safe.index:
        if top_safe[asset] > bil_momentum:
            selected.append(asset)
    if not selected:
        return ['130730.KS']
    if '130730.KS' not in selected:
        selected.append('130730.KS')
    return selected[:top_n]

def select_top_assets(momentum_df, canary_signal_series, top_n=4):
    """
    canary_signal_series.iloc[-1] == True -> 공격자산
    else -> 안전자산
    """
    if canary_signal_series.iloc[-1]:
        # 공격
        risky_keys = list(risky_assets.keys())
        recent_mom = momentum_df[risky_keys].iloc[-1]
        top_assets = recent_mom.nlargest(top_n)
        return top_assets.index.tolist()
    else:
        # 방어
        if '130730.KS' in data.columns:
            bil_mom = calculate_momentum(data['130730.KS']).iloc[-1]
        else:
            bil_mom = 0
        return select_safe_assets(momentum_df, bil_mom, top_n=2)

def normalize_weights(weights_dict, total=100):
    """
    정수화 후, 합이 total!=100이면 최댓값에 조정
    """
    from math import floor, ceil

    rounded = {k: round(v) for k, v in weights_dict.items()}
    diff = total - sum(rounded.values())
    if diff != 0 and len(rounded) > 0:
        max_key = max(rounded, key=rounded.get)
        rounded[max_key] += diff
    return rounded

# ----------------------------
# 8) "현재 시점" 기준 추천 자산 계산
# ----------------------------
if len(monthly_data) < 2:
    print("\n[오류] 월말 데이터가 2개월분 이하입니다. 모멘텀 계산이 어렵습니다.")
    exit()

# 1) 모멘텀 DataFrame
momentum_df = monthly_data.apply(calculate_momentum)

# 2) TIP 모멘텀, 카나리아 신호
if 'TIP' not in monthly_data.columns:
    print("\n[TIP] 데이터가 부족합니다. 카나리아 신호를 계산할 수 없습니다.")
    exit()

canary_signal_series, tip_mom_series = canary_momentum(monthly_data['TIP'])
latest_canary = canary_signal_series.iloc[-1]  # 최근 월말 TIP 모멘텀 > 0 여부

# 3) 자산 선정
top_assets = select_top_assets(momentum_df, canary_signal_series, top_n=4)
if len(top_assets) == 0:
    top_assets = ['130730.KS']

# 4) 고정자산 (35%) + 변동자산 (65%)
haa_share = 65
share_each = haa_share / len(top_assets)
combined_weights = dict(fixed_assets)
for asset in top_assets:
    combined_weights[asset] = combined_weights.get(asset, 0) + share_each

final_weights = normalize_weights(combined_weights, total=100)
sorted_weights = sorted(final_weights.items(), key=lambda x: x[1], reverse=True)

# ----------------------------
# 9) 결과 출력
# ----------------------------
print(f"\n[투자일: {specific_date}] 기준 추천 포트폴리오")
print(f"카나리아 신호: {latest_canary} (TIP 모멘텀이 0 이상이면 True)")
print("\n■ 선정된 변동자산:")
for asset in top_assets:
    # 자산명(한글) 붙이기
    if asset in risky_assets:
        asset_name = risky_assets[asset]
    elif asset in safe_assets:
        asset_name = safe_assets[asset]
    else:
        asset_name = asset
    print(f"  - {asset}: {asset_name}")

print("\n■ 최종 종목 비중(정수합=100):")
for k, v in sorted_weights:
    print(f"  {k}: {v}%")

print("\n프로그램 종료.")
