# tools/calcopy/engine/agy.py
# -*- coding: utf-8 -*-
"""agy(구독 제미나이 CLI) 창구 — 한 곳(관제 319 D11). 영상 파일을 통째로 보여 주고 칸이 정해진 JSON 을 받는다.
명령 실행 권한은 주지 않는다(--mode plan). 실패는 조용히 넘기지 않고 error 로 돌려 서버가 멈추게 한다."""
from __future__ import annotations

import json
import os
import shutil
import subprocess
import time
from pathlib import Path

RETRIES = 1


def find() -> str | None:
    p = shutil.which("agy")
    if p:
        return p
    cand = Path(os.environ.get("LOCALAPPDATA", "")) / "agy" / "bin" / "agy.exe"
    return str(cand) if cand.exists() else None


def _call(args: list[str], run, timeout: int) -> dict:
    r = (run or subprocess.run)(args, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=timeout)
    try:
        return json.loads(r.stdout or "{}")
    except ValueError:
        return {"status": "BAD_OUTPUT", "response": (r.stdout or "")[:300]}


def logged_in(run=None) -> bool:
    exe = find()
    if not exe:
        return False
    out = _call([exe, "--output-format", "json", "--mode", "plan", "-p", 'JSON 하나로만 답: {"ok": true}'], run, 120)
    return out.get("status") == "SUCCESS"


def _json_of(text: str):
    t = (text or "").strip()
    if t.startswith("```"):
        t = t.strip("`").split("\n", 1)[-1]
    try:
        v = json.loads(t)
        return v if isinstance(v, dict) else None
    except ValueError:
        return None


def ask_video(video: Path, ask: str, schema: dict, run=None, timeout: int = 600) -> dict:
    exe = find()
    if not exe:
        return {"error": "agy_missing"}
    video = Path(video)
    prompt = (f"이 폴더의 {video.name} 영상 파일을 통째로 보고 들어라(장면 그림으로 자르지 말고, 명령어는 쓰지 마라). "
              f"{ask}\n답은 JSON 하나만.")
    args = [exe, "--output-format", "json", "--mode", "plan", "--json-schema", json.dumps(schema, ensure_ascii=False),
            "--add-dir", str(video.parent), "-p", prompt]
    t0 = time.time()
    for _ in range(RETRIES + 1):
        out = _call(args, run, timeout)
        ans = _json_of(out.get("response") or "") if out.get("status") == "SUCCESS" else None
        if ans is not None:
            return {"answer": ans, "sec": round(time.time() - t0, 1)}
    return {"error": f"agy_bad_answer:{out.get('status')}"}
```

# tools/calcopy/hybrid/jev_gate.py
# -*- coding: utf-8 -*-
"""서버 Jev 창구 — 회원 엔진의 결정(후보 고르기)은 여기서만 Jev 에 묻는다(관제 319 D7, 전역 0원칙).
키 = 서버 환경 TYPESAFE_API_KEY. 회원 묶음에 이 파일·키 이름이 들어가면 benchmark_pack.check 가 막는다."""
from __future__ import annotations

import os

CONF_OK = 0.7


class JevDown(Exception):
    pass


def _client():
    if not os.environ.get("TYPESAFE_API_KEY"):
        raise JevDown("no_key")
    try:
        from typesafe_sdk import TypeSafeClient
    except ImportError as e:
        raise JevDown("no_sdk") from e
    return TypeSafeClient(model="jev-latest")


def decide(problem: str, facts: list[str], fixes: list[str], client=None) -> dict:
    if len(fixes) < 2:
        raise ValueError("후보는 2개 이상")
    from typesafe_sdk import Choice, Noul  # noqa: F401 — 실제 클라이언트일 때만 필요
    state = {"problem": problem, "facts": list(facts)} | {f"fix_{i}": f for i, f in enumerate(fixes)}
    Q = {f"fixroot_{i}": Noul(instructions=f"`fix_{i}` removes the root cause of `problem` for all future cases (not a one-off patch)") for i in range(len(fixes))}
    Q["best_fix"] = Choice(instructions="Which candidate fix best removes the root cause of `problem` with the least risk",
                           criteria={f"fix_{i}": f[:300] for i, f in enumerate(fixes)} | {"none": "None of the candidates"})
    own = client is None
    cl = _client() if own else client
    try:
        r = (cl.__enter__() if own else cl).system_one(state=state, questions=Q)
    except JevDown:
        raise
    except Exception as e:  # noqa: BLE001 — 결정 없이 넘어가지 않는다: 부르는 쪽이 멈춘다
        raise JevDown(type(e).__name__) from e
    finally:
        if own:
            cl.__exit__(None, None, None)
    a = r.answers
    best = a["best_fix"]
    pick = None if best.choice == "none" else int(str(best.choice).split("_")[1])
    conf = round(float(best.confidence), 2)
    return {"pick": pick, "conf": conf, "fixroot": [round(float(a[f"fixroot_{i}"].noul), 2) for i in range(len(fixes))],
            "ok": pick is not None and conf >= CONF_OK}
```

시험의 `FakeClient`는 `typesafe_sdk`가 없는 PC에서도 돌아야 하므로, `Choice`·`Noul` import를 `try`로 감싸 없으면 `dict`로 대신하는 작은 대체를 둔다:

```python
try:
    from typesafe_sdk import Choice, Noul
except ImportError:  # 시험·SDK 없는 PC — 질문 꼴만 만든다(실제 호출은 _client 가 JevDown)
    def Choice(**kw): return {"type": "choice", **kw}

```python
# tools/calcopy/engine/preset_brief.py
# -*- coding: utf-8 -*-
"""프리셋 정본(frame.json·manifest.json) → 읽기용 프리셋.md·CLAUDE.md (관제 319 D4). 렌더는 이 md 를 읽지 않는다.
머리 첫 줄 표식 = 두 파일의 md5. 서버 bm_make 가 표식이 다르면 다시 그리게 지시한다(D10)."""
from __future__ import annotations

import hashlib
import json
from pathlib import Path

MARK = "<!-- manifest_sha: {} -->"


def sha(preset_dir: Path) -> str:
    h = hashlib.md5()
    for n in ("frame.json", "manifest.json"):
        h.update((Path(preset_dir) / n).read_bytes())
    return h.hexdigest()


def _flat(d: dict, pre: str = "") -> list[tuple[str, object]]:
    out = []
    for k, v in (d or {}).items():
        key = f"{pre}{k}"
        out += _flat(v, key + ".") if isinstance(v, dict) else [(key, v)]
    return out


def render(preset_dir: Path, out_dir: Path, name: str) -> dict:
    preset_dir, out_dir = Path(preset_dir), Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    frame = json.loads((preset_dir / "frame.json").read_text(encoding="utf-8"))
    man = json.loads((preset_dir / "manifest.json").read_text(encoding="utf-8"))
    var = set(man.get("variables") or [])
    s = sha(preset_dir)
    lines = [MARK.format(s), f"# 내 프리셋 — {name}", "", f"쟀던 벤치 편: {len(man.get('sources') or [])}편", "",
             "| 칸 | 값 | 어떻게 |", "|---|---|---|"]
    for k, v in _flat(frame):
        lines.append(f"| {k} | {json.dumps(v, ensure_ascii=False)[:60]} | {'편마다 소재에 맞춰 정함' if k in var else '고정'} |")
    md = out_dir / "프리셋.md"
    md.write_text("\n".join(lines) + "\n", encoding="utf-8")
    cm = out_dir / "CLAUDE.md"
    cm.write_text(MARK.format(s) + f"\n# {name} 프리셋으로 만들 때 지킬 것\n\n- 만들기 전에 프리셋.md 를 끝까지 읽는다.\n"
                  "- '고정' 칸은 바꾸지 않는다. '편마다' 칸만 이 소재에 맞춰 정한다.\n- 프리셋.md 에 없는 규칙은 지어내지 않는다.\n", encoding="utf-8")
    return {"md": str(md), "claude_md": str(cm), "sha": s}


def stale(md_path: Path, preset_dir: Path) -> bool:
    first = Path(md_path).read_text(encoding="utf-8").splitlines()[:1]
    return not first or first[0] != MARK.format(sha(preset_dir))
```
