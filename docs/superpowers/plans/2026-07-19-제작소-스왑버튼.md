def api_mix_seg_thumb(job_id: str, seg_id: str):
    """[다른 화면으로] 피커용 seg 썸네일(중간지점 프레임 jpg, 캐시).
    seg_id는 인벤토리 검증 후에만 파일명에 쓴다(경로 방어)."""
    job = Store(DB_PATH).get_mix_job(job_id)

        return JSONResponse(status_code=404, content={"ok": False, "error": "데이터 없음"})
    seg_map, _ = _edit_plan._build_inventory(list(job["extract"].values()))
    seg = seg_map.get(seg_id)
    if not seg:
        return JSONResponse(status_code=404, content={"ok": False, "error": "없는 seg_id"})
    work = _MIX_WORK_DIR / job_id
    cached = work / "seg_thumbs" / f"{seg_id}.jpg"
    if not cached.exists():
        try:
            src = _resolve_sources(job, work)[seg["video_id"]]
        except Exception:
            return JSONResponse(status_code=404, content={"ok": False, "error": "소스 없음"})
        mid = (seg["start"] + seg["end"]) / 2
        frame = extract_frame_at(src, work / "seg_thumbs", mid, filename=f"{seg_id}.jpg")
        if not frame:
            return JSONResponse(status_code=404, content={"ok": False, "error": "프레임 추출 실패"})
    return FileResponse(str(cached), media_type="image/jpeg")

function renderSwapButton(beat, idx){
  const weak = beat.fit && beat.fit <= 2;
  const style = weak
    ? 'padding:8px 14px;font-size:14px;border:2px solid var(--fire);border-radius:8px;background:#1a0d0d;color:#fff;margin-top:6px'
    : 'padding:6px 12px;font-size:12px;border:1px solid var(--line);border-radius:8px;background:#0a1018;margin-top:6px';
  return `<button style="${style}" onclick="openSwapPicker(${idx})">🎞 다른 화면으로 바꾸기</button>`;
}

async function openSwapPicker(idx){
  if(!MIX_JOB) return;
  const box = document.getElementById('sceneSwap'+idx);
  if(box) box.innerHTML = '<div class="hint">⏳ 화면 목록 불러오는 중…</div>';
  let d;
  try{ d = await (await fetch('/api/mix/segments/'+encodeURIComponent(MIX_JOB))).json(); }
  catch(e){ if(box) box.innerHTML='<div class="hint" style="color:var(--fire)">네트워크 오류</div>'; return; }
  if(!d.ok){ if(box) box.innerHTML='<div class="hint" style="color:var(--fire)">목록 실패</div>'; return; }
  const cards = (d.segments||[]).map(s=>`
    <div onclick="doSwap(${idx},'${esc(s.seg_id)}')" style="flex:0 0 96px;cursor:pointer;opacity:${s.used?0.4:1}">
      <img src="${esc(s.thumb_url)}" style="width:96px;height:170px;object-fit:cover;border-radius:6px;background:#000" alt="" onerror="this.style.visibility='hidden'">

      <div class="hint" style="font-size:10px">${s.dur}s${s.used?' · 사용중':''}</div>
    </div>`).join('');
  if(box) box.innerHTML =
    `<div class="hint" style="margin:4px 0">아래 화면 중 하나를 <b>누르면</b> 이 문장에 그 화면이 들어갑니다.</div>`

async function doSwap(idx, segId){
  if(!MIX_JOB) return;
  const box = document.getElementById('sceneSwap'+idx);
  if(box) box.innerHTML = '<div class="hint">⏳ 화면 바꾸는 중…</div>';

      method:'POST', headers:{'Content-Type':'application/json'},
