import yfinance as yf
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# 2. NASDAQ 100 티커 가져오기
def get_nasdaq100_tickers():
    url = 'https://en.wikipedia.org/wiki/NASDAQ-100'
    tables = pd.read_html(url)
    table = tables[4]
    tickers = table['Ticker'].tolist()
    tickers = [ticker.replace('.', '-') for ticker in tickers]
    return tickers

# 3. S&P 100 티커 가져오기
def get_sp100_tickers():
    url = 'https://en.wikipedia.org/wiki/S%26P_100'
    tables = pd.read_html(url)
    table = tables[2]
    tickers = table['Symbol'].tolist()
    tickers = [ticker.replace('.', '-') for ticker in tickers]
    return tickers

# 4. NASDAQ 100 + S&P 100 합치기 + 중복 제거
nasdaq100 = get_nasdaq100_tickers()
sp100 = get_sp100_tickers()
combined_tickers = list(set(nasdaq100 + sp100))

# 5. 데이터 다운로드 (10년)
extra_tickers = ['QQQ', 'SPY', 'PSQ', 'IAU', 'BIL', 'SPMO']
all_tickers = list(set(combined_tickers + extra_tickers))

data = yf.download(all_tickers, period="10y", auto_adjust=False)['Close']
data = data.dropna(thresh=len(data) * 0.8, axis=1)
data = data.drop(columns=[ticker for ticker in extra_tickers if ticker not in data.columns])

available_tickers = [ticker for ticker in combined_tickers if ticker in data.columns]

# 6. 월말 가격 계산
monthly_prices = data.resample('ME').last()

# 7. 카나리아 신호 (QQQ, SPMO 각각)
qqq_prices = data['QQQ']
qqq_ma5 = qqq_prices.rolling(window=5).mean()
qqq_ma10 = qqq_prices.rolling(window=10).mean()
qqq_ma20 = qqq_prices.rolling(window=20).mean()

qqq_check = pd.DataFrame(index=monthly_prices.index)
qqq_check['Price'] = qqq_prices.resample('ME').last()
qqq_check['MA5'] = qqq_ma5.resample('ME').last()
qqq_check['MA10'] = qqq_ma10.resample('ME').last()
qqq_check['MA20'] = qqq_ma20.resample('ME').last()

qqq_check['Canary_OK'] = (
    (qqq_check['Price'] > qqq_check['MA5']) |
    (qqq_check['Price'] > qqq_check['MA10']) |
    (qqq_check['Price'] > qqq_check['MA20'])
)

spmo_prices = data['SPMO']
spmo_ma5 = spmo_prices.rolling(window=5).mean()
spmo_ma10 = spmo_prices.rolling(window=10).mean()
spmo_ma20 = spmo_prices.rolling(window=20).mean()

spmo_check = pd.DataFrame(index=monthly_prices.index)
spmo_check['Price'] = spmo_prices.resample('ME').last()
spmo_check['MA5'] = spmo_ma5.resample('ME').last()
spmo_check['MA10'] = spmo_ma10.resample('ME').last()
spmo_check['MA20'] = spmo_ma20.resample('ME').last()

# ★ 수정: SPMO는 10일선 + 20일선만 체크
spmo_check['Canary_OK'] = (
    (spmo_check['Price'] > spmo_check['MA10']) |
    (spmo_check['Price'] > spmo_check['MA20'])
)

# 8. 모멘텀 점수 계산
def calculate_weighted_momentum_score(prices):
    returns_3m = prices.pct_change(3)
    returns_6m = prices.pct_change(6)
    returns_12m = prices.pct_change(12)
    weighted_returns = (returns_12m * 4) + (returns_6m * 3) + (returns_3m * 2)
    return weighted_returns

momentum = calculate_weighted_momentum_score(monthly_prices[available_tickers])
momentum = momentum.shift(1)

# 9. 전략 수익률 계산 + 월별 포트 기록
strategy_returns = []
dates = []
portfolio_records = {}
monthly_returns = {}

for i in range(len(momentum) - 1):
    date = momentum.index[i]
    next_date = momentum.index[i + 1]

    if date not in qqq_check.index:
        continue

    canary_ok = qqq_check.loc[date, 'Canary_OK']
    spmo_canary_ok = spmo_check.loc[date, 'Canary_OK']

    current_momentum = momentum.iloc[i].dropna()
    if current_momentum.empty:
        continue

    try:
        top2 = current_momentum.sort_values(ascending=False).head(2).index.tolist()
        valid_top2 = [ticker for ticker in top2 if ticker in monthly_prices.columns]
        if len(valid_top2) < 2:
            continue

        assets = valid_top2 + ['QQQ', 'SPY', 'IAU', 'BIL', 'PSQ']
        current_prices = monthly_prices.loc[date, assets]
        next_prices = monthly_prices.loc[next_date, assets]

        individual_returns = (next_prices / current_prices) - 1
        top2_returns = individual_returns[valid_top2]
        qqq_return = individual_returns['QQQ']
        spy_return = individual_returns['SPY']
        iau_return = individual_returns['IAU']
        bil_return = individual_returns['BIL']
        psq_return = individual_returns['PSQ']

        if canary_ok:
            if not spmo_canary_ok:
                # QQQ 공격인데 SPMO 방어 → QQQ 단독 사용
                monthly_return = (
                    0.7 * qqq_return +
                    0.10 * spy_return +
                    0.10 * iau_return +
                    0.10 * bil_return
                )
                portfolio_records[date.strftime('%Y-%m')] = ['QQQ', 'SPY', 'IAU', 'BIL']
            else:
                # QQQ, SPMO 모두 공격 → 기존 Top2 전략
                monthly_return = (
                    0.35 * top2_returns.iloc[0] +
                    0.35 * top2_returns.iloc[1] +
                    0.10 * spy_return +
                    0.10 * iau_return +
                    0.10 * bil_return
                )
                portfolio_records[date.strftime('%Y-%m')] = valid_top2 + ['SPY', 'IAU', 'BIL']
        else:
            # 방어 모드
            monthly_return = (
                0.5 * psq_return +
                0.10 * iau_return +
                0.10 * bil_return +
                0.15 * top2_returns.iloc[0] +
                0.15 * top2_returns.iloc[1]
            )
            portfolio_records[date.strftime('%Y-%m')] = ['PSQ', 'IAU', 'BIL'] + valid_top2

    except:
        continue

    strategy_returns.append(monthly_return)
    dates.append(next_date)
    monthly_returns[date.strftime('%Y-%m')] = monthly_return

# 10. 누적 수익률 계산
strategy_returns = pd.Series(strategy_returns, index=dates)
strategy_cum_returns = (1 + strategy_returns).cumprod()

# 11. SPY 누적 수익률 계산
spy_prices_monthly = monthly_prices['SPY']
spy_returns = spy_prices_monthly.pct_change().dropna()
spy_cum_returns = (1 + spy_returns).cumprod()

start_date = strategy_cum_returns.index[0]
spy_cum_returns = spy_cum_returns.loc[start_date:]
spy_cum_returns = spy_cum_returns / spy_cum_returns.iloc[0]
aligned_strategy = strategy_cum_returns.reindex(spy_cum_returns.index, method='ffill')

# 12. 전략 vs SPY 누적 수익률 그래프
plt.figure(figsize=(12, 6))
plt.plot(aligned_strategy, label='Momentum + Top2 + SPY + IAU + BIL (+SPMO 10/20)')
plt.plot(spy_cum_returns, label='SPY Buy & Hold')
plt.title('Momentum Strategy (NASDAQ100 + S&P100) - QQQ & SPMO 10/20 Logic')
plt.xlabel('Date')
plt.ylabel('Cumulative Return')
plt.legend()
plt.grid(True)
plt.show()

# 13. CAGR, MDD 계산
def calculate_cagr(cum_returns):
    total_return = cum_returns.iloc[-1]
    years = len(cum_returns) / 12
    cagr = (total_return) ** (1 / years) - 1
    return cagr

def calculate_mdd(cum_returns):
    roll_max = cum_returns.cummax()
    drawdown = (cum_returns / roll_max) - 1
    mdd = drawdown.min()
    return mdd

strategy_cagr = calculate_cagr(strategy_cum_returns)
strategy_mdd = calculate_mdd(strategy_cum_returns)

# 14. 전략 성과 출력
print(f"Strategy CAGR: {strategy_cagr:.2%}")
print(f"Strategy MDD: {strategy_mdd:.2%}")

# 15. 최근 12개월 포트폴리오 및 수익률 출력
print("\nRecent 12 months Portfolio:")
recent_portfolios = pd.Series(portfolio_records).iloc[-12:]
recent_returns = pd.Series(monthly_returns).iloc[-12:]

for month, portfolio in recent_portfolios.items():
    ret = recent_returns.get(month, np.nan)
    if pd.notna(ret):
        print(f"{month}: {', '.join(portfolio)} | Return: {ret:.2%}")
    else:
        print(f"{month}: {', '.join(portfolio)} | Return: No data")

# 16. 다음달 추천 포트폴리오
print("\nNext Month Recommended Portfolio:")

latest_date = momentum.index[-1]

if latest_date in qqq_check.index:
    canary_ok = qqq_check.loc[latest_date, 'Canary_OK']
    spmo_canary_ok = spmo_check.loc[latest_date, 'Canary_OK']

    latest_momentum = momentum.iloc[-1].dropna()
    if not latest_momentum.empty:
        top2_next_month = latest_momentum.sort_values(ascending=False).head(2).index.tolist()

        if canary_ok:
            if not spmo_canary_ok:
                print(f"Portfolio: QQQ 70%, SPY 10%, IAU 10%, BIL 10%")
            else:
                print(f"Portfolio: {', '.join(top2_next_month)}, SPY, IAU, BIL (Top2 each 35%, SPY 10%, IAU 10%, BIL 10%)")
        else:
            print(f"Portfolio: PSQ 50%, IAU 10%, BIL 10%, {', '.join(top2_next_month)} (Top2 each 15%)")
    else:
        print("No data available to recommend.")
else:
    print("No canary data available for next month.")
