# 라이브러리 임포트
import pandas as pd
import yfinance as yf
import matplotlib.pyplot as plt

# 사용자로부터 티커 목록 입력받기
tickers = input("분석할 주식 티커들을 쉼표로 구분하여 입력하세요 (예: ^GSPC,AAPL,GOOGL): ").split(',')

# 각 티커에 대해 분석
for ticker in tickers:
    ticker = ticker.strip()  # 공백 제거
    print(f"\n===== {ticker} 분석 시작 =====")
    
    # Yahoo Finance 데이터를 가져오기 (일 단위 데이터)
    data = yf.download(ticker, start="2000-01-01", end=None, interval="1d")
    if data.empty:
        print(f"티커 {ticker}의 데이터를 가져오지 못했습니다.")
        continue
    
    data['Close'] = data['Adj Close']
    
    # 최신 데이터 날짜
    last_date_index = data.index[-1]
    last_date = last_date_index.strftime('%Y-%m-%d')
    print(f"가장 최신 데이터 날짜: {last_date}")
    
    # 월 단위 데이터로 변환 (최신 데이터 포함)
    data_monthly = data.resample('M').last()
    data_monthly.loc[last_date_index] = data.iloc[-1]  # 최신 일 데이터를 직접 추가
    data_monthly.sort_index(inplace=True)
    
    # 1. 기존 코폭 지표 계산
    data_monthly['ROC_14'] = data_monthly['Close'].pct_change(14) * 100  # 14개월 ROC
    data_monthly['ROC_11'] = data_monthly['Close'].pct_change(11) * 100  # 11개월 ROC
    data_monthly['Coppock'] = (data_monthly['ROC_14'] + data_monthly['ROC_11']).ewm(span=10, adjust=False).mean()  # 10개월 가중 이동평균
    
    # 코폭 지표가 적용된 마지막 날짜 추출
    last_coppock_date = data_monthly['Coppock'].dropna().index[-1]
    last_coppock_date_str = last_coppock_date.strftime('%Y-%m-%d')
    last_coppock_value = data_monthly['Coppock'].loc[last_coppock_date]
    print(f"코폭 지표가 적용된 마지막 날짜: {last_coppock_date_str}")
    print(f"코폭 지표 값 (마지막 날짜): {last_coppock_value:.2f}")
    
    # 2. RSI 계산
    delta = data_monthly['Coppock'].diff()
    gain = (delta.where(delta > 0, 0)).rolling(window=14).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(window=14).mean()
    rs = gain / loss
    data_monthly['RSI'] = 100 - (100 / (1 + rs))
    
    # 3. 스토케스틱 오실레이터 계산
    low_14 = data_monthly['Coppock'].rolling(window=14).min()
    high_14 = data_monthly['Coppock'].rolling(window=14).max()
    data_monthly['Stochastic'] = 100 * (data_monthly['Coppock'] - low_14) / (high_14 - low_14)
    
    # 4. MACD 계산
    data_monthly['MACD'] = data_monthly['Coppock'].ewm(span=12).mean() - data_monthly['Coppock'].ewm(span=26).mean()
    data_monthly['MACD_Signal'] = data_monthly['MACD'].ewm(span=9).mean()
    data_monthly['MACD_Oscillator'] = data_monthly['MACD'] - data_monthly['MACD_Signal']
    
    # 그래프 그리기
    fig, axs = plt.subplots(4, 1, figsize=(14, 24), sharex=True)
    
    # 1. 코폭 지표와 인덱스
    ax1 = axs[0]
    ax2 = ax1.twinx()
    ax1.plot(data_monthly.index, data_monthly['Close'], label=f'{ticker} Close Price', color='green', alpha=0.7)
    ax2.plot(data_monthly.index, data_monthly['Coppock'], label='Coppock Curve', color='purple')
    ax1.set_ylabel('Close Price', fontsize=12)
    ax2.set_ylabel('Coppock Value', fontsize=12)
    ax1.set_title(f'{ticker} Coppock Curve and Index')
    ax1.legend(loc='upper left', fontsize=10)
    ax2.legend(loc='upper left', bbox_to_anchor=(0.3, 1), fontsize=10)
    ax1.grid()
    
    # 2. RSI와 인덱스
    ax1 = axs[1]
    ax2 = ax1.twinx()
    ax1.plot(data_monthly.index, data_monthly['Close'], label=f'{ticker} Close Price', color='green', alpha=0.7)
    ax2.plot(data_monthly.index, data_monthly['RSI'], label='RSI', color='orange')
    ax1.set_ylabel('Close Price', fontsize=12)
    ax2.set_ylabel('RSI Value', fontsize=12)
    ax2.axhline(30, color='red', linestyle='--', label='Oversold (30)')
    ax2.axhline(70, color='red', linestyle='--', label='Overbought (70)')
    ax1.set_title(f'{ticker} RSI and Index')
    ax1.legend(loc='upper left', fontsize=10)
    ax2.legend(loc='upper left', bbox_to_anchor=(0.3, 1), fontsize=10)
    ax1.grid()
    
    # 3. 스토케스틱 오실레이터와 인덱스
    ax1 = axs[2]
    ax2 = ax1.twinx()
    ax1.plot(data_monthly.index, data_monthly['Close'], label=f'{ticker} Close Price', color='green', alpha=0.7)
    ax2.plot(data_monthly.index, data_monthly['Stochastic'], label='Stochastic Oscillator', color='brown')
    ax1.set_ylabel('Close Price', fontsize=12)
    ax2.set_ylabel('Stochastic Value', fontsize=12)
    ax2.axhline(20, color='red', linestyle='--', label='Oversold (20)')
    ax2.axhline(80, color='red', linestyle='--', label='Overbought (80)')
    ax1.set_title(f'{ticker} Stochastic Oscillator and Index')
    ax1.legend(loc='upper left', fontsize=10)
    ax2.legend(loc='upper left', bbox_to_anchor=(0.3, 1), fontsize=10)
    ax1.grid()
    
    # 4. MACD와 인덱스
    ax1 = axs[3]
    ax2 = ax1.twinx()
    ax1.plot(data_monthly.index, data_monthly['Close'], label=f'{ticker} Close Price', color='green', alpha=0.7)
    ax2.plot(data_monthly.index, data_monthly['MACD_Oscillator'], label='MACD Oscillator', color='blue')
    ax1.set_ylabel('Close Price', fontsize=12)
    ax2.set_ylabel('MACD Oscillator Value', fontsize=12)
    ax1.set_title(f'{ticker} MACD Oscillator and Index')
    ax1.legend(loc='upper left', fontsize=10)
    ax2.legend(loc='upper left', bbox_to_anchor=(0.3, 1), fontsize=10)
    ax1.grid()
    
    # X축에 날짜 추가
    for ax in axs:
        ax.tick_params(axis='x', labelrotation=45)
        ax.set_xticks(data_monthly.index[::12])  # 매 12개월 간격으로 주요 날짜 표시
        ax.set_xticklabels([date.strftime('%Y-%m') for date in data_monthly.index[::12]], fontsize=10)
    
    plt.tight_layout()
    plt.figtext(0.5, 0.02, f'Last Data Date: {last_date}, Coppock Applied: {last_coppock_date_str}, Coppock Value: {last_coppock_value:.2f}', ha='center', fontsize=12, color='blue')
    plt.show()
