import yfinance as yf
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from datetime import datetime, timedelta
from scipy.stats import percentileofscore

# 사용자로부터 두 개의 종목 티커 입력받기
symbol1 = input("Enter the first stock symbol: ").upper()
symbol2 = input("Enter the second stock symbol: ").upper()

# 각 티커에 대한 회사 이름 가져오기
company_name1 = yf.Ticker(symbol1).info['longName']
company_name2 = yf.Ticker(symbol2).info['longName']

# 오늘 날짜와 3년 전 날짜 계산
end_date = datetime.now()
start_date = end_date - timedelta(days=3*365)

# 3년치 주가 데이터를 yfinance로 가져오기 (수정 종가 사용)
data1 = yf.download(symbol1, start=start_date, end=end_date)['Close']
data2 = yf.download(symbol2, start=start_date, end=end_date)['Close']

# 데이터가 제대로 다운로드되었는지 확인
if data1.empty or data2.empty:
    print(f"Failed to download data for {symbol1} ({company_name1}) or {symbol2} ({company_name2}). Please check the symbols and try again.")
else:
    # 두 주식의 공통된 날짜만 남기기 (inner join)
    merged_data = pd.concat([data1, data2], axis=1, join="inner")
    merged_data.columns = [symbol1, symbol2]  # 각 주식의 심볼로 열 이름 지정

    # 로그 변환하여 가격 차이 계산 (로그 스프레드)
    log_data1 = np.log(merged_data[symbol1])
    log_data2 = np.log(merged_data[symbol2])
    log_spread = log_data1 - log_data2

    # 1표준편차와 2표준편차 계산
    std1 = log_spread.std()
    std2 = 2 * std1
    mean_spread = log_spread.mean()

    # 상관관계 계산
    correlation = merged_data[symbol1].pct_change().corr(merged_data[symbol2].pct_change())
    
    # 상관관계 값 출력
    print(f"Correlation between {company_name1} ({symbol1}) and {company_name2} ({symbol2}): {correlation:.4f}")

    # 로그 스프레드 그래프 시각화 (표준편차 포함)
    plt.figure(figsize=(14, 10))

    # 로그 스프레드 꺾은선 그래프
    plt.subplot(4, 1, 1)
    plt.plot(log_spread, label=f'Log Spread ({company_name1} - {company_name2})', color='blue')

    # 1표준편차 및 2표준편차 추가
    plt.axhline(mean_spread + std1, color='green', linestyle='--', label='+1 Std Dev')
    plt.axhline(mean_spread - std1, color='green', linestyle='--', label='-1 Std Dev')
    plt.axhline(mean_spread + std2, color='red', linestyle='--', label='+2 Std Dev')
    plt.axhline(mean_spread - std2, color='red', linestyle='--', label='-2 Std Dev')

    # 제목과 레이블 설정
    plt.title(f'Log Spread Between {company_name1} ({symbol1}) and {company_name2} ({symbol2}) with Standard Deviations')
    plt.xlabel('Date')
    plt.ylabel('Log Spread')
    plt.legend()
    plt.grid(True)

    # 기본 퍼센타일 계산
    basic_percentiles = log_spread.rolling(window=60).apply(lambda x: percentileofscore(x, x[-1]))

    # 지수 이동 평균(EMA) 적용 (기간 설정 예: 60일)
    price_diff_ema = log_spread.ewm(span=60, adjust=False).mean()

    # EMA 가공된 퍼센타일 계산
    ema_percentiles = price_diff_ema.rolling(window=60).apply(lambda x: percentileofscore(x, x[-1]))

    # 기본 퍼센타일 그래프
    plt.subplot(4, 1, 2)
    plt.plot(basic_percentiles, label=f'Basic Percentile of Log Spread ({symbol1} - {symbol2})', color='blue')
    plt.title(f'Basic Percentile of Log Price Difference Between {company_name1} and {company_name2}')
    plt.xlabel('Date')
    plt.ylabel('Percentile')
    plt.legend()
    plt.grid(True)

    # EMA 가공된 퍼센타일 그래프
    plt.subplot(4, 1, 3)
    plt.plot(ema_percentiles, label=f'EMA Percentile of Log Spread ({symbol1} - {symbol2})', color='orange')
    plt.title(f'EMA Percentile of Log Price Difference Between {company_name1} and {company_name2}')
    plt.xlabel('Date')
    plt.ylabel('Percentile')
    plt.legend()
    plt.grid(True)

    # 누적 수익률 계산 (첫날을 기준으로 한 누적 수익률)
    cumulative_returns1 = (merged_data[symbol1].pct_change() + 1).cumprod() - 1
    cumulative_returns2 = (merged_data[symbol2].pct_change() + 1).cumprod() - 1

    # 누적 수익률 그래프
    plt.subplot(4, 1, 4)
    plt.plot(cumulative_returns1, label=f'{company_name1} ({symbol1}) Cumulative Return', color='green')
    plt.plot(cumulative_returns2, label=f'{company_name2} ({symbol2}) Cumulative Return', color='red')
    plt.title('Cumulative Returns Comparison')
    plt.xlabel('Date')
    plt.ylabel('Cumulative Return')
    plt.legend()
    plt.grid(True)

    plt.tight_layout()
    plt.show()
