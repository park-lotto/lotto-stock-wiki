import yfinance as yf
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime, timedelta

# -----------------------------------------------------------
# 1. 데이터 다운로드 (월봉)
# -----------------------------------------------------------
tickers = ["QQQ", "SOXX", "QLD", "PSQ", "GLD", "SPMO", "TIP"]  # ✅ TIP 추가!

today = datetime.today()
first_day_current_month = today.replace(day=1)
first_day_next_month = (first_day_current_month + timedelta(days=32)).replace(day=1)

if today < first_day_next_month:
    last_complete_month = first_day_current_month - timedelta(days=1)
else:
    last_complete_month = today

end_date = last_complete_month.strftime('%Y-%m-%d')
start_date = "2010-01-01"

print(f"데이터 다운로드 기간: {start_date} ~ {end_date}")

df = yf.download(
    tickers=tickers,
    start=start_date,
    end=end_date,
    interval="1mo"
)

# -----------------------------------------------------------
# 2. 종가만 추출 및 멀티 인덱스 처리
# -----------------------------------------------------------
df_close = df["Close"].dropna(how="all")

if isinstance(df_close.columns, pd.MultiIndex):
    df_close.columns = df_close.columns.droplevel(0)

print("\ndf_close.columns =>", df_close.columns.tolist())

# -----------------------------------------------------------
# 3. QQQ + TIP 둘 다 6개월 이동평균선 위에 있을 때 상승추세
# -----------------------------------------------------------
required_tickers = ["QQQ", "TIP"]
for t in required_tickers:
    if t not in df_close.columns:
        raise ValueError(f"{t} 컬럼이 없습니다. 야후 파이낸스에서 {t} 데이터를 불러오지 못했습니다.")

qqq = df_close["QQQ"].dropna()
tip = df_close["TIP"].dropna()

qqq_ma6 = qqq.rolling(window=6).mean()
tip_ma6 = tip.rolling(window=6).mean()

uptrend_qqq = qqq > qqq_ma6
uptrend_tip = tip > tip_ma6

# ✅ 둘 다 위에 있어야 상승 추세
uptrend = uptrend_qqq & uptrend_tip

# -----------------------------------------------------------
# 4. 최근 투자 판단
# -----------------------------------------------------------
last_date = uptrend.dropna().index[-1]
is_uptrend = uptrend.loc[last_date]

print(f"\n=== {last_date.strftime('%Y-%m')} 기준 추세 판단 ===")
if is_uptrend:
    print("🚀 상승추세 → 다음 달에는 QQQ 60%, QLD 15%, SOXX 25% 투자")
else:
    print("📉 하락추세 → 다음 달에는 SPMO 50%, PSQ 50% 투자")

# -----------------------------------------------------------
# 5. 백테스트
# -----------------------------------------------------------
monthly_returns = df_close.pct_change().dropna()
uptrend_signal = uptrend.shift(1).dropna()
weights = pd.DataFrame(0.0, index=monthly_returns.index, columns=monthly_returns.columns)

for date in weights.index:
    if date not in uptrend_signal.index:
        continue

    if uptrend_signal.loc[date]:
        # 상승추세 → QQQ 60%, QLD 15%, SOXX 25%
        if "QQQ" in weights.columns:
            weights.loc[date, "QQQ"] = 0.60
        if "QLD" in weights.columns:
            weights.loc[date, "QLD"] = 0.15
        if "SOXX" in weights.columns:
            weights.loc[date, "SOXX"] = 0.25
    else:
        # 하락추세 → SPMO 50%, PSQ 50%
        if "SPMO" in weights.columns:
            weights.loc[date, "SPMO"] = 0.50
        if "PSQ" in weights.columns:
            weights.loc[date, "PSQ"] = 0.50

shifted_weights = weights.shift(1).dropna()
portfolio_returns = (shifted_weights * monthly_returns).sum(axis=1)
cumulative_returns = (1 + portfolio_returns).cumprod()

# 성과 지표
final_value = cumulative_returns.iloc[-1]
total_months = len(cumulative_returns)
years = total_months / 12
cagr = final_value**(1/years) - 1 if years > 0 else np.nan

cummax = cumulative_returns.cummax()
drawdown = cumulative_returns / cummax - 1
mdd = drawdown.min()

print("\n===== 백테스트 성과 (QQQ+TIP 기반 트렌드 전략) =====")
print(f"최종 누적수익률: {final_value:.2f} 배")
print(f"CAGR(연평균수익률): {cagr:.2%}")
print(f"MDD(최대낙폭): {mdd:.2%}")

# -----------------------------------------------------------
# 6. 그래프 시각화
# -----------------------------------------------------------
plt.figure(figsize=(12,6))
plt.plot(cumulative_returns, label="Trend Strategy")
plt.title("Trend-Following Strategy (QQQ & TIP Above 6-mo MA)")
plt.xlabel("Date")
plt.ylabel("Cumulative Returns")
plt.grid(True)
plt.legend()
plt.show()

# 벤치마크(QQQ Buy & Hold) 비교
if "QQQ" in monthly_returns.columns:
    qqq_returns = monthly_returns["QQQ"].dropna()
    qqq_cum = (1 + qqq_returns).cumprod()

    plt.figure(figsize=(12,6))
    plt.plot(cumulative_returns, label="Strategy (Trend Following)")
    plt.plot(qqq_cum, label="QQQ Buy & Hold", linestyle="--")
    plt.title("Strategy vs QQQ Buy & Hold")
    plt.xlabel("Date")
    plt.ylabel("Cumulative Returns")
    plt.grid(True)
    plt.legend()
    plt.show()
