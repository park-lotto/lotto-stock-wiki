import pandas as pd
import yfinance as yf
from datetime import datetime, timedelta

# -------------------- 1. 사용자 입력 --------------------
specific_date_str = input("한국 기준 투자일 입력 (YYYY-MM-DD): ")
try:
    kr_date = datetime.strptime(specific_date_str, "%Y-%m-%d")
except ValueError:
    print("날짜 형식이 잘못되었습니다.")
    raise SystemExit

us_close_date = (kr_date - timedelta(days=1)).strftime('%Y-%m-%d')
kr_date_str = kr_date.strftime('%Y-%m-%d')

# -------------------- 2. 티커 설정 --------------------
risky_assets = {
    "133690.KS": "TIGER 미국나스닥100",
    "448290.KS": "TIGER 미국S&P500",
    "458730.KS": "TIGER 미국배당다우존스",
    "101280.KS": "KODEX 일본TOPIX100",
    "251350.KS": "KODEX 선진국MSCI World",
    "474800.KS": "KOSEF 미국원유에너지기업",
    "182480.KS": "TIGER 미국MSCI리츠(합성 H)",
    "229200.KS": "KODEX 코스닥 150",
    "069500.KS": "KODEX 200",
}
safe_assets = {
    "476760.KS": "ACE 미국30년국채액티브",
    "302190.KS": "TIGER 중장기국채",
    "148070.KS": "KOSEF 국고채10년",
    "329750.KS": "TIGER 미국달러단기채권액티브",
    "130730.KS": "KOSEF 단기자금",
    "411060.KS": "ACE KRX금현물"
}
canary_assets = {"TIP": "TIP", "QQQ": "QQQ"}
fixed_assets = {
    "448290.KS": 10,
    "133690.KS": 10,
    "411060.KS": 5,
    "474800.KS": 5,
}

# -------------------- 3. 데이터 다운로드 --------------------
tickers = list(set(risky_assets) | set(safe_assets) | set(canary_assets) | set(fixed_assets))
start_date = (kr_date - pd.DateOffset(years=3)).strftime('%Y-%m-%d')

print(f"\n[INFO] 데이터 다운로드 중: {start_date} ~ {kr_date_str}")
data = yf.download(tickers, start=start_date, end=kr_date_str)['Close']
data = data.dropna(how='all')

# -------------------- 4. 미국 종목은 하루 전까지만 사용 --------------------
for us_ticker in canary_assets:
    if us_ticker in data.columns:
        data[us_ticker] = data[us_ticker].loc[:us_close_date]

# -------------------- 5. 모멘텀 계산 함수 --------------------
def calculate_momentum(series):
    df = pd.DataFrame(index=series.index)
    df['1M'] = series.pct_change(21)
    df['3M'] = series.pct_change(63)
    df['6M'] = series.pct_change(126)
    df['12M'] = series.pct_change(252)
    return df.mean(axis=1)

# -------------------- 6. 카나리아 신호 (TIP + QQQ 모두 양수) --------------------
def canary_momentum(tip, qqq):
    tip_mom = calculate_momentum(tip)
    qqq_mom = calculate_momentum(qqq)
    signal = (tip_mom > 0) & (qqq_mom > 0)
    return signal, tip_mom, qqq_mom

# -------------------- 7. 자산 선택 함수 --------------------
def select_safe_assets(momentum_df, bil_mom, top_n=2):
    recent = momentum_df[safe_assets.keys()].iloc[-1]
    top_safe = recent.nlargest(top_n)
    selected = [a for a in top_safe.index if top_safe[a] > bil_mom]
    if not selected:
        return ['130730.KS']
    if '130730.KS' not in selected:
        selected.append('130730.KS')
    return selected[:top_n]

def select_top_assets(momentum_df, canary_signal, top_n=4):
    if not canary_signal.iloc[-1]:
        bil_mom = calculate_momentum(data['130730.KS']).iloc[-1]
        return select_safe_assets(momentum_df, bil_mom, top_n)
    else:
        return momentum_df[risky_assets.keys()].iloc[-1].nlargest(top_n).index.tolist()

def normalize_weights(weights_dict, total=100):
    rounded = {k: round(v) for k, v in weights_dict.items()}
    diff = total - sum(rounded.values())
    if diff != 0:
        max_key = max(rounded, key=rounded.get)
        rounded[max_key] += diff
    return rounded

# -------------------- 8. 실행 --------------------
momentum_data = data.apply(calculate_momentum)

if not all(k in data.columns for k in canary_assets):
    print("\n[오류] TIP 또는 QQQ 데이터 부족 → 안전자산 모드로 전환")
    canary_signal = pd.Series([False], index=[data.index[-1]])
    top_assets = ['130730.KS']
    tip_mom, qqq_mom = pd.Series([0]), pd.Series([0])
else:
    canary_signal, tip_mom, qqq_mom = canary_momentum(data['TIP'], data['QQQ'])
    top_assets = select_top_assets(momentum_data, canary_signal, top_n=4)

# -------------------- 9. 비중 계산 --------------------
haa_share = 65
share_each = haa_share / len(top_assets)
combined = dict(fixed_assets)
for asset in top_assets:
    combined[asset] = combined.get(asset, 0) + share_each
final_weights = normalize_weights(combined)
sorted_weights = sorted(final_weights.items(), key=lambda x: x[1], reverse=True)

# -------------------- 10. 출력 --------------------
print(f"\n✅ {kr_date_str} 기준 HAA 추천 포트폴리오")
print(f"📉 TIP 모멘텀: {tip_mom.iloc[-1]:.2%}")
print(f"📉 QQQ 모멘텀: {qqq_mom.iloc[-1]:.2%}")
print(f"📌 카나리아 신호 (TIP + QQQ 양수?): {canary_signal.iloc[-1]}")
print(f"📌 선택된 변동 자산: {', '.join(top_assets)}\n")

print("📊 최종 자산 구성:")
for asset, weight in sorted_weights:
    name = risky_assets.get(asset) or safe_assets.get(asset) or canary_assets.get(asset) or asset
    print(f"  {asset:<12s} ({name}): {weight}%")

print("\n[완료]")
