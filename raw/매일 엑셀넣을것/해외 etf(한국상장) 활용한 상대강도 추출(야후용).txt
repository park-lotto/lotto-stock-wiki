# ETF 데이터를 다운로드 (수정주가, 기간: 1년)
data = yf.download(list(korea_industry_etfs.keys()), period="1y")['Close']
data = data.sort_index()  # 날짜를 기준으로 정렬

# 변동성 조정 수익률 계산 함수
def calculate_risk_adjusted_momentum(data, period_3m=63, period_6m=126, period_12m=252, last_n_days=10):
    daily_top_15_industries = {}

    # 최근 N 거래일 가져오기
    trading_days = data.index[-last_n_days:]

    for day in trading_days:
        end_date_index = data.index.get_loc(day)

        # 각 기간별 수익률 및 표준편차 계산
        start_3m = max(0, end_date_index - period_3m + 1)
        start_6m = max(0, end_date_index - period_6m + 1)
        start_12m = max(0, end_date_index - period_12m + 1)

        returns_3m = data.iloc[start_3m:end_date_index + 1].pct_change(fill_method=None).dropna().mean()
        returns_6m = data.iloc[start_6m:end_date_index + 1].pct_change(fill_method=None).dropna().mean()
        returns_12m = data.iloc[start_12m:end_date_index + 1].pct_change(fill_method=None).dropna().mean()

        # 표준편차 계산
        vol_3m = data.iloc[start_3m:end_date_index + 1].pct_change(fill_method=None).dropna().std()
        vol_6m = data.iloc[start_6m:end_date_index + 1].pct_change(fill_method=None).dropna().std()
        vol_12m = data.iloc[start_12m:end_date_index + 1].pct_change(fill_method=None).dropna().std()

        # 변동성 조정 수익률 계산 (수익률 / 표준편차)
        risk_adjusted_returns_3m = returns_3m / vol_3m
        risk_adjusted_returns_6m = returns_6m / vol_6m
        risk_adjusted_returns_12m = returns_12m / vol_12m

        # 3개월, 6개월, 12개월 변동성 조정 수익률의 평균
        average_risk_adjusted_momentum = (risk_adjusted_returns_3m + risk_adjusted_returns_6m + risk_adjusted_returns_12m) / 3

        # 상대강도 상위 15개 산업 추출
        top_industries = average_risk_adjusted_momentum.nlargest(15).index.tolist()

        # 티커를 업종명으로 매핑
        top_industry_names = [f"{ticker} ({korea_industry_etfs[ticker]})" for ticker in top_industries]

        # 해당 날짜에 대해 저장
        daily_top_15_industries[day] = top_industry_names

    # 딕셔너리를 DataFrame으로 변환
    daily_top_15_industries_df = pd.DataFrame.from_dict(daily_top_15_industries, orient='index', columns=[f'Top {i+1}' for i in range(15)])
    daily_top_15_industries_df.index.name = 'Date'
    return daily_top_15_industries_df

# 변동성 조정 모멘텀 기준 상대강도 상위 업종 계산
result_df = calculate_risk_adjusted_momentum(data, period_3m=63, period_6m=126, period_12m=252, last_n_days=10)
